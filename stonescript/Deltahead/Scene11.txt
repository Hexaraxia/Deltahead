func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(p, layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  p.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(p, x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  p.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// canvases max out at 100x27, so screen-sized ones are tiled
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(root, i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

// a light: additive canvas fading out from the center, filled a few hundred cells a frame.
// a light that's off waits off screen
func QueueHalo(p, w, h, r, g, b, bgk)
  var c = MakeCanvas(p, OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

// a light that's off goes off screen with no width (the game draws every canvas, hidden or not)
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

// the O on frame i of a shot, and the streak it leaves (the arrays are whole cells)
func Head(h, s, x, y, k)
  Show(h, true)
  h.x = SX(x)
  h.y = SY(y)
  Show(s, k > 0)
  s.x = h.x - 2
  s.y = h.y - 2
  s.frame = k

func Puff(a, x, y, u)
  Show(a, u >= 0 & u < 16)
  a.x = SX(x) - 6
  a.y = SY(y) - 3
  a.frame = IMin(3, IMax(0, u) / 4)

func Dust(a, x, y, u)
  Show(a, u >= 0 & u < 12)
  a.x = SX(x) - 4
  a.y = SY(y) - 2
  a.frame = IMin(5, IMax(0, u) / 2)

// up off the shoulder (he stays out of the picture), out over the snow, a bounce to the end, over
// the lip and down into the cloud. the camera follows a whole cell at a time, never back
func ShotA(t)
  var n = fall.aX.Count()
  var i = IMax(0, IMin(t, n - 1))
  var ox = fall.aX[i]
  var oy = fall.aY[i]
  var y0 = 24 - (cy + 3)
  camX = IMin(46 - W, ox - cx)
  camY = IMin(fall.cloudRow - (cy + 1), IMax(y0, fall.aYmax[i] - (cy - 2)))
  var dy = camY - y0
  var sea = cy - 3
  aSky.x = -60 - math.RoundToInt(camX * 0.05)
  aSky.y = -20 - math.RoundToInt(dy * 0.05)
  aMoon.y = 2 - math.RoundToInt(dy * 0.05)
  var af = (f / 6) % 8
  var cols = ["#136328", "#1f9a40", "#32da5a"]
  for k = 0 .. 2
    var a = aAurora[k]
    a.x = -20 - math.RoundToInt(camX * 0.08)
    a.y = lb - math.RoundToInt(dy * 0.08)
    a.frame = af
    a.color = cols[k]
  aFar.x = -40 - math.RoundToInt(camX * 0.25)
  aFar.y = sea - hts.farSea - math.RoundToInt(dy * 0.25)
  aSummit.x = 14 - math.RoundToInt(camX * 0.3)
  aSummit.y = cy - 14 - math.RoundToInt(dy * 0.3)
  aBack.x = -90 - math.RoundToInt(camX * 0.6)
  aBack.y = sea + 2 - math.RoundToInt((dy - 1) * 0.6)
  // mist drifts the way the camera pans, like the cloud, or the two would wobble it
  aMist.x = -math.RoundToInt(Wrap(30 - f / 20.0 + camX * 0.7, 360))
  aMist.y = sea + 3 - math.RoundToInt(dy * 0.7)
  aMist2.x = aMist.x + 360
  aMist2.y = aMist.y
  aFieldA.x = SX(0)
  aFieldA.y = SY(0)
  aFieldB.x = SX(120)
  aFieldB.y = SY(0)
  aEdge.x = SX(fall.edgeX)
  aEdge.y = SY(0)
  // drifting the way the camera pans, or the two would wobble it
  aCloud.x = SX(-200 + f * 0.015)
  aCloud.y = SY(fall.cloudRow)
  aNear.x = -300 - math.RoundToInt(camX * 1.5)
  aNear.y = H - 12 - math.RoundToInt(dy * 1.5)
  for k = 0 .. 9
    var fl = aFlakes[k]
    fl.x = math.RoundToInt(Wrap((k * 53) % 97 - f * 0.25 + math.Sin(f * 0.05 + k) * 1.5, W + 4)) - 2
    fl.y = math.RoundToInt(Wrap((k * 37) % 33 + f * 0.18, H + 4)) - 2
    fl.frame = k % 3
    fl.color = "#9aa6ac"
  Light(aGlow, jobs.Count() = 0, cx, lb + 5 - math.RoundToInt(dy * 0.08))
  Show(aHead, t >= 0)
  Show(aStreak, false)
  ?t >= 0 & t < n
    Head(aHead, aStreak, ox, oy, fall.aK[i])
  :?t >= n
    Show(aHead, false)
  for k = 0 .. 1
    var h = fall.aHits[k]
    Puff(aPuffs[k], fall.aX[h], fall.aY[h], t - h)

// from far off like scene 6: a speck going back down the trail, the view following a row at a time
func ShotB(t)
  var n = fall.bX.Count()
  var i = IMax(0, IMin(t, n - 1))
  var mx = mtn.peakX - cx
  var my = IMax(0, fall.bY[i] - (cy - 4))
  bMtn.x = -mx
  bMtn.y = -my
  bClouds.x = -mx - 40 + f / 25
  bClouds.y = mtn.cloudY - my
  bSpeck.x = fall.bX[i] - mx
  bSpeck.y = fall.bY[i] - my
  // once it stops it fades out while the shot holds, its light going halfway
  var k = math.Clamp((t - n + 1) * 1.0 / IMax(1, fall.sC - fall.sB - n), 0.0, 1.0)
  bSpeck.color = color.Lerp(fall.speck[0][0], "#000000", k)
  Light(bGlow, jobs.Count() = 0 & k < 0.5, bSpeck.x, bSpeck.y)

  cSky.y = -math.RoundToInt((camY - summitCam) * 0.3)
  for k = 0 .. 5
    var b = cFogBack[IMin(2, k)]
    var kk = 0.8
    ?k >= 3
      b = cFogFront[k - 3]
      kk = 1.25
    var y = math.RoundToInt(Wrap(FOG_Y[k] - camY * kk, H + 6)) - 3
    b.x = math.RoundToInt(Wrap(FOG_X[k] + f * FOG_V[k], W + 70)) - 70
    b.y = y
    Show(b, camY + y > climb.crestRow + 8)
  for k = 0 .. 13
    var s = cFlakes[k]
    // snow hangs still as the camera goes down, falling the other way would wobble against it
    var y = math.RoundToInt(Wrap((k * 37) % 33 - camY, H + 4)) - 2
    s.x = math.RoundToInt(Wrap((k * 53) % 97 - f * 0.12 + math.Sin(f * 0.05 + k) * 1.5, W + 4)) - 2
    s.y = y
    s.frame = k % 3
    Show(s, camY + y < 92 & camY + y > climb.crestRow + 3)
  cTop.x = SX(0)
  cTop.y = SY(0)
  cBot.x = SX(0)
  cBot.y = SY(90)
  cGone.x = SX(climb.goneX)
  cGone.y = SY(climb.goneY)
  cFront.x = SX(0)
  cFront.y = SY(0)
  cNear.x = SX(0)
  cNear.y = -math.RoundToInt(camY * 1.3)
  Show(cHead, t < n - 1)
  Show(cStreak, false)
  ?t < n - 1
    Head(cHead, cStreak, fall.cX[i], fall.cY[i], fall.cK[i])
  for k = 0 .. 1
    var u = 99
    var h = 0
    for j = 0 .. fall.cHits.Count() - 1
      ?j % 2 = k & t >= fall.cHits[j]
        h = fall.cHits[j]
        u = t - h
    Puff(cPuffs[k], fall.cX[h], fall.cY[h], u)

// down past the crystal into the pool like his first fall, up once and under, then one mote of Ki comes up
func ShotE(t)
  var n = fall.eX.Count()
  var i = IMax(0, IMin(t, n - 1))
  var lit = jobs.Count() = 0
  camX = IMax(0, fount.landX - (cx - 6))
  camY = IMax(0, IMin(fount.floor - (cy + 7), fall.eY[i] - (cy - 2)))
  eFar.x = -math.RoundToInt(camX * 0.6)
  eFar.y = -math.RoundToInt(camY * 0.6)
  eMidL.x = SX(0)
  eMidL.y = SY(0)
  eMidR.x = SX(fount.midSplit)
  eMidR.y = SY(0)
  ePool.x = SX(0)
  ePool.y = SY(0)
  eFront.x = -math.RoundToInt(camX * 1.4)
  eFront.y = SY(0)
  // the crystal: its light, the Ki going up inside it, the glints
  var x = fount.crystalX
  var tip = fount.crystalTip
  Light(eGlowLow, lit, SX(x - 4), SY(76))
  Light(eGlowHigh, lit, SX(x - 2), SY(54))
  Light(eTipGlow, lit, SX(x), SY(tip + 2))
  Light(ePoolGlow, lit, SX(54), SY(fount.floor))
  for k = 0 .. 2
    var s = eSurges[k]
    var age = (f + k * 37) % 110
    var y = fount.backEdge - 3 - age * 0.5
    Show(s, SURGE_X[k] > x - (y - tip))
    s.x = SX(SURGE_X[k])
    s.y = SY(y)
  for k = 0 .. eGlints.Count() - 1
    var g = eGlints[k]
    var ph = (f / 4 + k * 9) % 29
    g.frame = 1
    ?ph = 1
      g.frame = 2
    :?ph < 3
      g.frame = 0
    g.x = SX(fount.glintX[k])
    g.y = SY(fount.glintY[k])
  Show(eHead, t < fall.eSink)
  Show(eStreak, false)
  ?t < fall.eSink
    Head(eHead, eStreak, fall.eX[i], fall.eY[i], fall.eK[i])
  // the splash and the rings it leaves
  var b = t - fall.eLand
  Show(eSplash, b >= 0 & b < 20)
  eSplash.frame = IMax(0, b) / 4
  eSplash.x = SX(fount.landX) - 4
  eSplash.y = SY(fount.floor) - 4
  for k = 0 .. 1
    var a = b - 4 - k * 9
    var r = eRings[k]
    Show(r, a >= 0 & a < 24)
    r.frame = IMax(0, a) / 4
    r.x = SX(fount.landX) - 4
    r.y = SY(fount.floor - 1 + k * 2)
  Light(eSplashGlow, lit & b >= 0 & b < 12, SX(fount.landX), SY(fount.floor))
  // one mote coming up where it went under
  var m = t - fall.eSink - 6
  Show(eMote, m >= 0)
  eMote.x = SX(fount.landX)
  eMote.y = SY(fount.floor - 1 - m * 0.12)
  eMote.frame = (f / 6) % 3
  Light(eMoteGlow, lit & m >= 0, eMote.x, eMote.y)

// ---------------------------------------------------------------- the stars

// the game's way of writing a star rating
func Stars(n)
  ?n <= 0
    return ""
  ?n = 1
    return "☆"
  ?n = 2
    return "☆☆"
  ?n = 3
    return "☆☆☆"
  return "☆" + n + "☆"

func TierColor(n)
  ?n <= 5
    return "#ffffff"
  ?n <= 10
    return "#7fe3de"
  ?n <= 15
    return "#ffff62"
  return "#32da5a"

// each place's name and stars like the banner on the way up, counting down to none, a falling note each
func Banner()
  var k = -1
  for j = 0 .. 3
    ?f >= BN_ON[j] & f < BN_OFF[j] + 15
      k = j
  Show(nameText, k >= 0)
  Show(starText, k >= 0)
  ?k < 0
    return
  var count = BN_FROM[k] - BN_TO[k]
  var gone = 0
  ?f >= BN_TICK[k]
    gone = IMin(count, (f - BN_TICK[k]) / BN_STEP[k] + 1)
  var n = BN_FROM[k] - gone
  var a = Prog(BN_ON[k], BN_ON[k] + 12) - Prog(BN_OFF[k], BN_OFF[k] + 15)
  nameText.text = BN_NAME[k]
  nameText.w = string.Size(BN_NAME[k]) + 4
  nameText.color = color.Lerp("#000000", "#e7e3da", a)
  starText.text = Stars(n)
  starText.w = string.Size(Stars(n)) + 4
  starText.color = color.Lerp("#000000", TierColor(n), a)
  ?f >= BN_TICK[k] & (f - BN_TICK[k]) % BN_STEP[k] = 0 & (f - BN_TICK[k]) / BN_STEP[k] < count
    play progress_@IMin(9, count - (f - BN_TICK[k]) / BN_STEP[k])@

// ---------------------------------------------------------------- the frame

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  var s = 0
  ?f >= fall.sB
    s = 1
  ?f >= fall.sC
    s = 2
  ?f >= fall.sD
    s = 3
  ?f >= fall.sE
    s = 4
  ?f >= fall.sF
    s = 5
  for k = 0 .. 5
    shots[k].x = OFF
  shots[s].x = 0
  ?s = 0
    ShotA(f - fall.aOn)
  :?s = 1
    ShotB(f - fall.sB)
  :?s = 2
    ShotC(f - fall.sC)
  :?s = 3
    ShotD(f - fall.sD)
  :?s = 4
    ShotE(f - fall.sE)
  :
    ShotF(f - fall.fOn)
  Banner()
  // the dark at the start, before the Ring, and at the end
  ?f < fall.aOn | (f >= fall.sF & f < fall.fOn)
    SetShade("#000000")
  :?f >= fall.fOn
    SetShade(color.Lerp("#000000", GRADES[5], Prog(fall.fOn, fall.fOn + 20)))
  :
    SetShade(GRADES[s])
  Sfx()


// ---------------------------------------------------------------- the film calls these

// a scene built ahead waits off to the side, but the game still draws its canvases,
// so they get no width till it's on
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built near the end of the scene before so its lights are ready, then waits off to
// the side (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // nothing else on the frame it's built in, that's heavy enough
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

func Reset()
  f = 0
  over = false
