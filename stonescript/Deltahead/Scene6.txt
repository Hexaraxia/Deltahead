// ---------------------------------------------------------------- building

func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// canvases max out at 100x27, so screen-sized ones are tiled
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

func MakeText(s)
  var t = ui.AddText(s)
  root.Add(t)
  t.anchor = "top_left"
  t.dock = "top_left"
  t.x = 0
  t.y = 0
  t.w = string.Size(s) + 2
  t.h = 1
  t.align = "center"
  t.color = "#000000"
  return t

// a light: additive canvas fading out from the center, filled a few hundred cells a frame.
// a light that's off waits off screen
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// a light that's off goes off screen with no width (the game draws every canvas, hidden or not)
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

// Puts a mine layer where the camera sees it; par < 1 is farther away
func Layer(c, par, ox, oy)
  c.x = ox - camX0 - math.RoundToInt((camX - camX0) * par)
  c.y = oy - camY0 - math.RoundToInt((camY - camY0) * par)

func SX(x)
  return x - camX

func SY(y)
  return y - camY

// the Pushers going by with the camera drifting along, each line typed over whoever says it.
// the Controller at the back stops, cracks his whip, walks on
func Mine(lit)
  camX = camX0 - 20 + f * 2 / 30
  camY = camY0
  Layer(dFar, 0.5, 0, 0)
  Layer(dBack, 0.75, 0, 0)
  Layer(dMain, 1.0, 0, 0)
  Layer(dFore, 1.3, 0, camY0 + 2)
  var shimmer = (f / 12) % 2
  Layer(liqA, 1.3, -6 + shimmer, 27)
  Layer(liqB, 1.3, -6 - (f / 16) % 2, 29)
  for i = 0 .. 3
    Layer(bubbles[i], 1.3, 60 + i * 17 + (i % 2) * 5, 26)
    bubbles[i].frame = (f / 4 + i * 7) % 10
    var pos = (f + i * 28) % 40
    Show(puffs[i], pos >= 20)
    Layer(puffs[i], 1.3, 59 + i * 17 + (i % 2) * 5, 21)
    puffs[i].frame = IMax(0, pos - 20) / 4
  for i = 0 .. 2
    Layer(drips[i], 1.3, 70 + i * 19, camY0 + 4)
    drips[i].frame = IMin(8, ((f + i * 23) % 70) / 3)
  for i = 0 .. 1
    var lf = (f / 5 + i * 3) % 4
    lampFlames[i].frame = lf
    lampFlames[i].x = SX(mine.lampX[i])
    lampFlames[i].y = SY(mine.lampY[i])
    Light(lampHalo[i * 2], lit & lf < 3, SX(mine.lampX[i]), SY(mine.lampY[i]))
    Light(lampHalo[i * 2 + 1], lit & lf = 3, SX(mine.lampX[i]), SY(mine.lampY[i]))
  // the pools' heat on everything near them, flickering with the boil
  for i = 0 .. 1
    var px = 18 + i * 92 - camX0 - math.RoundToInt((camX - camX0) * 1.3)
    var py = 29 - camY0
    var pd = (f / 5 + i * 2) % 3 = 0
    Light(poolHalo[i * 2], lit & !pd, px, py)
    Light(poolHalo[i * 2 + 1], lit & pd, px, py)
  // and embers off them, going up through the mine
  for i = 0 .. 5
    var e = mineEmbers[i]
    var life = 70 + (i * 17) % 40
    var age = (f + i * 31) % life
    Show(e, true)
    e.frame = (f / 3 + i) % 4
    e.x = 8 + (i * 29) % (W - 16) + math.RoundToInt(math.Sin(f * 0.07 + i * 1.7) * 2)
    e.y = H - 4 - age * (H - 6) / life
    ?age < 4 | age > life - 6
      e.color = "#e0461c"
    :
      e.color = "#ffb35c"
  // the line of Pushers and, behind them, their Controller
  for i = 0 .. N_PUSHERS - 1
    var w = walkers[i]
    Show(w, true)
    w.x = SX(70 + f * 4 / 30 - i * 14)
    w.y = SY(19)
    w.frame = (f / 6 + i) % 4
  Show(ctrl, true)
  var lt = f - T_CRACK
  var held = IMax(0, IMin(lt, 30))
  ctrl.x = SX(70 + f * 4 / 30 - N_PUSHERS * 14 - 2 - held * 4 / 30) + 8 - spr.ctrlHeadR
  ctrl.y = SY(19) - 2
  ctrl.frame = 17 + (f / 6 + 2) % 4
  ?lt >= 0 & lt < 30
    ctrl.frame = 21 + lt / 3
  :?lt >= 30 & lt < 39
    ctrl.frame = 8 + (lt - 30)
  // the other line, further back, going the other way
  for i = 0 .. 2
    var bw = backWalkers[i]
    Show(bw, true)
    Layer(bw, 0.75, 170 - f * 12 / 100 + i * 15, mine.backTrack - 3)
    bw.frame = (f / 7 + i) % 4
  // what they say
  for k = 0 .. LINES.Count() - 1
    var on = f >= LINE_AT[k] & f < LINE_END[k]
    ?on
      var n = IMin(string.Size(LINES[k]), (f - LINE_AT[k]) * 6 / 5 + 1)
      var who = walkers[LINE_WHO[k]]
      bub.Show(speech[k], who.x + HEAD_X, who.y, n, W, "#f0f0f0")
    :
      bub.Hide(speech[k])
    ?f = LINE_AT[k]
      play slave_npc @104 - LINE_WHO[k] * 7@

// the mountain from foot to top: mine lamps, Pushers going in, cloud, and him at the top. then
// on up past it into the sky, which doesn't move (the stars and moon are that far off)
func Mountain(lit)
  var mx = mtn.peakX - cx
  var myf = (mtn.h - H) * (1.0 - Ease(Prog(T_PAN, T_PAN + PAN_T))) - UP * Ease(Prog(T_UP, T_UP + UP_T))
  var my = math.RoundToInt(myf)
  pMtn.x = -mx
  pMtn.y = -my
  pClouds.x = -mx - 40 + f / 25
  pClouds.y = mtn.cloudY - my
  for i = 0 .. 1
    var l = lamps[i]
    var lx = mtn.mineX - 8 + i * 15
    l.x = lx - mx
    l.y = mtn.mineY - 2 - my
    l.frame = (f / 4 + i * 2) % 4
    Light(mineGlow[i], lit, lx - mx, mtn.mineY - 2 - my)
  // Pushers, specks with their carts, going in
  for i = 0 .. 3
    var d = dots[i]
    var dx = mtn.mineX - 34 + ((f / 6 + i * 9) % 32)
    Show(d, dx < mtn.mineX - 5)
    d.x = dx - mx
    d.y = mtn.mineY - my
  // him: along the last stretch of the trail, slowly
  var n = mtn.trailX.Count()
  var u = Prog(T_MTN, T_UP + 60)
  Show(speck, true)
  speck.x = math.RoundToInt(mtn.trailX[n - 2] + (mtn.trailX[n - 1] - mtn.trailX[n - 2]) * u) - mx
  // he only climbs a row on a frame the pan steps (or while it's still), or he jitters against it
  var row = math.RoundToInt(mtn.trailY[n - 2] + (mtn.trailY[n - 1] - mtn.trailY[n - 2]) * u)
  var panning = (f > T_PAN & f < T_PAN + PAN_T) | (f > T_UP & f < T_UP + UP_T)
  ?speckRow = -1000 | !panning | my ! lastMy
    speckRow = row
  lastMy = my
  speck.y = speckRow - my
  speck.frame = (f / 20) % 2
  // a faint light on him, so the eye finds him
  Light(speckGlow, lit, speck.x, speck.y)
  Drift(mx, myf)
  ShootingStar()
  Twinkle()

// once the mountain's gone, stars twinkle here and there (none on the moon), fading out with
// the title's Ki since scene 7's sky has none
func Twinkle()
  var fade = 1.0 - Prog(T_END - 80, T_END - 20)
  var on = f >= T_UP + UP_T - 30 & fade > 0
  var c = color.Lerp("#000000", TWINKLE, fade)
  for i = 0 .. N_TWINKLE - 1
    var tw = twinkles[i]
    var x = (i * 37 + 11) % W
    var y = IMax(2, H / 12) + (i * 7 + 3) % IMax(4, H - 2 * IMax(2, H / 12) - 2)
    var onMoon = x >= MOON_X - 1 & x < MOON_X + 15 & y < MOON_Y + 8
    Show(tw, on & !onMoon)
    tw.x = x
    tw.y = y
    tw.color = c
    var ph = (f + i * 29) % (70 + (i % 4) * 15)
    tw.frame = 0
    ?ph < 24
      tw.frame = ph / 6

// snow blown off the summit to the right, thrown up after us as we leave the top, thinning out.
// each flake is rounded once against the view so none wobble
func Drift(mx, myf)
  for i = 0 .. N_DRIFT - 1
    var life = 150 + (i * 37) % 80
    var since = f - T_DRIFT - (i * 23) % life
    ?since >= 0
      var age = since % life
      ?f - age <= T_DRIFT_END
        var d = drift[i]
        var vx = 0.10 + (i % 5) * 0.04
        var vy = 0.03 + (i % 4) * 0.03
        ?f - age >= T_UP - 30
          vx = 0.18 + (i % 5) * 0.04
          vy = 0.18 + (i % 6) * 0.06
        var x = mtn.peakX + 1 + vx * age
        var y = mtn.peakY - vy * age
        Show(d, true)
        d.x = math.RoundToInt(x - mx)
        d.y = math.RoundToInt(y - myf)
        d.frame = (i + age / 9) % 3
        var a = math.Clamp(age / 8.0, 0.0, 1.0) * math.Clamp((life - age) / (life * 0.4), 0.0, 1.0)
        d.color = color.Lerp("#000000", "#ccd6dc", a)

// shooting stars (STARS) once the mountain's gone: a head with a trail that burns out from the head back
func ShootingStar()
  for n = 0 .. STARS.Count() / 5 - 1
    var t = f - STARS[n * 5]
    ?t >= 0 & t < STAR_T + N_GLINT
      Meteor(t, W * STARS[n * 5 + 1], IMax(2, H / 12) + STARS[n * 5 + 2], STARS[n * 5 + 3], STARS[n * 5 + 4])

func Meteor(t, x0, y0, vx, vy)
  for k = 0 .. N_GLINT - 1
    var u = t - k * 0.5
    ?u >= 0 & u < STAR_T
      var g = glints[k]
      Show(g, true)
      g.x = math.RoundToInt(x0 + u * vx)
      g.y = math.RoundToInt(y0 + u * vy)
      g.frame = 3
      ?k = 0
        g.frame = 0
        ?u > STAR_T - 6
          g.frame = 1
      :?k < 7
        g.frame = 2
      var a = (1.0 - k * 1.0 / N_GLINT) * math.Clamp((STAR_T - u) / 6.0, 0.0, 1.0)
      g.color = color.Lerp("#000000", "#e8ecf0", a)

// ---------------------------------------------------------------- the film calls these

// a scene built ahead waits off to the side, but the game still draws its canvases,
// so they get no width till it's on
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built near the end of the scene before so its lights are ready, then waits off to
// the side (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // nothing else on the frame it's built in, that's heavy enough
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

// take its picture off the screen (another scene is playing)
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start, for watching again
func Reset()
  f = 0
  over = false
