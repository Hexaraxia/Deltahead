// One must imagine the deltahead happy
// Scene 7: The mountain heights
//
// green stars. comes down out of scene 6's sky onto a night ridge whose summit never
// gets closer. the aurora comes up and pillars rise out of the cloud to carry the
// path on, past Hans's signposts. one frame counter, 30 a second

// timing, in frames

var T_CARD_OUT = 140
var T_BANNER = 175
var T_GREEN = 0         // stars and sky go green, set in Build for when the banner hits 16
var T_WALK = 60         // he comes in along the ridge
var T_EDGE = 0          // at the end of it
var T_RISE1 = 0         // the first pillars come up
var T_ON1 = 0           // he goes on
var T_FORK = 0          // under the signposts
var T_ON2 = 0
var T_WAVE = 0          // the whole path ahead rises
var T_END = 0
var V = 0.35            // his pace
var DESC = 160          // how far above the ridge it opens, in rows
var DESC_T = 150        // how long the way down takes
var SKY_X = -10         // where scene 6 left the sky and the moon
var SKY_Y = -20
var MOON_X = 4
var MOON_Y = 2
var X0 = -6.0
var EDGE_X = 66.0       // where he stops at the first gap
var FORK_X = 216.0      // where he gets his breath, under the signs
var WAVE_X = 245.0      // where he is when the path ahead comes up

var f = 0
var built = false
var over = false
var attached = false
var parked = false

var W = 0

var cx = 0

var camX = 0
var camY = 0
var heroX = 0.0
var heroLeft = false

var HOLES = [70, 100, 150, 190, 262, 400]  // no ground to glint until the pillars come up
var shadeCol = ""

// ---------------------------------------------------------------- building

func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// Canvases top out at 100 x 27 cells, so screen-sized ones are tiled.
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

func MakeText(s)
  var t = ui.AddText(s)
  root.Add(t)
  t.anchor = "center_center"
  t.dock = "center_center"
  t.x = 0
  t.y = 0
  t.w = string.Size(s) + 2
  t.h = 1
  t.align = "center"
  t.color = "#000000"
  return t

// a light: additive canvas fading out from the center, filled a bit each frame
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// when he gets where
func Schedule()
  T_GREEN = T_BANNER + 38
  T_EDGE = T_WALK + math.CeilToInt((EDGE_X - X0) / V)
  T_RISE1 = T_EDGE + 34
  T_ON1 = T_RISE1 + 6 * 10 + 40
  T_FORK = T_ON1 + math.CeilToInt((FORK_X - EDGE_X) / V)
  T_ON2 = T_FORK + 60
  T_WAVE = T_ON2 + math.CeilToInt((WAVE_X - FORK_X) / V)
  // no fade, scene 8 cuts in on him still walking
  T_END = T_WAVE + 110

func Build()
  built = true
  W = screen.w
  H = screen.h
  cx = W / 2
  cy = H / 2
  Schedule()
  disable hud
  disable player
  disable banner
  storage.Set("hidHud", true)

  root = ui.AddPanel()
  root.style = invis
  root.anchor = "top_left"
  root.dock = "top_left"
  root.x = 0
  root.y = 0
  root.w = W
  root.h = H
  attached = true

  // back to front: sky, aurora, ranges, summit, ridge, pillars, him, cloud, snow, rocks, lights
  backdrop = Tiles("opaque", 0, H)
  Solid(backdrop)
  pSky = MakeAnim(skyArt.sky, SKY_X, SKY_Y)
  moon = MakeAnim(skyArt.moon, MOON_X, MOON_Y)
  auroraLo = MakeAnim(spr.auroraLo, 0, 0)
  auroraMid = MakeAnim(spr.auroraMid, 0, 0)
  auroraHi = MakeAnim(spr.auroraHi, 0, 0)
  pFar = MakeAnim(world.far, 0, 0)
  summit = MakeAnim(spr.summit, 0, 0)
  pBack = MakeAnim(world.back, 0, 0)
  mist = MakeAnim(spr.mist, 0, 0)
  ridgeA = MakeAnim(world.ridgeA, 0, 0)
  ridgeB = MakeAnim(world.ridgeB, 0, 0)
  // moonlight catching the snow
  glints = snowLight.Make(root, 6)
  for i = 0 .. world.pillarX.Count() - 1
    pillars.Add(MakeAnim(spr.pillar, 0, 0))
  for i = 0 .. 5
    puffs.Add(MakeAnim(spr.puff, 0, 0))
  heroR = MakeAnim(spr2.heroR, 0, 0)
  heroL = MakeAnim(spr2.heroL, 0, 0)
  cloud = MakeAnim(spr.cloud, 0, 0)
  for i = 0 .. 9
    flakes.Add(MakeAnim(spr.flake, 0, 0))
  near = MakeAnim(spr.near, 0, 0)
  auroraGlow = QueueHalo(80, 14, 14, 52, 22, 0.25)
  moonGlow = QueueHalo(31, 13, 40, 46, 58, 0.12)

  shade = Tiles("multiply", 0, H)
  vig = look.Vignette(root, W, H, 0.35)
  for c : vig
    parkable.Add(c)
  var lb = IMax(2, H / 12)
  for c : Tiles("opaque", 0, lb)
    bars.Add(c)
  for c : Tiles("opaque", H - lb, lb)
    bars.Add(c)
  Solid(bars)
  card = MakeText("The higher he climbed, the farther away the top got.")
  // above him, out of his way
  card.y = -(H / 4)
  banner.Build(root, "The Mountain Heights", 13, 16, T_BANNER, lb + 1)


// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// off lights get no width too: the game draws every canvas cell, shown or not
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

// the row his feet are on at x: the ridge climbs as it goes
func Feet(x)
  return world.feet0 + (world.feet1 - world.feet0) * x * 1.0 / world.ww

func HeroX()
  ?f < T_WALK
    return X0
  ?f < T_EDGE
    return X0 + V * (f - T_WALK)
  ?f < T_ON1
    return EDGE_X
  ?f < T_FORK
    return EDGE_X + V * (f - T_ON1)
  ?f < T_ON2
    return FORK_X
  return FORK_X + V * (f - T_ON2)


// ---------------------------------------------------------------- him

func HeroAt(hx, r, frame, left)
  var b = heroR
  ?left
    b = heroL
  Show(b, true)
  // same rounded x the camera uses so they never disagree
  var sx = math.RoundToInt(hx) - camX
  b.x = sx - 1
  ?left
    b.x = sx - 4
  b.y = SY(r) - 3
  b.frame = frame

// walks, looks down at the first gap while the pillars rise, gets his breath at the signposts
func Him()
  heroX = HeroX()
  var r = math.FloorToInt(Feet(math.RoundToInt(heroX)))
  var frame = (f / 6) % 4
  ?f >= T_EDGE & f < T_ON1
    frame = 4
    ?f >= T_EDGE + 10 & f < T_RISE1 + 30
      frame = 9
  :?f >= T_FORK & f < T_ON2
    frame = 9
    ?f >= T_ON2 - 14
      frame = 4
  HeroAt(heroX, r, frame, false)


// ---------------------------------------------------------------- the path

// pillars: one by one at the first gap, just ahead of him at the second, all along at
// the last. each slows as it rises, overshoots a row and settles with a puff
var RISE = 36
func Pillars()
  var n = world.pillarX.Count()
  var k1 = 0
  var k3 = 0
  var pu = 0
  for i = 0 .. n - 1
    var px = world.pillarX[i]
    var p = 0.0
    var age = -1.0
    ?px < world.gapB[0]
      var s = T_RISE1 + k1 * 12
      p = Prog(s, s + 40)
      age = f - (s + 40)
      k1 = k1 + 1
    :?px < world.gapB[1]
      p = math.Clamp((heroX - (px - 16)) / 11.0, 0.0, 1.0)
      age = (heroX - (px - 5)) / V
    :
      var s = T_WAVE + k3 * 3
      p = Prog(s, s + 36)
      age = f - (s + 36)
      k3 = k3 + 1
    var pl = pillars[i]
    var r = math.FloorToInt(Feet(px))
    var e = 1.0 - (1.0 - p) * (1.0 - p) * (1.0 - p)
    var off = math.RoundToInt(RISE * (1.0 - e))
    ?p >= 0.86 & p < 0.97
      off = -1
    :?p >= 0.97
      off = 0
    Show(pl, p > 0)
    pl.x = SX(px)
    pl.y = SY(r - world.floorBack + off)
    ?age >= 0 & age < 12 & pu < puffs.Count()
      var pf = puffs[pu]
      Show(pf, true)
      pf.frame = IMin(3, math.FloorToInt(age / 3))
      pf.x = SX(px) - 4
      pf.y = SY(r) - 3
      pu = pu + 1


// ---------------------------------------------------------------- sounds

// wind, the sky going green, and the pillars grinding up
func Sfx()
  ?f = 0
    ambient.Stop()
    ambient.Add("ambient_icy")
    music.Stop()
  :?f = T_GREEN + 4
    play elementalist_vigor_blink 80
  :?f = T_RISE1 - 6
    play yeti_blow_ice_wall 70
  :?f = T_FORK + 4
    play air_hiss 125
  :?f = T_WAVE
    play fissure_break_apart 75
  :?f = T_WAVE + 20
    play yeti_blow_ice_wall 60
  // each of the first lot landing
  var k = f - T_RISE1 - 22
  ?k >= 0 & k < 60 & k % 10 = 0
    play stone_throw_hit @70 + k / 10 * 4@
  // each pillar as it comes up under him on the way
  ?f > T_ON1 & f < T_FORK
    for i = 0 .. world.pillarX.Count() - 1
      var px = world.pillarX[i]
      ?px >= world.gapA[1] & px < world.gapB[1] & math.FloorToInt(heroX) = px - 13 & math.FloorToInt(heroX - V) < px - 13
        play pickup_stone @78 + (i % 3) * 8@


// ---------------------------------------------------------------- the frame

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  Show(heroR, false)
  Show(heroL, false)
  ShowAll(puffs, false)
  var lit = jobs.Count() = 0

  // camera trails him with room ahead, leads when the path rises. whole cells from
  // where he's drawn so he never wobbles against the ground
  var hx = math.RoundToInt(HeroX())
  var lead = 0
  ?f >= T_WAVE
    lead = math.RoundToInt(26 * Ease(Prog(T_WAVE, T_WAVE + 70)))
  camX = IMax(0, IMin(world.ww - W, hx - (cx - 14) + lead))
  // coming down out of the sky at the start
  var high = DESC * (1.0 - Ease(Prog(0, DESC_T)))
  camY = math.FloorToInt(Feet(hx)) - (cy + 3) - math.RoundToInt(high)
  Him()

  // parallax: sky and ranges barely move, summit not at all, near rocks the most
  var sea = cy - 3
  pSky.x = SKY_X - math.RoundToInt(camX * 0.05)
  pSky.y = SKY_Y
  moon.x = MOON_X
  moon.y = MOON_Y
  pFar.x = -10 - math.RoundToInt(camX * 0.25)
  pFar.y = sea - world.farSea + math.RoundToInt(high * 0.25)
  pBack.x = -20 - math.RoundToInt(camX * 0.6)
  pBack.y = sea + 2 - math.RoundToInt((camY - (world.feet0 - (cy + 3))) * 0.6)
  mist.x = -30 + math.RoundToInt(-f / 20.0 - camX * 0.7)
  mist.y = sea + 3 + math.RoundToInt(high * 0.7)
  near.x = -10 - math.RoundToInt(camX * 1.5)
  near.y = H - 12 + math.RoundToInt(high * 1.3)
  for i = 0 .. 9
    var s = flakes[i]
    s.x = math.RoundToInt(Wrap((i * 53) % 97 - f * 0.25 + math.Sin(f * 0.05 + i) * 1.5, W + 4)) - 2
    s.y = math.RoundToInt(Wrap((i * 37) % 33 + f * 0.18, H + 4)) - 2
    s.frame = i % 3
    // fades in once it's down from the sky
    s.color = color.Lerp("#000000", "#9aa6ac", Prog(DESC_T - 60, DESC_T))
  var g = Prog(T_GREEN, T_GREEN + 60)
  var ax = -20 - math.RoundToInt(camX * 0.08)
  var af = (f / 7) % 8
  auroraHi.x = ax
  auroraHi.y = IMax(2, H / 12) + math.RoundToInt(high * 0.08)
  auroraHi.frame = af
  auroraHi.color = color.Lerp("#000000", "#32da5a", g)
  auroraMid.x = ax
  auroraMid.y = IMax(2, H / 12) + math.RoundToInt(high * 0.08)
  auroraMid.frame = af
  auroraMid.color = color.Lerp("#000000", "#1f9a40", g)
  auroraLo.x = ax
  auroraLo.y = IMax(2, H / 12) + math.RoundToInt(high * 0.08)
  auroraLo.frame = af
  auroraLo.color = color.Lerp("#000000", "#136328", g)
  ShowAll([auroraHi, auroraMid, auroraLo], g > 0)
  Light(auroraGlow, lit & g > 0.5, cx, IMax(2, H / 12) + 5)
  Light(moonGlow, lit, MOON_X + 5, MOON_Y + 3)
  summit.x = W - 28
  summit.y = sea - 7 + math.RoundToInt(high * 0.25)
  ridgeA.x = SX(0)
  ridgeA.y = SY(0)
  ridgeB.x = SX(200)
  ridgeB.y = SY(0)
  Pillars()
  // risen pillars are ground to glint on
  var holes = HOLES
  ?f >= T_RISE1 + 100
    holes = [150, 190, 262, 400]
  ?f >= T_FORK
    holes = [262, 400]
  ?f >= T_WAVE + 60
    holes = []
  snowLight.Update(glints, f, camX, camY, W, world.feet0, world.feet1, world.ww, world.floorBack, world.floorFront, holes, f > DESC_T)
  // drifts the way the camera goes so it never steps back
  cloud.x = SX(-20 - f * 0.03)
  cloud.y = SY(Feet(heroX) + 7 + world.floorFront)

  card.color = Fade(T_CARD, T_CARD + 25, T_CARD_OUT, T_CARD_OUT + 25)
  Show(card, f < T_CARD_OUT + 26)
  banner.Update(f)
  SetShade(color.Lerp(NORMAL, GRADE_H, Prog(0, DESC_T)))
  Sfx()


// ---------------------------------------------------------------- the film calls these

// the game draws every canvas cell even off screen, so they get no width while parked
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so its lights are ready, parked off screen (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

// off the screen while another scene plays
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start, for watching again
func Reset()
  f = 0
  over = false
