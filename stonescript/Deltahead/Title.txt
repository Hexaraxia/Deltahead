// One must imagine the deltahead happy
// The title card
//
// Ki rises and settles into stone letters with a big ∆ for the D, the ∆ lights in rainbow on
// the swell of the bridge music, then it all breaks back into Ki and floats off, ∆ last.
// new it, Build(root, start frame) once, then Update(frame) every frame until over

var art = import Deltahead/ArtTitle
// viewer options (flashes, shake). no comment on an import line or the game can't read the file
var opt = import Deltahead/Options

var canvas = null
var delta = null
var sparks = null
var t0 = 0
var over = false
var keepMusic = false     // leave the music on at the end, for a scene that carries on under it
var playMusic = true      // false: the scene plays it and times the ∆ (630 frames in) to a hit
var kiAt = 90             // when the Ki starts rising, below 0 means before the card
var CW = 0
var CH = 0
var offX = 0
var offY = 0
var T_END = 990
var N_KI = 26
var KI = "#bcd0ec"
var KI_DIM = "#7486a0"   // bright enough to read rising over a starry sky
var INK = "#e8e8e8"
var SMALL = "#a4a4a4"
var HOT = "#ffffff"
var OFF = -1000
// per title cell, worked out once in Build: x, y, on the canvas, glyph, color, part of the ∆
var TX = []
var TY = []
var TIN = []
var TG = []
var TB = []
var TD = []
// color tables: fade in (25 frames) and flash (24) per cell, mote fading out (70)
var TF = []
var TH = []
var LEAVE_C = []

func Build(root, at)
  t0 = at
  CW = IMin(100, screen.w)
  CH = IMin(27, screen.h)
  offX = (CW - art.w) / 2
  offY = (CH - art.h) / 2
  canvas = ui.AddCanvas()
  root.Add(canvas)
  canvas.anchor = "top_left"
  canvas.dock = "top_left"
  // zero width while hidden, the game draws every canvas wherever it is
  canvas.w = 0
  canvas.h = CH
  canvas.x = OFF
  canvas.y = 0
  canvas.SetFG("#000000")
  canvas.SetBG("#000000")
  // '#' cells aren't drawn, so it's see-through
  canvas.Set("#")
  delta = MakeAnim(root, art.delta)
  sparks = MakeAnim(root, art.sparks)
  var fadeInk = []
  var fadeSmall = []
  for k = 0 .. 24
    fadeInk.Add(color.Lerp(KI, INK, k / 25.0))
    fadeSmall.Add(color.Lerp(KI, SMALL, k / 25.0))
  var hotInk = []
  var hotSmall = []
  for k = 0 .. 23
    hotInk.Add(color.Lerp(HOT, INK, k / 24.0))
    hotSmall.Add(color.Lerp(HOT, SMALL, k / 24.0))
  for k = 0 .. 69
    LEAVE_C.Add(color.Lerp(KI, "#000000", k / 70.0))
  for i = 0 .. art.cellX.Count() - 1
    var tx = art.cellX[i] + offX
    var ty = art.cellY[i] + offY
    TX.Add(tx)
    TY.Add(ty)
    TIN.Add(tx >= 0 & tx < CW & ty >= 0 & ty < CH)
    TG.Add(string.Sub(art.cellS, i, 1))
    var base = INK
    var fades = fadeInk
    var hots = hotInk
    ?i < art.nTop | art.cellY[i] = 9
      base = SMALL
      fades = fadeSmall
      hots = hotSmall
    TB.Add(base)
    TF.Add(fades)
    TH.Add(hots)
    TD.Add(art.cellX[i] < 13 & art.cellY[i] >= 2 & art.cellY[i] <= 7)

func MakeAnim(root, layers)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  a.anchor = "top_left"
  a.dock = "top_left"
  a.color = layers[0][0]
  a.visible = false
  return a

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Clamp01(v)
  return math.Clamp(v, 0.0, 1.0)

// a glyph in a color, if it's on the canvas
func Put(x, y, c, g)
  var cx = math.RoundToInt(x)
  var cy = math.RoundToInt(y)
  ?cx >= 0 & cx < CW & cy >= 0 & cy < CH
    canvas.Set(cx, cy, g)
    canvas.SetFG(cx, cy, c)

func Update(f)
  var t = f - t0
  ?t < IMin(0, kiAt) | t >= T_END
    canvas.x = OFF
    canvas.w = 0
    delta.visible = false
    sparks.visible = false
    over = t >= T_END
    return
  over = false
  var ox = (screen.w - CW) / 2
  var oy = (screen.h - CH) / 2
  canvas.x = ox
  canvas.w = CW
  canvas.y = oy
  canvas.Set("#")
  ?t = 0 & playMusic
    music.Play("bridge_crossing")
  ?t = art.lightAt
    play treasure_item_rainbow
  ?t = 800
    play ui_starfirst 90
  ?t = T_END - 1 & playMusic & !keepMusic
    music.Stop()
  Mask(t)
  Ki(t)
  Letters(t)
  // the lit ∆, going up on its own at the end
  var dx = ox + offX + art.deltaX
  var dy = oy + offY + art.deltaY
  var lit = t >= art.lightAt
  delta.visible = lit
  ?lit
    var up = IMax(0, t - art.deltaLeaves)
    delta.x = dx
    delta.y = dy - math.RoundToInt(up * up * 0.012)
    delta.color = "#rainFF"
  sparks.visible = t >= art.lightAt & t < art.lightAt + 18
  sparks.frame = IMin(5, (t - art.lightAt) / 3)
  sparks.x = dx + 6 - 17
  sparks.y = dy + 3 - 7

// blanks the gaps in and between letters while they're in, so the sky doesn't show through
func Mask(t)
  // the gaps are all inside the card, so if it fits a bare Set is fine
  ?t < art.gapFirst | t >= art.gapLast | offX < 0 | offY < 0
    return
  for i = 0 .. art.gapX.Count() - 1
    ?t >= art.gapFrom[i] & t < art.gapTo[i]
      canvas.Set(art.gapX[i] + offX, art.gapY[i] + offY, " ")

// Ki rising across the dark the whole time
func Ki(t)
  var a = Clamp01((t - kiAt) / 90.0) - Clamp01((t - (T_END - 80)) / 60.0)
  ?a <= 0
    return
  var c = color.Lerp("#000000", KI_DIM, a)
  for k = 0 .. N_KI - 1
    var v = 0.08 + (k % 5) * 0.025
    var y = CH - 1 - ((t + 600 + k * 41) * v) % (CH + 3)
    var x = (k * 37 + 7) % CW + math.Sin(t * 0.03 + k) * 1.2
    Put(x, y, c, ".")

// each title cell: a mote rising into place, the glyph, then a mote rising away
func Letters(t)
  var light = t - art.lightAt
  var hot = light >= 0 & light < 24 & opt.flashes
  var dark = light < 0
  for i = 0 .. TX.Count() - 1
    var a = art.arrive[i]
    // lit ∆ cells and cells before their mote just get skipped (the game warns on a comment-only branch)
    ?t >= a - art.fly & t < a
      var tx = TX[i]
      var ty = TY[i]
      var p = (t - a + art.fly) / (art.fly * 1.0)
      var sx = tx + art.riseDX[i]
      var sy = CH - 1.0
      var k = 1.0 - (1.0 - p) * (1.0 - p)
      var g = "·"
      ?(i + t / 4) % 3 = 0
        g = "."
      Put(sx + (tx - sx) * p + math.Sin(p * 6 + i) * (1 - p) * 1.5, sy + (ty - sy) * k, KI, g)
    :?t >= a & (dark | !TD[i])
      ?t < art.leave[i]
        // the glyph, if it's on the canvas
        ?TIN[i]
          var c = TB[i]
          ?t - a < 25
            c = TF[i][t - a]
          ?hot
            c = TH[i][light]
          canvas.Set(TX[i], TY[i], TG[i])
          canvas.SetFG(TX[i], TY[i], c)
      :?t < art.leave[i] + 70
        var q = (t - art.leave[i]) / 70.0
        var g = "·"
        ?q < 0.08
          g = TG[i]
        var ty = TY[i]
        Put(TX[i] + math.Sin(q * 5 + i) * 1.2, ty - q * q * (ty + 4), LEAVE_C[t - art.leave[i]], g)
