// a light: additive canvas fading out from the center, filled a bit each frame

  var c = MakeCanvas(OFF, 0, w, h, "add")

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// extra real frames a fight's marks add
func Extra(m)
  var n = 0
  ?m.Count() > 0
    for k = 0 .. m.Count() / 2 - 1
      n = n + IMax(m[k * 2 + 1], -m[k * 2 + 1])
  return n

// real frame for story time S in fight i
func RealAt(i, S)
  var m = MARKS[i]
  var r = S
  ?m.Count() > 0
    for k = 0 .. m.Count() / 2 - 1
      var s = m[k * 2]
      var n = m[k * 2 + 1]
      ?s < S
        ?n > 0
          r = r + n
        :
          r = r + IMin(-n, S - s)
  return A[i] + r

// a fight's story time, t real frames into it
func Story(t, m)
  var cum = 0
  ?m.Count() > 0
    for k = 0 .. m.Count() / 2 - 1
      var s = m[k * 2]
      var n = m[k * 2 + 1]
      ?t < s + cum
        return (t - cum) * 1.0
      ?n > 0
        ?t < s + cum + n
          return s * 1.0
        cum = cum + n
      :
        ?t < s + cum - 2 * n
          return s + (t - s - cum) * 0.5
        cum = cum - n
  return (t - cum) * 1.0

// when he gets to each stop, plus shakes and impacts in real time
func Schedule()
  var t = T_WALK + math.CeilToInt((P[0] - X0) / V)
  for i = 0 .. P.Count() - 1
    D[i] = SL[i] + Extra(MARKS[i])
    ?i > 0
      t = A[i - 1] + D[i - 1] + math.CeilToInt((P[i] - P[i - 1] - EXIT[i - 1]) / V)
    A[i] = t
    var sh = SHAKES[i]
    ?sh.Count() > 0
      for k = 0 .. sh.Count() / 3 - 1
        SHK.Add(RealAt(i, sh[k * 3]))
        SHK.Add(sh[k * 3 + 1])
        SHK.Add(sh[k * 3 + 2])
    var im = IMPACTS[i]
    ?im.Count() > 0
      for k = 0 .. im.Count() / 2 - 1
        IMP.Add(RealAt(i, im[k * 2]))
        IMP.Add(im[k * 2 + 1])
  T_MUSIC = A[0] - 30
  // no fade, scene 9 cuts in with him walking on
  T_FADE = A[7] + 260
  T_END = T_FADE

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

// u through n frames, 0 to 1
func Ramp(u, n)
  return math.Clamp(u * 1.0 / n, 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

// from a to b as u goes from u0 to u1, eased
func Glide(u, u0, u1, a, b)
  return a + (b - a) * Ease(Ramp(u - u0, u1 - u0))

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// off lights get no width too: the game draws every canvas cell, shown or not
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

// row his feet are on at x, the ridge climbs as it goes
func Feet(x)
  return world.feet0 + (world.feet1 - world.feet0) * x * 1.0 / world.ww

// his row at x (the creatures' too)
func Row(x)
  return math.FloorToInt(Feet(math.RoundToInt(x)))

// walks from stop to stop
func HeroX()
  walking = false
  ?f < T_WALK
    return X0
  var from = X0
  var left = T_WALK * 1.0
  for i = 0 .. P.Count() - 1
    ?f < A[i]
      walking = true
      return from + V * (f - left)
    ?f < A[i] + D[i]
      ?i = ZIP
        return P[i] + ZipDX(f - A[i])
      return P[i] * 1.0
    from = P[i] + EXIT[i] * 1.0
    left = A[i] + D[i] * 1.0
  walking = true
  return from + V * (f - left)

// story time passes x this frame (sp last frame's, st this one's)
func Cross(sp, st, x)
  return sp < x & st >= x

// how hard it shakes this frame
func ShakeNow()
  var amp = 0
  ?!opt.shake
    return 0
  ?SHK.Count() > 0
    for k = 0 .. SHK.Count() / 3 - 1
      ?f >= SHK[k * 3] & f < SHK[k * 3] + SHK[k * 3 + 1]
        amp = IMax(amp, SHK[k * 3 + 2])
  return amp

func ImpactNow()
  // impact frames count as a flash, so skip them with flashes off
  ?!opt.flashes
    return false
  ?IMP.Count() > 0
    for k = 0 .. IMP.Count() / 2 - 1
      ?f >= IMP[k * 2] & f < IMP[k * 2] + IMP[k * 2 + 1]
        return true
  return false


// ---------------------------------------------------------------- him

// walking, standing, or whatever the fight wants. swings get the blade arc, dashes get afterimages
func Him()
  var frame = 4
  ?walking
    frame = (f / 5) % 4
  ?hFrame >= 0
    frame = hFrame
  var hx = heroX + hDX
  var r = math.FloorToInt(Feet(math.RoundToInt(hx)))
  var b = heroR
  ?hLeft
    b = heroL
  Show(b, true)
  // same rounded x the camera follows so they never disagree
  var sx = math.RoundToInt(hx) - camX
  var sy = SY(r) - 3 + math.RoundToInt(hDY)
  b.x = sx - 1
  ?hLeft
    b.x = sx - 4
  // dash and run frames have the blade out behind him, so shift them a column
  ?frame = 13 | frame >= 24
    b.x = b.x - 1
    ?hLeft
      b.x = b.x + 2
  b.y = sy
  b.frame = frame
  ?frame = 10
    var sl = slashR
    sl.x = sx + 4
    ?hLeft
      sl = slashL
      sl.x = sx - 7
    Show(sl, true)
    sl.y = sy
  // position history for the afterimages
  for k = 7 .. 1
    ghX[k] = ghX[k - 1]
    ghY[k] = ghY[k - 1]
    ghF[k] = ghF[k - 1]
  ghX[0] = b.x
  ghY[0] = sy
  ghF[0] = frame
  ?ghostOn & !hLeft
    for k = 0 .. 1
      var g = ghosts[k]
      Show(g, true)
      g.x = ghX[2 + k * 3]
      g.y = ghY[2 + k * 3]
      g.frame = ghF[2 + k * 3]
      g.color = color.Lerp("#7a7a7a", "#2a2a2a", k * 1.0)

// his frame u frames into a swing (blade lands at 9), -1 once it's done
func Swing(u)
  ?u < 0 | u >= 23
    return -1
  ?u < 6
    return 6
  ?u < 9
    return 10
  ?u < 17
    return 7
  return 8

// quick swing (blade lands at 4)
func Quick(u)
  ?u < 0 | u >= 12
    return -1
  ?u < 2
    return 6
  ?u < 4
    return 10
  ?u < 9
    return 7
  return 8

func DoSwing(u)
  var s = Swing(u)
  ?s >= 0
    hFrame = s

func DoQuick(u)
  var s = Quick(u)
  ?s >= 0
    hFrame = s


// ---------------------------------------------------------------- the creatures

// frame u frames into an animation with these holds
func Seq(u, holds, base, rate)
  var s = u / rate
  for i = 0 .. holds.Count() - 1
    ?s < holds[i]
      return base + i
    s = s - holds[i]
  return base + holds.Count() - 1

// puts a creature's feet on the path at world x (its left edge)
func OnPath(a, x, w, h)
  Show(a, true)
  var ix = math.RoundToInt(x)
  a.x = ix - camX
  a.y = SY(math.FloorToInt(Feet(ix + w / 2))) - (h - 1)

// buried till it wakes, then frame fr, and once dead it sinks back into the snow and fades
func Life(a, x, t, wake, dead, fr, col, w, h, wakeBase, wakeN, wakeRate)
  ?dead >= 0 & t >= dead
    var u = t - dead
    ?u >= 70
      return
    OnPath(a, x, w, h)
    a.frame = wakeBase + IMax(0, wakeN - 1 - u / 4)
    a.color = color.Lerp(col, GONE, Ramp(u - 25, 45))
    return
  OnPath(a, x, w, h)
  a.color = col
  ?t < wake
    a.frame = wakeBase
  :?t < wake + wakeN * wakeRate
    a.frame = wakeBase + (t - wake) / wakeRate
  :
    a.frame = fr

func Eater(a, x, t, wake, dead, fr, col)
  Life(a, x, t, wake, dead, fr, col, spr.eW, spr.eH, spr.eWake, 5, 3)

func Gobbler(a, x, t, wake, dead, fr, col)
  Life(a, x, t, wake, dead, fr, col, spr.gW, spr.gH, spr.gWake, 5, 3)

// thrown from x0 to x1 over n frames in an arc hgt rows high, tumbling
func Thrown(a, x0, x1, u, n, hgt, w, h, wakeBase, col)
  var p = Ramp(u, n)
  OnPath(a, x0 + (x1 - x0) * p, w, h)

// a thrown Eater that dies where it lands

// x0 to x1 between t0 and t1
func Walk(t, t0, t1, x0, x1)
  return x0 + (x1 - x0) * Ramp(t - t0, t1 - t0)

// the broken stretch: he hooks his sword over one of Hans's haul lines, rides it across
// throwing sparks, lets go short of the far post and lands past the edge

// distance from the near post, t frames into the crossing
func ZipDX(t)
  var ride = ZIP_LET - world.zipA
  ?t < 14
    return 0.0
  // speeding up the whole way, still moving at the end for the leap off
  ?t < 14 + ZIP_RUN
    var u = Ramp(t - 14, ZIP_RUN)
    return ride * (3 * u * u - u * u * u) / 2
  // off the line, slowing in the air
  var p = Ramp(t - 14 - ZIP_RUN, ZIP_FLY)
  return ride + 9.0 * (1.0 - (1.0 - p) * (1.0 - p))

// the line's row at x, with a fraction, matching the art
func CableRow(x)
  var u = math.Clamp((x - world.zipA) * 1.0 / (world.zipB - world.zipA), 0.0, 1.0)
  var ya = math.FloorToInt(Feet(world.zipA)) - world.zipH
  var yb = math.FloorToInt(Feet(world.zipB)) - world.zipH
  return ya + (yb - ya) * u + world.zipSag * math.Sin(u * 3.1416)

// camera offset it wants: back to show the broken edge during the Berserkers, else wherever a fight leaves him
func Want()
  for i = 0 .. P.Count() - 1
    ?f >= A[i] & f < A[i] + D[i]
      var s = Story(f - A[i], MARKS[i])
      ?i = ZIP
        return 0
      ?i = 4 & s < 290
        return -18
      ?i = 0 & s < 222
        return 0
      ?i = 6 & s < 340
        return 0
      return EXIT[i]
  return 0

// the game draws every canvas cell even off screen, so they get no width while parked
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so its lights are ready, parked off screen (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

// off screen while another scene plays
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start for a rewatch
func Reset()
  f = 0
  over = false
  camLead = 0
  lastHx = 0
  stillFor = 0
