func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// Canvases top out at 100 x 27 cells, so screen-sized ones are tiled.
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

func MakeText(s, centered)
  var t = ui.AddText(s)
  root.Add(t)
  t.anchor = "top_left"
  t.dock = "top_left"

// a light: an additive canvas fading out from the center, filled a few hundred cells a frame
// so no frame gets heavy. canvases ignore .visible, so a light that's off waits off screen
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

func Center(c, x, y)

  c.y = y - c.h / 2

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// a light on its source this frame, or off screen. they're all put away each frame, so only the ones a shot places show
// an off light gets no width too, the game draws every canvas wherever it is (its cells stay filled)
func Light(c, on, x, y)
  ?on
    c.w = lightW[allLights.IndexOf(c)]
    Center(c, x, y)
  :
    c.x = OFF
    c.w = 0

// places a layer for the camera: par < 1 is farther and moves less, par > 1 nearer. they line up at the mine view
func Layer(c, par, ox, oy)
  c.x = ox - camX0 - math.RoundToInt((camX - camX0) * par)
  c.y = oy - camY0 - math.RoundToInt((camY - camY0) * par)

// world position on the main layer -> screen
func SX(x)
  return x - camX

func SY(y)
  return y - camY

// where on his lap the Pusher is, counted so he's at the end of the near straight at T_DEP
func RingIdx()
  var n = ring.pathX.Count()
  return ((f - T_DEP + 4 * ring.lap) * n / ring.lap + ring.depIdx) % n

// the Ring's chamber: back wall drifts slower than the track, front rocks faster. the bend breaks when the rock lands
func Chamber()
  cBack.x = ring.backX - camXe - math.RoundToInt((camX - camXe) * 0.8)
  cBack.y = ring.backY - camYe - math.RoundToInt((camY - camYe) * 0.8)
  cFront.x = ring.frontX - camXe - math.RoundToInt((camX - camXe) * 1.3)
  cFront.y = ring.frontY - camYe - math.RoundToInt((camY - camYe) * 1.3)
  post.x = SX(ring.postX)
  post.y = SY(ring.postY)
  bend.x = SX(ring.bendX)
  bend.y = SY(ring.bendY)
  bend.frame = 0
  ?f >= T_ROCK
    bend.frame = 1
  Show(cBack, true)
  Show(cFront, true)
  Show(post, true)
  Show(bend, true)

// grit trickles down, then the rock smashes the bend ahead of him. he doesn't look up
func Rock()
  var rx = SX(ring.rockX)
  var ry = SY(ring.rockY)
  for i = 0 .. 2
    var g0 = T_ROCK - 36 + i * 9
    var falling = f >= g0 & f < g0 + 16
    Show(specks[i], falling)
    ?falling
      specks[i].frame = f - g0
      specks[i].x = rx + 4 + i * 2
      specks[i].y = ry - 15
  // the fall, quicker as it goes
  ?f >= T_ROCK - 8
    var left = IMax(0, T_ROCK - f)
    Show(rock, true)
    rock.x = rx
    rock.y = ry - left * left * 24 / 64
  ?f >= T_ROCK & f < T_ROCK + 12
    Show(dust, true)
    dust.frame = IMin(5, (f - T_ROCK) / 2)
    dust.x = SX(ring.dustX)
    dust.y = SY(ring.dustY)
  ?f >= T_ROCK & f < T_ROCK + 6 & opt.shake
    root.x = (f % 2) * 2 - 1
    root.y = ((f + 1) % 2) * 2 - 1
  // what it knocks off the bend, thrown out and coming down
  for i = 0 .. 4
    var d = debris[i]
    var t = f - T_ROCK
    Show(d, t >= 0 & t < 90)
    ?t >= 0
      d.frame = i % 5
      var dy = DEB_VY[i] * t + 0.07 * t * t
      var floorY = ring.dustY + 4 + i % 2
      var y = ring.dustY + dy
      var x = ring.dustX + 2 + DEB_VX[i] * t
      ?y >= floorY & t > 3
        var tl = (-DEB_VY[i] + math.Sqrt(DEB_VY[i] * DEB_VY[i] + 0.28 * 4)) / 0.14
        x = ring.dustX + 2 + DEB_VX[i] * tl
        y = floorY
      d.x = SX(math.RoundToInt(x))
      d.y = SY(math.RoundToInt(y))
  // the dust it raises, settling in the lantern light
  for i = 0 .. 1
    var st = settle[i]
    var t2 = f - T_ROCK - 6 - i * 8
    Show(st, t2 >= 0 & t2 < 40)
    st.frame = IMin(4, IMax(0, t2) / 8)
    st.x = SX(ring.dustX - 3 + i * 5 + IMax(0, t2) / 10)
    st.y = SY(ring.dustY - 2 - IMax(0, t2) / 14)
  ?f = T_ROCK - 36
    play bomb_cart_move 45
  :?f = T_ROCK
    play stone_throw_hit 55
    play mushroom_boss_fat_slam 80

// the Boiling Mine, then down the pit to the Ring in one camera move: past the Pushers,
// over to the pit, down, and it holds on the Ring while it breaks and he walks off
func Descent()
  var lit = jobs.Count() = 0
  var u = Ease(Prog(T_DESCENT, T_RING - 10))
  camX = camX0 + math.RoundToInt(u * (camXe - camX0))
  camY = camY0 + math.RoundToInt(u * (camYe - camY0))
  Layer(dFar, 0.5, 0, 0)
  Layer(dBack, 0.75, 0, 0)
  Layer(dMain, 1.0, 0, 0)
  Layer(dFore, 1.3, 0, camY0 + 2)
  var shimmer = (f / 12) % 2
  Layer(liqA, 1.3, -6 + shimmer, 27)
  Layer(liqB, 1.3, -6 - (f / 16) % 2, 29)
  for i = 0 .. 3
    Layer(bubbles[i], 1.3, 60 + i * 17 + (i % 2) * 5, 26)
    bubbles[i].frame = (f / 4 + i * 7) % 10
    // steam rises where a bubble pops
    var pos = (f + i * 28) % 40
    Show(puffs[i], pos >= 20)
    Layer(puffs[i], 1.3, 59 + i * 17 + (i % 2) * 5, 21)
    puffs[i].frame = IMax(0, pos - 20) / 4
  for i = 0 .. 2
    Layer(drips[i], 1.3, 70 + i * 19, camY0 + 4)
    drips[i].frame = IMin(8, ((f + i * 23) % 70) / 3)
  // the pool's heat on everything near it, flickering with the boil
  for i = 0 .. 1
    var px = 18 + i * 92 - camX0 - math.RoundToInt((camX - camX0) * 1.3)
    var py = 29 - camY0 - math.RoundToInt((camY - camY0) * 1.3)
    var pd = (f / 5 + i * 2) % 3 = 0
    Light(poolHalo[i * 2], lit & !pd, px, py)
    Light(poolHalo[i * 2 + 1], lit & pd, px, py)
  // embers rising off it, past us on the way down
  for i = 0 .. 5
    var e = mineEmbers[i]
    var life = 70 + (i * 17) % 40
    var age = (f + i * 31) % life
    Show(e, true)
    e.frame = (f / 3 + i) % 4
    var ex = 8 + (i * 29) % (W - 16) + math.RoundToInt(math.Sin(f * 0.07 + i * 1.7) * 2)
    var ey = H - 4 - age * (H - 6) / life - math.RoundToInt(u * 6)
    e.x = ex
    e.y = ey
    ?age < 4 | age > life - 6
      e.color = "#e0461c"
    :
      e.color = "#ffb35c"

  // the Pushers and their Controller walk the rails
  for i = 0 .. 2
    var wx = 46 + (f - 350) * 4 / 30 - i * 16
    var w = walkers[i]
    w.x = SX(wx) + WALK_DX[i]
    w.y = SY(19) + WALK_DY[i]
    w.frame = (f / 6 + i) % 4
  // the Controller walks with his whip trailing (frames 17-20), stops at his line to crack it, then walks on
  walkers[2].frame = 17 + (f / 6 + 2) % 4
  ?f >= LINE_AT[CTRL_LINE]
    var lt = f - LINE_AT[CTRL_LINE]
    walkers[2].x = walkers[2].x - IMin(lt, 30) * 4 / 30
    ?lt < 30
      walkers[2].frame = 21 + lt / 3
    :?lt < 39
      walkers[2].frame = 8 + (lt - 30)
  // further back, another line going the other way, slower for the distance
  for i = 0 .. 2
    var bw = backWalkers[i]
    Layer(bw, 0.75, 150 - (f - T_MINE) * 12 / 100 + BACK_DX[i], mine.backTrack - 3 + WALK_DY[i])
    bw.frame = (f / 7 + i) % 4

  // what they say, typed out in a bubble over the speaker
  for k = 0 .. LINES.Count() - 1
    var on = f >= LINE_AT[k] & f < LINE_END[k]
    ?on
      var n = IMin(string.Size(LINES[k]), (f - LINE_AT[k]) * 6 / 5 + 1)
      var who = walkers[LINE_WHO[k]]
      // everyone's mouth on the same row, the Controller's sprite is taller
      bub.Show(speech[k], who.x + HEAD_X[LINE_WHO[k]], who.y - WALK_DY[LINE_WHO[k]], n, W, "#f0f0f0")
    :
      bub.Hide(speech[k])
    ?f = LINE_AT[k]
      ?LINE_WHO[k] = 2
        play controller_whip_attack
      :
        play slave_npc @100 - LINE_WHO[k] * 15@
  ?f = LINE_AT[CTRL_LINE] + 12
    play controller_whip_hit

  // fire and light below
  for i = 0 .. 1
    var gx = SX(world.geyserX[i])
    var gy = SY(world.geyserY[i])
    flames[i].x = gx - 1

// life down here: heat through the floor cracks, embers, steam vents, the boiling pool, drips, chains, bats, far fires
func Passage(t, mx, top, lit)
  for i = 0 .. 1
    var gx = mx + track.glowX[i]
    var gy = top + track.glowY[i]
    var dim = (t / 4 + i * 2) % 3 = 0
    Light(glowHalo[i * 2], lit & !dim, gx, gy)
    Light(glowHalo[i * 2 + 1], lit & dim, gx, gy)
  for i = 0 .. 3
    var e = tEmbers[i]
    var c = i % 2
    var rise = (t + i * 23) % 60
    Show(e, true)
    e.frame = (t / 3 + i) % 4
    e.x = mx + track.glowX[c] + (i / 2) * 3 - 1 + math.RoundToInt(math.Sin(t * 0.09 + i * 2) * 1.5)
    e.y = top + track.glowY[c] - 1 - rise / 5
  for i = 0 .. 1
    var st = tSteam[i]
    var sp = (t + i * 35) % 70
    Show(st, sp < 20)
    st.frame = sp / 4
    st.x = mx + track.ventX[i] - 2
    st.y = top + track.ventY[i] - 6
  for i = 0 .. 1
    var bb = tBubbles[i]
    Show(bb, true)
    bb.frame = (t / 4 + i * 5) % 10
    bb.x = mx + track.poolX - 3 + i * 3
    bb.y = top + track.poolY - 1
  for i = 0 .. 2
    var d = tDrips[i]
    var ph = (t + i * 37) % 70
    Show(d, ph < 17)
    d.frame = IMin(16, ph)
    d.x = mx + track.dripX[i] - 1
    d.y = top + track.dripTop
  for i = 0 .. 1
    var ch = tChains[i]
    Show(ch, true)
    ch.frame = (t / 12 + i * 2) % 4
    ch.x = mx + track.chainX[i] - 1
    ch.y = top + track.chainTop[i]
  for i = 0 .. 2
    var fire = tFires[i]
    Show(fire, true)
    fire.frame = (t / 4 + i) % 2
    fire.x = tFar.x + track.fireX[i] - 1
    fire.y = top + track.fireY - 1
  for i = 0 .. 1
    var b = tBats[i]
    var bt = t - 30 - i * 160
    var flying = bt >= 0 & bt < 150
    Show(b, flying)
    ?flying
      b.x = -4 + math.RoundToInt(bt * 0.75)
      b.y = top + 14 + i + math.RoundToInt(math.Sin(bt * 0.18 + i) * 1.5)
      b.frame = BAT_FRAMES[(bt / 3) % 4]


// after the dip: the long dark way off the Ring. he keeps pushing at the same pace and the camera
// keeps up. the Controller's lash sweeps the floor from behind and he hops it without looking
func Track()
  var lit = jobs.Count() = 0
  var t = f - T_TRACK
  var wx = t * ring.speed
  // his wheels run 7 rows below the middle of the screen
  var top = cy + 7 - track.wheels
  var mx = -math.RoundToInt(wx)
  tFar.x = -math.RoundToInt(wx * 0.55)
  tFar.y = top
  tMid.x = mx
  tMid.y = top
  // the front layer starts mid-screen so its columns pass him at the same time on any screen
  tFront.x = cx - math.RoundToInt(wx * 1.45)
  tFront.y = top
  Passage(t, mx, top, lit)

  var base = cy + 5
  cart.x = cx - 4
  cart.y = base
  body.x = cx - 7
  body.y = base
  body.frame = (f / 5) % 4

  Show(lash, false)
  root.x = 0
  for ls : [LASH1, LASH2]
    // flung out along the floor from back there; a loop rolls along it and cracks at his heels
    var k = f - ls
    ?f >= ls & k < spr.lashN
      Show(lash, true)
      lash.frame = k
      lash.x = cx - 6 - spr.lashReach
      lash.y = base + 3 - spr.lashH
    var hop = f - ls - spr.lashCrack + 7
    ?hop >= 0 & hop < 14
      body.frame = 4
      body.y = base - 2
      ?hop < 3 | hop >= 11
        body.y = base - 1
    ?f >= ls + spr.lashCrack & f < ls + spr.lashCrack + 4 & opt.shake
      root.x = (f % 2) * 2 - 1
    ?f = ls
      play controller_whip_attack
    :?f = ls + spr.lashCrack
      play controller_whip_hit

  SetShade(color.Lerp("#000000", GRADE_PASS, Prog(T_TRACK, T_TRACK + 25)))

// built ahead under the loading ∆ so its lights are filled before it starts, waits off to the side till then
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    attached = true
    root.x = OFF
    // nothing more on the frame it's built in, that's heavy enough
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
    attached = true
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  ?key = "bumpLBegin"
    debug = !debug
  Frame()
  ?f <= T_END
    f++
  :
    over = true
  ?debug
    >`1,1,#7d7a73,frame @f@   screen @W@ x @H@   lights left to build @jobs.Count()@

// take its picture off the screen (the next scene is playing)
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start, for watching again
func Reset()
  f = 0
  over = false
