// nothing's made till it's needed: making a scene reads all its art, which stalls the game

var lt = 0               // frames of the loading ∆
var jobs0 = -1           // how many lights the first scene had to fill
var ready = 0.0
var made = 0              // how many of the other scenes have been made, under the loading ∆
var lights = 0.0
var LOAD_RATE = 6         // how many times as fast the first scene's lights fill under the loading ∆

// the worn Deltahead stays off screen while the film is on
opt.filmOn = false
?string.Equals(loc.id, "waterfall")
  opt.filmOn = part > 0
  ?menu ! null
    ?menu.stage > menu.WAIT
      opt.filmOn = true
  // each scene is built ahead of its cut (its lights take a bit to fill) and takes over the frame the last one ends
  ?part = 0
    ?menu = null
      menu = new Deltahead/Menu
    menu.Update()
    // the first scene is made, then built, on the still black before the options open (a heavy frame each)
    ?menu.stage = menu.HOLD
      ?film1 = null
        film1 = new Deltahead/Scene1
      :?!film1.built
        film1.Warm()
        jobs0 = film1.jobs.Count()
      :
        menu.go = true
    // its lights filling while the options are up, then under the loading ∆
    ?menu.stage > menu.HOLD & menu.stage < menu.DONE
      film1.Warm()
    ?menu.stage = menu.DONE
      ?loader = null
        loader = new Deltahead/Loader
        loader.Build(ui.root, 0)
      film1.Warm()
      // nothing else is on, so fill them faster
      for i = 2 .. LOAD_RATE
        film1.RunJobs()
      // the other scenes are made one a frame, since each reads all its art (built later, near its turn)
      ?made < 16
        made++
        ?made = 1
          film2 = new Deltahead/Scene2
        :?made = 2
          film3 = new Deltahead/Scene3
        :?made = 3
          film4 = new Deltahead/Scene4
        :?made = 4

        :?made = 12
          film13 = new Deltahead/Scene13
        :?made = 13
          film14 = new Deltahead/Scene14
        :?made = 14
          film15 = new Deltahead/Scene15
        :?made = 15
          film16 = new Deltahead/Scene16
        :
          film17 = new Deltahead/Scene17
      lights = 1.0
      ?jobs0 > 0
        lights = 1.0 - film1.jobs.Count() * 1.0 / jobs0
      ready = (made + lights) / 17.0
      loader.Update(lt, ready)
      lt++
      ?loader.done
        menu.Off()
        part = 1
        film1.Update()
  :?part = 1
    film1.Update()
    ?film1.f >= film1.T_END - 90
      ?film2 = null
        film2 = new Deltahead/Scene2
      film2.Warm()
    ?film1.over
      film1.Detach()
      part = 2
      film2.Update()
  :?part = 2

    ?film2.f >= film2.T_END - 90
