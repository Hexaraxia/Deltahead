func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(p, layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  p.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

// a dark copy of a figure, made just before it so it's drawn behind: its shadow
func MakeShadow(p, layers)
  var a = ui.AddAnim(layers[0][1])
  p.Add(a)
  Place(a, OFF, 0)
  a.color = "#000000"
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = "#000000"
  return a

// what the ∆'s light doesn't reach on a place or figure, made right after it, one piece per color: [anim, color]
func MakeDim(p, pieces)
  var list = []
  for d : pieces
    list.Add([MakeAnim(p, d, OFF, 0), d[0][0]])
  return list

// the ∆'s light on a place, made right after its picture: the shadow, the unlit part, and the lit part ring by ring
func MakeLit(p, shadow, darks, lits)
  var rings = []
  for r : lits
    rings.Add(MakeAnim(p, r, OFF, 0))
  return [MakeAnim(p, shadow, OFF, 0), MakeDim(p, darks), rings]

func MakeCanvas(p, x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  p.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// Canvases top out at 100 x 27 cells, so screen-sized ones are tiled.
func Tiles(p, blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(p, i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)

    c.Set(" ")

// a light: an additive canvas fading out from the center, filled a few hundred cells a frame. off ones wait off screen
func QueueHalo(p, w, h, r, g, b, bgk)
  var c = MakeCanvas(p, OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Show(c, on)
  c.visible = on

// an off light gets no width too, the game draws every canvas wherever it is (its cells stay filled)
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func ShowShot(p, on)
  ?on
    p.x = 0
  :
    p.x = OFF

// which color a glow is on (they go round slowly)
func Hue(k)
  return (f / 24 + k) % 8

// what the ∆'s light doesn't reach on the place or figure at x, y (frame fr), going dark as the light comes up
func Dim(list, x, y, fr, on)
  for d : list
    var a = d[0]
    Show(a, on & glowU > 0.01)
    a.x = x
    a.y = y
    a.frame = fr
    a.color = color.Lerp(d[1], "#000000", DIM * glowU)

// the ∆'s light on a place at x, y (frame fr): its shadow comes in, the unlit part goes dark, and the
// lit part goes rainbow ring by ring out from the ∆
func Lit(set, x, y, shadowY, litY, fr)
  var a = set[0]
  Show(a, glowU > 0.02)
  a.x = x
  a.y = y + shadowY
  a.frame = fr
  a.color = color.Lerp("#000000", SHADOW_C, glowU)
  Dim(set[1], x, y, fr, true)
  var rings = set[2]
  for k = 0 .. rings.Count() - 1
    var r = rings[k]
    Show(r, glowU > (k + 1) * 0.18)
    r.x = x
    r.y = y + litY[k]
    r.frame = fr

// a figure's shadow a cell down and away from the ∆ (at lx), the figure going dark except its top,
// which goes rainbow once the light is half up
func Figure(a, shadow, dims, top, lx)
  var side = 0
  ?a.x + 2 > lx + 3
    side = 1
  :?a.x + 2 < lx - 3
    side = -1
  Show(shadow, a.visible & glowU > 0.02)
  shadow.x = a.x + side
  shadow.y = a.y + 1
  shadow.frame = a.frame
  shadow.color = color.Lerp("#000000", SHADOW_C, glowU)
  Dim(dims, a.x, a.y, a.frame, a.visible)
  Show(top, a.visible & glowU > 0.5)
  top.x = a.x
  top.y = a.y
  top.frame = a.frame

// a place going dark as the light comes over it
func Darken(col, dark, t0)
  return color.Lerp(col, dark, Ease(Prog(t0, t0 + GLOW_IN)))

func Wide(early)
  // the camera holds on the summit, then with the music goes down the mountain to its foot
  my = math.RoundToInt(myTop + (myFoot - myTop) * Ease(Prog(T_M, T_M + PAN_T)))
  pMtn.x = -mx
  pMtn.y = -my
  pClouds.x = -mx - 40 + f / 25
  pClouds.y = mtn.cloudY - my
  // the Stream's line, far off over the top
  beam.x = mtn.peakX - mx
  beam.y = mtn.peakY - 23 - my
  beam.frame = (f / 4) % 4
  // his ∆: up the line, and on up out of the sky
  Show(delta, f >= T_RISE & f < T_RISE + RISE_T)
  delta.x = mtn.peakX - mx
  delta.y = math.RoundToInt(mtn.peakY - 1 - (mtn.peakY + 6 - myTop) * Ease(Prog(T_RISE, T_RISE + RISE_T))) - my
  // the ∆ in the stars (the sky doesn't move as we go down), and its glow once it's whole
  var x0 = mtn.peakX - mx - 16
  cons.Update(c1, x0, lb, f - T_STARS, STAR_GAP, STAR_RAIN, T_LINES - T_STARS, LINE_RATE, f >= T_STARS)
  Light(skyGlows[Hue(0)], early & f >= T_FORMED, x0 + 15, lb + 7)
  // the mine's lamps, and the Pushers going in, specks with their carts
  for i = 0 .. 1
    var l = lamps[i]
    var x = mtn.mineX - 8 + i * 15
    l.x = x - mx
    l.y = mtn.mineY - 2 - my
    l.frame = (f / 4 + i * 2) % 4
    Light(mineGlow[i], early, x - mx, mtn.mineY - 2 - my)
  for i = 0 .. 3
    var d = dots[i]
    var x = mtn.mineX - 34 + ((f / 6 + i * 9) % 32)
    Show(d, x < mtn.mineX - 5)
    d.x = x - mx
    d.y = mtn.mineY - my

// where Pusher k (0 at the front) stands once he's stopped, and when he stops
func StopX(k)
  return a16.postL - 9 - GAP * k

func StopAt(k)
  return T_STOP0 + STOP_GAP * (N - 1 - k)

func Close(lit, early)
  var camYf = cy0 - TILT * Ease(Prog(T_TILT, T_TILT + TILT_T))
  camY = math.RoundToInt(camYf)
  mouthA.x = -camX
  mouthA.y = -camY
  mouthB.x = a16.half - camX
  mouthB.y = -camY
  nearA.x = -camX
  nearA.y = -camY
  nearB.x = a16.half - camX
  nearB.y = -camY
  for i = 0 .. 1
    var fl = flames[i]
    fl.x = a16.lampX[i] - camX
    fl.y = a16.lampY[i] - camY
    fl.frame = (f / 4 + i * 2) % 4
    Light(lampGlow[i], early, fl.x, fl.y)
  Light(innerGlow, early, a16.mouthX + 4 - camX, a16.feet - 4 - camY)
  // the ∆ over the cliff from down here, in the sky so it doesn't move as we tilt up
  cons.Update(c2, 16, 1, f - T_MOUTH - T_C2, 4, 20, 45, 1.6, true)
  // the mine's heat rising from under the lintel, one rounding against the camera so it doesn't wobble on the tilt
  for i = 0 .. steam.Count() - 1
    var s = steam[i]
    var age = (f + i * 23) % 120
    s.x = a16.mouthX - 6 + (i * 5) % 12 - age / 14 - camX
    s.y = math.RoundToInt(a16.feet - 9 - age / 9.0 - camYf)
    s.frame = (age / 20 + i) % 3
    s.color = color.Lerp("#6a6e74", "#000000", age / 120.0)
  // the Pushers go on at the Ring's pace till each stops, stand a moment, then shade their eyes and look up
  for k = 0 .. N - 1
    var b = bodies[k]
    var c = carts[k]
    var t = StopAt(k)
    var x = StopX(k) - V * IMax(0, t - f)
    b.x = math.RoundToInt(x) - camX
    b.y = a16.feet - 2 - camY
    c.x = b.x + 3
    c.y = b.y
    var fr = (f / 8 + k * 2) % 4
    ?f >= t + LOOK_AFTER
      fr = 5
    :?f >= t + 12
      fr = 4
    :?f >= t
      fr = 3
    b.frame = fr
  // the ∆'s light on the place, and on them
  Lit(mouthLit, -camX, -camY, lt.mouthShadowY, lt.mouthLitY, 0)
  Lit(nearLit, -camX, -camY, lt.nearShadowY, lt.nearLitY, 0)
  var lx = 16 + cons.VIEWS[1][0] / 2
  for k = 0 .. N - 1

// the Icy Ridge: the ∆ lights in its sky and the Ki creatures stop one by one to stare at it
func Ridge(lit)
  var u = f - T_RIDGE
  rDunes.frame = (f / 9) % 4
  cons.Update(c3, 47, 1, u - 6, 3, 15, 30, 2.0, true)
  var wF = [s8.gWalk, s8.eWalk, s8.eWalk]
  var rest = [s8.gWake + 4, s8.eWake + 4, s8.eWake + 4]
  var x0 = [58, 34, 74]
  var row = [lb + 13, lb + 17, lb + 19]
  var stop = [70, 50, 90]
  for k = 0 .. 2
    var a = rFoes[k]
    // on along the snow toward us (the game draws them facing left), until it stops
    var walked = IMin(u, stop[k])
    a.x = x0[k] - math.RoundToInt(walked * 0.25)
    a.y = row[k]
    a.color = GREEN
    ?u < stop[k]
      a.frame = wF[k] + (u / 6 + k) % 4
    :
      a.frame = rest[k]
  // the ∆'s light on the snow, and on them
  Lit(dunesLit, -10, lb + 9, lt.dunesShadowY, lt.dunesLitY, rDunes.frame)
  Lit(fieldLit, -10, lb + 14, lt.fieldShadowY, lt.fieldLitY, 0)
  for k = 0 .. 2
    Figure(rFoes[k], foeShadow[k], foeDim[k], foeTop[k], 47 + cons.VIEWS[2][0] / 2)

  // the ∆'s light on the canyon, and on them
  Lit(riverLit, -25, 1, lt.riverShadowY, lt.riverLitY, 0)
  var lx = 55 + cons.VIEWS[3][0] / 2

// ---------------------------------------------------------------- the frame

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  var lit = jobs.Count() = 0
  var early = lit
  var shot = 0
  ?f >= T_MOUTH
    shot = 1
  ?f >= T_RIDGE
    shot = 2
  ?f >= T_RIVER
    shot = 3
  ?f >= T_BACK
    shot = 1
  ShowShot(shotA, shot = 0)
  ShowShot(shotB, shot = 1)
  ShowShot(shotC, shot = 2)
  ShowShot(shotD, shot = 3)
  // the shared moon is only for the mountain and the mine, the places have their own
  Show(moon, shot < 2)
  // the ∆'s light over the place, coming up as the ∆ forms over it
  glowU = 0.0
  ?shot = 1
    glowU = Ease(Prog(T_MOUTH + T_GLOW, T_MOUTH + T_GLOW + GLOW_IN))
  :?shot = 2
    glowU = Ease(Prog(T_RIDGE + 45, T_RIDGE + 45 + GLOW_IN))
  :?shot = 3
    glowU = Ease(Prog(T_RIVER + 45, T_RIVER + 45 + GLOW_IN))
  ?!lit
    glowU = 0.0
  ?shot = 0
    Wide(early)
  :?shot = 1
    Close(lit, early)
  :?shot = 2
    Ridge(lit)
  :
    River(lit)
  card.color = Fade(T_CARD, T_CARD + 25, T_CARD_OUT, T_CARD_OUT + 25)
  Show(card, f >= T_CARD & f < T_CARD_OUT + 26)
  // straight in from scene 15 on the mountain, the card over it
  ?shot = 0
    SetShade(GRADE_MOON)
  :?shot = 2
    SetShade(Darken(GRADE_RIDGE, DARK_RIDGE, T_RIDGE + 45))
  :?shot = 3
    SetShade(Darken(GRADE_RIVER, DARK_RIVER, T_RIVER + 45))
  :
    SetShade(color.Lerp(Darken(GRADE_MOUTH, DARK_MOUTH, T_MOUTH + T_GLOW), NORMAL, Prog(T_TILT, T_TILT + TILT_T)))
  Sfx()


// ---------------------------------------------------------------- the film calls these

// a scene built ahead waits off to the side, but the game still draws its canvases there, so no width till it's on
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built ahead near the end of the scene before so its lights are ready, waits off to the side (ui.Remove would recycle it)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // nothing more on the frame it's built in, that's heavy enough
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

func Reset()
  f = 0
  over = false
