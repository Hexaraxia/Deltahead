func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// canvases top out at 100 x 27 cells, so screen sized ones get tiled
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

func MakeText(s)

  t.anchor = "center_center"

// a light: an additive canvas fading out from the center, filled a bit each frame
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// a light on its source, or off screen with zero width (the game draws every canvas wherever it is)
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

// passage cell -> screen
func SX(x)
  return x - camX

func SY(y)
  return y + top

// the flash: k from 0 (none) to 1 (white)
func Flash(k)
  ?!opt.flashes
    k = 0
  var i = 0
  for c : flashTiles
    ?k > 0
      c.x = (i % ((W - 1) / 100 + 1)) * 100
      var v = math.RoundToInt(255 * k)
      c.SetFG(color.FromRGB(v, v, v))
      c.SetBG(color.FromRGB(v, v, v))
    :
      c.x = OFF
    i++

// where the Bomb Cart is: still, then picking up speed, then rolling on
func BombX()
  return BombXAt(f)

func BombXAt(at)
  ?at < T_ROLL
    return pass.bombStart
  var t = at - T_ROLL
  ?t < 40
    return pass.bombStart + 0.75 * t * t / 80.0
  return pass.bombStart + 15 + 0.75 * (t - 40)

// where the Pusher is while we follow him
func PusherX()
  return P0 + 0.3 * (f - T_CHASE)

// the Controller, by his head's cell, with his feet on row r
func CtrlAt(hx, r, frame)
  Show(ctrl, true)
  ctrl.x = SX(hx) - spr1.ctrlHeadR
  ctrl.y = SY(r) - 4
  ctrl.frame = frame

// the Controller walks in, lights the Bomb Cart's fuse with a flint, kicks it off and grabs his whip
func Sending()
  camX = 0
  SetShade(GRADE)
  var hx = C_STOP
  var row = pass.wheels
  var frame = C_STEP + 3
  ?f < T_BACK
    // walking in, whip trailing
    hx = -6 + math.RoundToInt((f - T_WALK) * (C_STOP + 6) / 120.0)
    frame = C_STEP + (f / 6) % 4
  :?f < T_BACK + 12
    // stepping back between the rails to the cart
    hx = C_STOP + (f - T_BACK) * (C_CART - C_STOP) / 12
    row = pass.wheels - (f - T_BACK) * 3 / 12
    frame = C_STEP + (f / 4) % 4
  :
    hx = C_CART
    row = pass.wheels - 3
    ?f >= T_REACH & f < T_STRIKE
      frame = C_REACH
    :?f >= T_STRIKE & f < T_FUSE
      frame = C_STRIKE + ((f - T_STRIKE) / 5) % 2
    :?f >= T_FUSE & f < T_KICK
      frame = C_JERK
    :?f >= T_KICK & f < T_PICK
      frame = C_KICK
    :?f >= T_PICK & f < T_PICK + 16
      frame = C_PICKUP
  CtrlAt(hx, row, frame)
  ?f = T_STRIKE | f = T_STRIKE + 10 | f = T_STRIKE + 20
    play stone_throw_hit @160 + (f - T_STRIKE) * 3 / 2@
  :?f = T_ROLL
    play bomb_cart_move 90
  Rolling()


// the Bomb Cart, its lit fuse and glow, and its smoke
func Rolling()
  var bxf = BombX() - camX
  ?f >= T_CHASE & f < T_BOOM
    // placed off him, not the rounded camera, so it doesn't jitter back a cell when the camera steps
    bxf = BombX() - PusherX() - 7 + cx
  var sx = math.RoundToInt(bxf)
  Show(bomb, true)
  bomb.x = sx
  bomb.y = SY(pass.trackRow - 5)
  bomb.frame = 0
  ?f >= T_ROLL
    bomb.frame = (f / 3) % 2
  ?f >= T_FUSE
    Show(fuse, true)
    fuse.frame = (f / 2) % 4
    fuse.x = sx + 2
    fuse.y = SY(pass.trackRow - 6)
    Light(fuseHalo, jobs.Count() = 0, sx + 3, SY(pass.trackRow - 6))
    for i = 0 .. 2
      var sp = sparks[i]
      var life = 9 + i * 3
      var age = (f + i * 5) % life
      Show(sp, f < T_BOOM)
      sp.frame = (f / 2 + i) % 4
      // rounded once so it can't step back against the cart
      sp.x = math.RoundToInt(bxf + 3 + (i - 1) * age * 0.35)
      sp.y = SY(pass.trackRow - 7) - math.RoundToInt(age * 0.45)
      sp.color = "#ffd27a"
      ?age > life - 4
        sp.color = "#e0461c"
    for i = 0 .. 1
      var p = (f + i * 20) % 40
      var s = smoke[i]
      Show(s, p < 20 & f - p >= T_FUSE)
      s.frame = p / 4
      // each puff stays where the cart left it and drifts back
      var px = BombXAt(f - p) + 1 - p * 0.05
      s.x = SX(math.RoundToInt(px))
      ?f >= T_CHASE & f < T_BOOM
        s.x = math.RoundToInt(px - PusherX()) - 7 + cx
      s.y = SY(pass.trackRow - 12)


// following the Pusher while the Bomb Cart comes up behind, passes him and races to the end
func Chase()
  var bx = PusherX()
  camX = math.RoundToInt(bx) + 7 - cx
  SetShade(GRADE)
  Show(cart, true)
  Show(body, true)
  cart.x = cx - 4
  cart.y = SY(pass.wheels) - 2
  body.x = cx - 7
  body.y = SY(pass.wheels) - 2
  body.frame = (f / 5) % 4
  Rolling()
  ?f = T_CHASE + 20
    play bomb_cart_move 80


// the blast and after: his cart wrecked, the chest and the sword, the fight, then into the crack
func After()
  var b = f - T_BOOM
  camX = camB
  var lit = jobs.Count() = 0

  // the flash and the shake
  Flash(IMax(0, 14 - b) / 14.0)
  ?b < 14 & opt.shake
    var amp = 1
    ?b < 6
      amp = 2
    root.x = ((b % 2) * 2 - 1) * amp
    root.y = (((b + 1) % 2) * 2 - 1) * amp / 2
  ?b = 0
    play bomb_cart_explosion
  :?b = 4
    play mushroom_boss_fat_slam 70
  :?b = 12
    play stone_throw_hit 55
  :?b = 20
    play stone_throw_hit 70

  // the blast itself, then smoke
  Show(blast, b < 40)
  blast.x = SX(pass.deadX - 9)
  blast.y = SY(pass.trackRow - 7)
  blast.frame = IMin(5, b / 7)
  ?b < 2
    blast.frame = 0

  // what it opened: the rock, the floor, the heat coming up
  Show(holeOv, true)
  holeOv.x = SX(pass.holeX)
  holeOv.y = SY(pass.holeY)
  Show(crackOv, true)
  crackOv.x = SX(pass.crackX)
  crackOv.y = SY(pass.crackY)
  Show(lipOv, true)
  lipOv.x = SX(pass.crackX)
  lipOv.y = SY(pass.crackY)
  var dim = (f / 4) % 3 = 0
  Light(crackHalo[0], lit & !dim, SX(pass.crackX + 7), SY(pass.crackY + 1))
  Light(crackHalo[1], lit & dim, SX(pass.crackX + 7), SY(pass.crackY + 1))

  // debris flying out and landing
  for i = 0 .. 7
    var d = debris[i]
    Show(d, true)
    d.frame = i % 5
    var t = b
    var y = 20 + DEB_VY[i] * t + 0.06 * t * t
    var x = pass.deadX + DEB_VX[i] * t
    ?y >= DEB_FLOOR[i] & t > 4
      // landed, stays where it came down
      var tl = (-DEB_VY[i] + math.Sqrt(DEB_VY[i] * DEB_VY[i] + 0.24 * (DEB_FLOOR[i] - 20))) / 0.12
      x = pass.deadX + DEB_VX[i] * tl
      y = DEB_FLOOR[i]
    d.x = SX(math.RoundToInt(x))
    d.y = SY(math.RoundToInt(y))

  // grit sifting down from the ceiling for a while
  for i = 0 .. 2
    var g0 = 20 + i * 17
    Show(specks[i], b >= g0 & b < g0 + 16)
    specks[i].frame = IMin(15, IMax(0, b - g0))
    specks[i].x = SX(pass.deadX - 14 + i * 9)
    specks[i].y = SY(10)

  for i = 0 .. 1
    var fl = fires[i]
    var burning = b >= 10 + i * 6 & b < 620 + i * 90
    Show(fl, burning)
    fl.frame = (f / 3 + i * 2) % 4
    fl.x = SX(pass.deadX - 6 + i * 9) - 1
    fl.y = SY(pass.trackRow + 1 - i) - 1
    var fd = (f / 4 + i) % 3 = 0
    Light(fireHalo[i * 2], lit & burning & !fd, fl.x + 1, fl.y + 1)
    Light(fireHalo[i * 2 + 1], lit & burning & fd, fl.x + 1, fl.y + 1)
    var sm = fireSmoke[i]
    var sa = (f + i * 23) % 48
    Show(sm, burning & sa < 32)
    sm.frame = IMin(4, sa / 7)
    sm.x = fl.x - sa / 8
    sm.y = fl.y - 3 - sa / 6
  Wreck(b)
  Chest(b, lit)
  Fight(b)
  Him(b)


// his cart tips over, a wheel rolls into the crack, ore spills. later it rocks when he heaves at it
func Wreck(b)
  var wx = camB + cx - 4
  Show(wreck, true)
  wreck.frame = 2
  var jolt = 0
  ?b >= 100 & b < 176 & (b / 6) % 4 = 2
    jolt = 1
  wreck.x = SX(wx - IMin(2, b / 3) + jolt)
  ?b < 6
    wreck.frame = IMin(1, b / 3)
  wreck.y = SY(pass.wheels) - 2
  Show(ore, b >= 6)
  ore.x = SX(wx - 4)
  ore.y = SY(pass.wheels + 1)
  // the wheel rolls off to the crack and drops in
  var rx = wx + 5 + math.RoundToInt(IMax(0, b - 4) * 0.5)
  var ry = pass.wheels
  ?rx >= pass.crackX + 4
    rx = pass.crackX + 4
    ry = pass.wheels + 1 + (b - 4 - (pass.crackX + 4 - wx - 5) * 2) / 2
  Show(wheel, b >= 4 & ry < pass.wheels + 6)
  wheel.frame = (b / 3) % 2
  wheel.x = SX(rx)
  wheel.y = SY(ry)


// the chest falls out of the rock, the lid flies off, glints inside and then on the sword

  Show(chest, true)

// the horn t frames after the blow: over his shoulder to the floor, then a bounce
func HornX(t)
  ?t < 30
    return HIT_X + 1 + (HORN_X - HIT_X - 1) * t / 30.0
  return HORN_X + 2 * IMin(8, t - 30) / 8.0

func HornY(t)
  ?t < 30
    var p = t / 30.0
    return pass.wheels - 2 + 3 * p - 7 * math.Sin(p * 3.14159)
  return pass.wheels + 1 - math.Sin(IMin(8, t - 30) / 8.0 * 3.14159)

// down the crack: the camera falls with him, embers rushing up, then the dark
func Falling()
  var t = f - T_BOOM - T_FALL
  camX = camB
  root.x = 0
  root.y = 0
  Flash(0)
  // the camera falls with him, then slows and stops as he drops away, scene 3 picks it up from there
  var fall = 0.5 * t + 0.008 * t * t
  var cam = fall
  var s = t - 50
  ?s > 0
    var v = 1.3
    var k = 0.04
    var sk = math.Min(s * 1.0, v / k)
    cam = 0.5 * 50 + 0.008 * 50 * 50 + v * sk - 0.5 * k * sk * sk
  var depth = math.RoundToInt(cam)
  var below = math.RoundToInt(fall - cam)
  Show(shaftArt, true)
  shaftArt.x = cx - shaft.w / 2
  shaftArt.y = -depth
  Show(faller, below < H - cy + 4)
  faller.frame = 5
  faller.x = cx - 2 + (t / 8) % 2
  faller.y = cy - 3 + below
  for i = 0 .. 3
    var e = embers[i]
    Show(e, true)
    e.frame = (t / 2 + i) % 4
    e.x = cx - 6 + i * 4 + math.RoundToInt(math.Sin(t * 0.2 + i) * 2)
    e.y = H - ((t * 2 + i * 11) % H)
  var lit = jobs.Count() = 0
  var hot = depth > 40 & below < 8
  Light(fallHalo[0], lit & !hot & below < H, cx, cy - 2 + below)
  Light(fallHalo[1], lit & hot, cx, cy - 2 + below)
  // the walls go dark behind him
  SetShade(color.Lerp(GRADE, "#5a4c44", Prog(T_BOOM + T_FALL + 60, T_END)))
  ?t = 2
    play fire_geyser 70

// ---------------------------------------------------------------- the film calls these

// a scene built ahead waits off screen and the game still draws its canvases, so zero their width till it's on
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so the lights are ready, parked off to the side (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

// take its picture off the screen (another scene is playing)
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start, for watching again
func Reset()
  f = 0
  over = false
  camB = 0
