// ---------------------------------------------------------------- building

func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// Canvases top out at 100 x 27 cells, so screen-sized ones are tiled.
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ramp(u, n)
  return math.Clamp(u * 1.0 / n, 0.0, 1.0)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func AnimIn(p, layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  p.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

// a light: additive canvas fading out from the center, filled a bit each frame
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ki.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// on its source this frame, or off with no width: the game draws every canvas cell, shown or not
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Show(c, on)
  c.visible = on

// walks out, plants the sword, stands at the lip, holds up the bud, looks over the edge,
// takes the sword back. from the lip on each pose is placed so his head stays put
func Him()
  Show(heroR, false)
  Show(heroL, false)
  ?f < T_PLANT
    HeroAt(X0 + V * IMax(0, f - T_WALK), (f / 6) % 4, false)
  :?f < T_PLANT + 10
    HeroAt(XP, 4, false)
  :?f < T_PLANT + 20
    HeroAt(XP, 5, false)
  :?f < T_PLANT + 28
    HeroAt(XP, 6, false)
  :?f < T_PLANT + 36
    HeroAt(XP, 7, false)
  :?f < T_PLANT + 51
    HeroAt(XP + (HEAD - 1 - XP) * Ramp(f - T_PLANT - 36, 15), 15 + (f / 4) % 2, false)
  :?f < T_JUNK[0]
    HeroAt(HEAD - 1, 7, false)
  :?f < T_HOLD
    HeroAt(HEAD - 1, PocketFrame(), false)
  :?f < T_CRACK + 12
    // up into the moonlight
    HeroAt(HEAD - 1, 11, false)
  :?f < T_CRACK + 26
    HeroAt(HEAD - 1, 7, false)
  :?f < T_UP
    HeroAt(HEAD - 1, 12, false)
  :?f < T_UP + 14
    HeroAt(HEAD - 1, 7, false)
  :?f < T_UP + 26
    HeroAt(HEAD - 5, 7, true)
  :?f < T_UP + 36
    HeroAt(HEAD - 5 - 2 * Ramp(f - T_UP - 26, 10), 15 + (f / 4) % 2, true)
  :?f < T_UP + 50
    HeroAt(HEAD - 7, 4, true)
  :
    HeroAt(HEAD - 3, 4, false)

// turning out his pockets: reach in, look, throw. the bud glows mid throw so he stops
func PocketFrame()
  for k = 0 .. 1
    var u = f - T_JUNK[k]
    ?u >= 0 & u < J_IN
      return 18
    ?u >= J_IN & u < J_IN + J_LOOK[k]
      return 17
    ?u >= J_IN + J_LOOK[k] & u < J_IN + J_LOOK[k] + J_THROW
      return 11
  var b = f - T_BUD
  ?b >= 0 & b < J_IN
    return 18
  ?b >= J_IN & f < T_GLOW - 4
    return 17
  ?f >= T_GLOW - 4 & f < T_GLOW + 16
    return 11
  ?f >= T_GLOW + 16
    return 17
  return 7

// junk in his hand, then thrown over the lip and down the drop
func Junk()
  for k = 0 .. 1
    var a = junk[k]
    var u = f - T_JUNK[k] - J_IN
    var thrown = u - J_LOOK[k]
    a.x = OFF
    ?u >= 0 & thrown < 0
      a.x = HEAD + 2
      a.y = SEAT_S - 1
    :?thrown >= 0
      var y = SEAT_S - 2 - 0.9 * thrown + 0.07 * thrown * thrown
      ?y < H - lb
        a.x = HEAD + 2 + math.RoundToInt(0.75 * thrown)
        a.y = math.RoundToInt(y)

// his sword, stuck in the rock until he takes it back
func Sword()
  Show(sword, f >= T_PLANT + 28 & f < T_UP + 36)
  sword.x = SWORD_X
  sword.y = SEAT_S - 1

// his breath in the cold, every so often while he stands there
func Breath()
  var u = (f - T_SIT - 20) % 80
  var on = ((f >= T_SIT + 20 & f < T_CRACK) | f >= T_UP + 50) & u < 24
  // his head's a couple of columns back once he's turned
  var hc = HEAD
  ?f >= T_UP + 50
    hc = HEAD - 2
  Show(breath, on)
  ?on
    breath.x = hc + 1 + math.RoundToInt(u * 0.15)
    breath.y = SEAT_S - 2 - math.RoundToInt(u * 0.08)
    breath.frame = IMin(2, u / 8)
    breath.color = color.Lerp("#8a949a", "#000000", Ramp(u - 8, 16))

// the bud: held out, flashes white mid throw, then up to the moon where it opens
func Bloom()
  var on = f >= T_BUD + J_IN & f < T_CRACK + 14
  Show(bloom, on)
  bloom.color = color.Lerp("#ffffff", "#c8e0b0", Ramp(f - T_GLOW, 14))
  ?on & f < T_HOLD
    // stem just past his hand so the blank beside it stays off his head
    bloom.x = HEAD - 3
    bloom.y = SEAT_S - 6
    ?f >= T_GLOW - 4 & f < T_GLOW + 16
      bloom.x = HEAD - 4
      bloom.y = SEAT_S - 7
    bloom.frame = (f / 10) % 2
  :?on
    // its stem in his raised hand
    bloom.x = HEAD - 4
    bloom.y = SEAT_S - 7
    bloom.frame = (f / 10) % 2
    ?f >= T_OPEN & f < T_OPEN + 6
      bloom.frame = 2
    :?f >= T_OPEN + 6
      bloom.frame = 3
  for k = 0 .. 3
    var u = f - T_OPEN - 4 - k * 9
    var a = kis[k]
    Show(a, u >= 0 & u < 60)
    ?u >= 0 & u < 60
      a.x = HEAD - 1 + k - math.RoundToInt(math.Sin(u * 0.1 + k) * 0.6)
      a.y = SEAT_S - 7 - math.RoundToInt(u * 0.12)
      a.frame = 1
      a.color = color.Lerp(KI, "#000000", Ramp(u - 20, 40))

// the wide shot: him as a speck (lit a little so the eye finds him) up the trail, and the
// gold hauled straight up the face a stretch at each crack, getting there first
func Wide(u, lit)
  var n = mtn.trailX.Count()
  // steady, then faster: once the gold's past him he runs
  var q = math.Clamp((u - 10) / 210.0, 0.0, 1.0)
  var p = q * 0.75
  ?q > 0.6
    p = 0.45 + (q - 0.6) * 1.375
  var hx = 0.0
  var hy = 0.0
  ?p < 0.62
    var q = p / 0.62
    hx = mtn.trailX[n - 3] + (mtn.trailX[n - 2] - mtn.trailX[n - 3]) * q
    hy = mtn.trailY[n - 3] + (mtn.trailY[n - 2] - mtn.trailY[n - 3]) * q
  :
    var q = (p - 0.62) / 0.38
    hx = mtn.trailX[n - 2] + (mtn.trailX[n - 1] - mtn.trailX[n - 2]) * q
    hy = mtn.trailY[n - 2] + (mtn.trailY[n - 1] - mtn.trailY[n - 2]) * q
  wSpeck.x = math.RoundToInt(hx) - WIDE_MX
  wSpeck.y = math.RoundToInt(hy) - WIDE_MY
  wSpeck.frame = (f / 20) % 2
  Light(wSpeckGlow, lit, wSpeck.x, wSpeck.y)
  var g = 0.0
  for k = 0 .. GOLD_AT.Count() - 1
    g = g + Ease(Ramp(u - GOLD_AT[k], 16)) / GOLD_AT.Count()
  wGold.x = math.RoundToInt(GOLD_X0 + (GOLD_X1 - GOLD_X0) * g) - WIDE_MX
  wGold.y = math.RoundToInt(GOLD_Y0 + (GOLD_Y1 - GOLD_Y0) * g) - WIDE_MY
  wGold.frame = (f / 6) % 3
  Light(wGoldGlow, lit & wGold.y < H - lb, wGold.x, wGold.y)
  // cloud drifting, aurora rippling like on the ledge
  wClouds.x = -WIDE_MX - 40 + f / 25
  for a : wAurora
    a.frame = (f / 10) % 8
  wAurora[0].color = "#0b3a18"
  wAurora[1].color = "#136328"
  wAurora[2].color = "#1f9a40"
  // a crack before each stretch
  for k = 0 .. GOLD_AT.Count() - 1
    ?u = GOLD_AT[k] - 2
      play controller_whip_hit @CRACK_P[k]@

// the gold glinting far below on the lower slopes, creeping up
func Glint()
  var on = f >= T_GLINT
  Show(glint, on)
  ?on
    var c = math.RoundToInt(30 - 6 * Ramp(f - T_GLINT, 260))
    glint.x = flank.x + c
    glint.y = flank.y + art.flankTop[c] - 1
    glint.frame = (f / 6) % 3
    glint.color = color.Lerp("#000000", "#d7af42", Ramp(f - T_GLINT, 20))

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  var lit = jobs.Count() = 0
  // stars twinkling, aurora rippling slow and faint
  for i = 0 .. stars.Count() - 1
    var s = stars[i]
    var b = 0.5 + 0.5 * math.Sin(f * 0.05 + i * 1.7)
    s.frame = (f / 40 + i) % 3
    s.color = color.Lerp("#2a2e34", "#d0d4dc", b)
  var af = (f / 10) % 8
  auroraHi.frame = af
  auroraMid.frame = af
  auroraLo.frame = af
  auroraHi.color = "#1f9a40"
  auroraMid.color = "#136328"
  auroraLo.color = "#0b3a18"
  Him()
  Sword()
  Breath()
  Junk()
  Bloom()
  Glint()
  Light(moonGlow, lit, W - 11, 5)
  // the bud's light: a flash when it glows, flickering while he looks, bright as it opens
  var bx = bloom.x + 4
  var by = bloom.y + 1
  var bright = (f >= T_GLOW & f < T_GLOW + 12) | (f >= T_OPEN & f < T_OPEN + 40)
  var faint = (f >= T_GLOW + 12 & f < T_OPEN & ((f / 14) % 3 > 0 | f >= T_HOLD)) | (f >= T_OPEN + 40 & f < T_OPEN + 80)
  Light(budGlow, lit & bright, bx, by)
  Light(budDim, lit & faint, bx, by)
  Light(goldGlow, lit & f >= T_GLINT + 20, glint.x + 1, glint.y)
  // the wide shot, over everything
  ?f >= T_WIDE
    wide.x = 0
    Wide(f - T_WIDE, lit)
  :
    wide.x = OFF
    wSpeckGlow.x = OFF
    wGoldGlow.x = OFF
  // cloud sea drifting far below
  cloud.x = LIP_S - 10 - f / 40
  // snow blowing past the ledge
  for i = 0 .. 7
    var s = flakes[i]
    s.x = math.RoundToInt(Wrap((i * 53) % 97 - f * 0.25 + math.Sin(f * 0.05 + i) * 1.5, W + 4)) - 2
    s.y = math.RoundToInt(Wrap((i * 37) % 33 + f * 0.18, H + 4)) - 2
    s.frame = i % 3

  card.color = Fade(T_CARD, T_CARD + 25, T_CARD_OUT, T_CARD_OUT + 25)
  Show(card, f < T_CARD_OUT + 26)
  // comes apart into Ki for scene 10
  ?f >= T_END - ki.OUT_T
    ki.At(veil, f - T_END)
    ?f = T_END - ki.OUT_T
      play elementalist_aether_blink 90
  :
    ki.Off(veil)
  // no fade either end: cut in from scene 8, Ki out into scene 10
  SetShade(GRADE_H)
  Sfx()


// ---------------------------------------------------------------- the film calls these

// the game draws every canvas cell even off screen, so they get no width while parked
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so its lights are ready, parked off screen (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

// off the screen while another scene plays
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start, for watching again
func Reset()
  f = 0
  over = false
