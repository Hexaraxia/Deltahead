func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  ?layers[0].Count() > 2
    return MakeGroup(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a


// big art as one anim per color, each cropped to its own cells, in a see-through panel.
// the game walks every cell of an anim whenever any of it shows, so this keeps it cheap
func MakeGroup(layers, x, y)
  var g = ui.AddPanel()
  root.Add(g)
  g.style = invis
  Place(g, x, y)
  for l : layers
    var a = ui.AddAnim(l[1])
    g.Add(a)
    Place(a, l[2], l[3])
    a.color = l[0]
  return g

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// canvases top out at 100 x 27 cells, so screen sized ones get tiled
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

// a light: an additive canvas fading out from the center, filled a bit each frame
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ki.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ramp(u, n)
  return math.Clamp(u * 1.0 / n, 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

// seeded so it plays the same every time
func Rnd()
  seed = (seed * 75 + 74) % 65537
  return seed / 65537.0

// a new speed each time a flake wraps, within its layer's range
func SnowVel(i)
  var l = snL[i]
  snVX[i] = SN_VX0[l] + (SN_VX1[l] - SN_VX0[l]) * Rnd()
  snVY[i] = SN_VY0[l] + (SN_VY1[l] - SN_VY0[l]) * Rnd()

// row of something on the Stream from its place along the flow (held in frame while we ride up)
func ConvY(c)
  ?f < T_UP0
    return CV_A0 + c - f / 6
  ?f < T_UP1
    return CV_A0 + c - T_UP0 / 6 - math.RoundToInt(D0 * (low - camY) / ((low - high) * 1.0))
  return CV_A0 + c - (f - (T_UP1 - T_UP0)) / 6 - D0

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

// lights that are off get zero width too, the game draws every canvas wherever it is
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

// which color the ∆'s light is on
func Hue()
  return (f / 10) % 8

func HeroX()
  var x0 = art.beamX - (cx + 8) - 4
  ?f < T_WALK
    return x0 * 1.0
  return math.Min(X_STOP * 1.0, x0 + V * (f - T_WALK))


// ---------------------------------------------------------------- him

// walks out to the gap, leans out to watch, straightens as the Delta passes (his ∆ answers), steps to the edge
func Him()
  var hx = math.RoundToInt(HeroX())
  ?f >= T_EDGE
    hx = hx + 1
  var fr = 4
  ?(f >= T_WALK & f < T_STOP) | (f >= T_EDGE & f < T_EDGE + 8)
    fr = (f / 5) % 4
  :?f >= T_LEAN & f < T_STRAIGHT
    fr = 5
  var here = f >= T_WALK
  Show(hero, here)
  Show(head, here)
  hero.x = SX(hx) - 1
  hero.y = SY(art.feet) - 3
  hero.frame = fr
  // the sword goes behind a tower like he does (3 either side of its middle)
  var behind = false
  for t : art.towerX
    ?math.Abs(hx - 1 + art.swordCol[fr] - t) <= 3
      behind = true
  Show(sword, here & !behind)
  sword.x = hero.x
  sword.y = hero.y
  sword.frame = fr
  head.x = hero.x + art.headX[fr]
  head.y = hero.y + art.headY[fr]
  head.color = DELTA_C
  ?here & jobs.Count() = 0
    Light(headGlows[Hue()], true, head.x, head.y)
  var u = f - T_STRAIGHT
  Show(answer, u >= 0 & u < 12)
  answer.x = head.x - 4
  answer.y = head.y - 2
  answer.frame = IMin(2, IMax(0, u) / 4)
  answer.color = DELTA_C


// ---------------------------------------------------------------- the Stream

// the beam's light, the game's lines and motes, our streaks, and everything it carries up, Delta last
func Stream()
  var lit = jobs.Count() = 0
  var bx = SX(art.beamX)
  Light(beamGlow[0], lit, bx, SY(12))
  Light(beamGlow[1], lit, bx, SY(38))
  Light(beamGlow[2], lit, bx, SY(64))
  // the game's Stream is on a front layer that slides faster than the camera (still once we're at it)
  var px = bx - math.FloorToInt((camX + cx + 8 - art.beamX) * 0.75)
  Light(beamGlow[3], lit, px, SY(art.gameY0 + 8))
  var ry = ConvY(RIDE_C)
  Light(flare, lit & opt.flashes & f >= T_UP0 & f < T_UP1 & math.Abs(ry + 1 - art.medalY) <= 2, bx, SY(art.medalY))
  var gy = SY(art.gameY0 + 20) - 26
  var fr = (2 * f) / 3
  for k = 0 .. gameStream.Count() - 1
    var n = LINE_N[k]
    var a = gameStream[k]
    a.x = px + n
    ?n < 0
      a.x = px + 2 * n
    a.y = gy
    a.frame = (LINE_F0[k] + fr) % 30
  for k = 0 .. 8
    var n = k - 4
    var m = gameMotes[k]
    m.x = px + n
    ?n < 0
      m.x = px + 2 * n
    m.y = gy
    m.frame = (MOTE_F0[k] + fr) % 15
  // the silent going up behind its lines
  var stripTop = art.gameY0 - 21
  var stripFoot = art.gameY0 + 23
  var sk = (STRIP_P + f / 6) % 45
  for j = 0 .. 4
    var b = backFigs[j]
    var r = art.figR[j] - sk
    ?r < 0
      r = r + 45
    b.x = px + art.figX[j]
    b.y = SY(stripTop + r)
    Show(b, f < T_UP0 & stripTop + r + art.figH[j] - 1 <= stripFoot)
  // our streaks, in screen space so they never slide down as the camera goes up
  for k = 0 .. streaks.Count() - 1
    var s = streaks[k]
    s.x = bx - art.beamW / 2 + (k * 5) % art.beamW
    s.y = math.RoundToInt(Wrap(ST_SEED[k] - f * ST_V[k], H + 4)) - 2
    s.frame = (k + f / 6) % 3
    s.color = "#c8ffd8"
    ?k % 3 = 1
      s.color = "#5ab07a"
    Show(s, camY < 56)
  // riding up with us, then on past ours with our chests mixed in. each one only shows once it's
  // been in the strip or below the picture, so nothing pops in from nowhere
  var u = Prog(T_UP0, T_UP1)
  var dOn = false
  for i = 0 .. conv.Count() - 1
    var a = conv[i]
    var y = ConvY(convC[i])
    ?f = T_UP0
      convArm[i] = y >= stripTop & y + convH[i] - 1 <= stripFoot
    :?f > T_UP0 & !convArm[i] & SY(y) >= H
      convArm[i] = true
    var on = f >= T_UP0 & convArm[i] & SY(y) < H & SY(y) + convH[i] > 0
    Show(a, on)
    a.x = px + convX[i]
    a.y = SY(y)
    ?convFig[i]
      a.color = color.Lerp(G_WHITE, BODY_GRAY, u)
    ?i = conv.Count() - 1
      dOn = on
  delta.color = DELTA_C
  ?lit & dOn
    Light(chestGlows[Hue()], true, delta.x + 2, delta.y + 1)

// lanterns flickering and lighting the deck, pennants stirring
func Ours()
  var lit = jobs.Count() = 0
  for k = 0 .. flames.Count() - 1
    var fl = flames[k]
    fl.x = SX(art.lampX[k])
    fl.y = SY(art.lampY[k])
    fl.frame = ((f + k * 5) / 4) % 4
    Light(lampGlow[k], lit & fl.x > -6 & fl.x < W + 6 & fl.y > -3 & fl.y < H + 3, fl.x, fl.y)
  for k = 0 .. flags.Count() - 1

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  var hx = math.RoundToInt(HeroX())
  // camera: start of the game's bridge, along to the Stream, eased up to ours, then down to the deck
  var top = art.feet - (cy + 5)
  low = art.gameY0 + 11 - (cy + 5)
  var startX = art.gameX0 - 57
  var streamX = art.beamX - (cx + 8)
  camX = math.RoundToInt(startX + (streamX - startX) * Ease(Prog(T_ALONG0, T_ALONG1)))
  high = art.top - 2
  camY = math.RoundToInt(low + (high - low) * Ease(Prog(T_UP0, T_UP1)))
  ?f >= T_DOWN0
    camY = math.RoundToInt(high + (top - high) * Ease(Prog(T_DOWN0, T_DOWN1)))
  var down = camY - top
  // the sky comes down into the picture as we go up
  pSky.x = -10 - math.RoundToInt(camX * 0.05)
  pSky.y = -20 - down
  moon.y = 2 - down
  var af = (f / 6) % 8
  var cols = ["#136328", "#1f9a40", "#32da5a"]
  for k = 0 .. 2
    var a = aurora[k]
    a.x = -20 - math.RoundToInt(camX * 0.08)
    a.y = lb - down
    a.frame = af
    a.color = cols[k]
  gameA.x = SX(0)
  gameA.y = SY(art.gameTop)
  gameB.x = SX(art.half)
  gameB.y = SY(art.gameTop)
  // the game's blizzard: flakes wrap around the screen and only show below the cloud
  for i = 0 .. snow.Count() - 1
    snX[i] = snX[i] + snVX[i] / 30.0
    snY[i] = snY[i] + snVY[i] / 30.0
    var ix = math.RoundToInt(snX[i])
    var iy = math.RoundToInt(snY[i])
    ?ix < 0 | ix > W | iy < 0 | iy > H
      ?ix < 0
        snX[i] = snX[i] + W
        snY[i] = (H + 1) * Rnd()
      ?ix > W
        snX[i] = snX[i] - W
        snY[i] = (H + 1) * Rnd()
      ?iy < 0
        snX[i] = (W + 1) * Rnd()
        snY[i] = snY[i] + H
      ?iy > H
        snX[i] = (W + 1) * Rnd()
        snY[i] = snY[i] - H
      SnowVel(i)
    var sn = snow[i]
    sn.x = ix
    sn.y = iy
    Show(sn, ix >= 0 & ix < W & iy >= 0 & iy < H & iy + camY >= SNOW_TOP)
  cloudA.x = SX(-100)
  cloudA.y = SY(36)
  cloudB.x = SX(-160)
  cloudB.y = SY(46)
  pFar.x = -10 - math.RoundToInt(camX * 0.25)
  pFar.y = SY(28)
  // the cloud sea's top, drifting the way the camera goes
  mist.x = -math.RoundToInt(Wrap(30 + f / 20.0 + camX * 0.5, 360))
  mist.y = SY(31)
  mist2.x = mist.x + 360
  mist2.y = mist.y
  bridgeA.x = SX(0)
  bridgeA.y = SY(art.top)
  bridgeB.x = SX(art.half)
  bridgeB.y = SY(art.top)
  frontA.x = SX(0)
  frontA.y = SY(art.top)
  frontB.x = SX(art.half)
  frontB.y = SY(art.top)
  Ours()
  Stream()
  Him()
  // the place name like the game shows it (no stars here), once we're up at ours
  var on = f >= T_UP1 & f < T_UP1 + 220
  Show(nameText, on)
  nameText.color = color.Lerp("#000000", "#e7e3da", Prog(T_UP1, T_UP1 + 20) - Prog(T_UP1 + 190, T_UP1 + 220))
  card.color = Fade(T_CARD, T_CARD + 25, T_CARD_OUT, T_CARD_OUT + 25)
  Show(card, f >= T_CARD & f < T_CARD_OUT + 26)
  // scene 13's grade at first, cold in the blizzard, plain once we're above the cloud sea
  SetShade(color.Lerp(color.Lerp(GRADE_H, GRADE_LOW, Prog(0, 36)), NORMAL, Prog(T_UP0 + 60, T_UP0 + 150)))
  ?f < ki.IN_T + 1
    ki.At(veil, f + 1)
  :
    ki.Off(veil)
  Sfx()


// ---------------------------------------------------------------- the film calls these

// a scene built ahead waits off screen and the game still draws its canvases, so zero their width till it's on
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so the lights are ready, parked off to the side (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

func Reset()
  f = 0
  over = false
