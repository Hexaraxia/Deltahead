func RunJobs()
  look.Run()
  ki.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// real frame for story time S
func RealAt(S)
  var r = S
  for k = 0 .. MARKS.Count() / 2 - 1
    var s = MARKS[k * 2]
    var n = MARKS[k * 2 + 1]
    ?s < S
      ?n > 0
        r = r + n
      :
        r = r + IMin(-n, S - s)
  return r

// story time at real frame t
func Story(t)
  var cum = 0
  for k = 0 .. MARKS.Count() / 2 - 1
    var s = MARKS[k * 2]
    var n = MARKS[k * 2 + 1]
    ?t < s + cum
      return (t - cum) * 1.0
    ?n > 0
      ?t < s + cum + n
        return s * 1.0
      cum = cum + n
    :
      ?t < s + cum - 2 * n
        return s + (t - s - cum) * 0.5
      cum = cum - n
  return (t - cum) * 1.0

func Schedule()
  // run, then slow to a stop over STOP_T frames, landing exactly on P
  var tc = math.CeilToInt((P - X0) / V - STOP_T / 2.0)
  V = (P - X0) / (tc + STOP_T / 2.0)
  T_ARRIVE = tc + STOP_T
  T_END = RealAt(T_CUT) + 30
  for k = 0 .. SHAKES.Count() / 3 - 1
    SHK.Add(RealAt(SHAKES[k * 3]))
    SHK.Add(SHAKES[k * 3 + 1])
    SHK.Add(SHAKES[k * 3 + 2])
  for k = 0 .. IMPACTS.Count() / 2 - 1
    IMP.Add(RealAt(IMPACTS[k * 2]))
    IMP.Add(IMPACTS[k * 2 + 1])

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ramp(u, n)
  return math.Clamp(u * 1.0 / n, 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func Glide(u, u0, u1, a, b)
  return a + (b - a) * Ease(Ramp(u - u0, u1 - u0))

// 0 to 1 between story times a and b, fast then slowing (a throw)
func Out(st, a, b)
  var u = Ramp(st - a, b - a)
  return 1.0 - (1.0 - u) * (1.0 - u)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// off lights get no width too: the game draws every canvas cell, shown or not
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

func Feet(x)
  return world.feet0 + (world.feet1 - world.feet0) * x * 1.0 / world.ww

func Row(x)
  return math.FloorToInt(Feet(math.RoundToInt(x)))

// runs in from the left and stops where he'll fight
func HeroX()
  walking = f < T_ARRIVE
  var tc = T_ARRIVE - STOP_T
  ?f < tc
    return X0 + V * f
  ?f < T_ARRIVE
    var s = f - tc
    return X0 + V * tc + V * s - V * s * s / (2.0 * STOP_T)
  return P * 1.0

func Cross(sp, st, x)
  return sp < x & st >= x

func ShakeNow()
  var amp = 0
  ?!opt.shake
    return 0
  for k = 0 .. SHK.Count() / 3 - 1
    ?f >= SHK[k * 3] & f < SHK[k * 3] + SHK[k * 3 + 1]
      amp = IMax(amp, SHK[k * 3 + 2])
  return amp

func ImpactNow()
  // impact frames count as a flash, so skip them with flashes off
  ?!opt.flashes
    return false
  for k = 0 .. IMP.Count() / 2 - 1
    ?f >= IMP[k * 2] & f < IMP[k * 2] + IMP[k * 2 + 1]
      return true
  return false

func Seq(u, holds, base, rate)
  var s = u / rate
  for i = 0 .. holds.Count() - 1
    ?s < holds[i]
      return base + i
    s = s - holds[i]
  return base + holds.Count() - 1

func OnPath(a, x, w, h)
  Show(a, true)
  var ix = math.RoundToInt(x)
  a.x = ix - camX
  a.y = SY(math.FloorToInt(Feet(ix + w / 2))) - (h - 1)

func Walk(t, t0, t1, x0, x1)
  return x0 + (x1 - x0) * Ramp(t - t0, t1 - t0)

func Hit(k, x, y, u, col)
  ?u >= 0 & u < 9
    var a = hits[k]
    Show(a, true)
    a.x = math.RoundToInt(x) - camX - 1
    a.y = SY(y) - 1
    a.frame = math.FloorToInt(u) / 3
    a.color = col

func Burst(k, x, y, u, col)
  ?u >= 0 & u < 12
    var a = bursts[k]
    Show(a, true)
    a.x = math.RoundToInt(x) - camX - 4
    a.y = SY(y) - 2
    a.frame = IMin(2, math.FloorToInt(u) / 4)
    a.color = col
    ?u < 5
      Light(burstGlow, jobs.Count() = 0 & opt.flashes, math.RoundToInt(x) - camX, SY(y))

func Puff(k, x, y, u)
  ?u >= 0 & u < 16
    var a = puffs[k]
    Show(a, true)
    a.x = math.RoundToInt(x) - camX - 6
    a.y = SY(y) - 3
    a.frame = IMin(3, math.FloorToInt(u) / 4)

func Lines(k, x, y, dir, on)
  ?on
    var a = lines[k]
    Show(a, true)
    a.x = math.RoundToInt(x) - camX
    ?dir > 0
      a.x = a.x - 8
    a.y = SY(y) - 1
    a.frame = (f / 2) % 2

func Mote(k, x, y, u, col)
  ?u >= 0 & u < 30
    var a = motes[k]
    Show(a, true)
    a.x = math.RoundToInt(x) - camX
    a.y = SY(y - u * 0.15)
    a.frame = IMin(2, math.FloorToInt(u) / 10)
    a.color = color.Lerp(col, "#000000", Ramp(u - 12, 18))


// ---------------------------------------------------------------- him

func Him()
  var frame = 4
  ?walking
    // running, pulling up, then catching his breath
    frame = 24 + (f / 3) % 4
    ?f >= T_ARRIVE - STOP_T
      frame = 16
  :?f < T_ARRIVE + BREATH_T
    frame = 9
  ?hFrame >= 0
    frame = hFrame
  var hx = heroX + hDX
  var r = math.FloorToInt(Feet(math.RoundToInt(hx)))
  var b = heroR
  ?hLeft
    b = heroL
  Show(b, true)
  var sx = math.RoundToInt(hx) - camX
  var sy = SY(r) - 3 + math.RoundToInt(hDY)
  b.x = sx - 1
  ?hLeft
    b.x = sx - 4
  // dash and run frames have the blade out behind him, so shift them a column
  ?frame = 13 | frame >= 24
    b.x = b.x - 1
    ?hLeft
      b.x = b.x + 2
  b.y = sy
  b.frame = frame
  ?frame = 10
    var sl = slashR
    sl.x = sx + 4
    ?hLeft
      sl = slashL
      sl.x = sx - 7
    Show(sl, true)
    sl.y = sy
  for k = 7 .. 1
    ghX[k] = ghX[k - 1]
    ghY[k] = ghY[k - 1]
    ghF[k] = ghF[k - 1]
  ghX[0] = b.x + shX
  ghY[0] = sy + shY
  ghF[0] = frame
  ?ghostOn & !hLeft
    for k = 0 .. 1
      var g = ghosts[k]
      Show(g, true)
      g.x = ghX[2 + k * 3] - shX
      g.y = ghY[2 + k * 3] - shY
      g.frame = ghF[2 + k * 3]
      g.color = color.Lerp("#7a7a7a", "#2a2a2a", k * 1.0)

func Swing(u)
  ?u < 0 | u >= 23
    return -1
  ?u < 6
    return 6
  ?u < 9
    return 10
  ?u < 17
    return 7
  return 8

func Quick(u)
  ?u < 0 | u >= 12
    return -1
  ?u < 2
    return 6
  ?u < 4
    return 10
  ?u < 9
    return 7
  return 8

func DoSwing(u)
  var s = Swing(u)
  ?s >= 0
    hFrame = s

func DoQuick(u)
  var s = Quick(u)
  ?s >= 0
    hFrame = s

// the Controller: walk in, idle, lash, backhand and swat, step in, final lash. gold with the odd shine
func Controller(s, st)
  var x = Walk(st, 170, 230, P + 50, P + 20)
  x = Walk(st, 744, 784, x, P + 6)
  var fr = 0
  ?s >= 240 & ((s + 30) % 96) < 8
    fr = 1
  ?(s >= 170 & s < 230) | (s >= 744 & s < 784)
    fr = 17 + (s / 6) % 4
  :?s >= 290 & s < 305
    fr = 21 + (s - 290) / 5
  :?s >= 305 & s < 305 + world.lashN
    fr = 32
  :?s >= 798 & s < 813
    fr = 21 + (s - 798) / 5
  :?s >= 813 & s < 813 + world.lashN
    fr = 32
  :?s >= 492 & s < 504
    fr = 26 + (s - 492) / 4
  :?s >= 504 & s < 516
    fr = 29 + (s - 504) / 4
  :?s >= 622 & s < 634
    fr = 26 + (s - 622) / 4
  :?s >= 634 & s < 646
    fr = 29 + (s - 634) / 4
  :?s >= 880 & s < 922
    fr = 2 + (s - 880) / 7
  :?s >= 922 & s < 934
    fr = 7
  :?s >= 934
    fr = IMin(23, 21 + (s - 934) / 6)
  OnPath(ctrl, x, world.ctrlW, world.ctrlH)
  ctrl.frame = fr
  var shine = math.Sin(f * 0.13)
  // kept for the whip too, reading .color back gives a Color not a string, so it cant be copied
  var col = color.Lerp(GOLD, SHINE, math.Clamp((shine - 0.75) * 4.0, 0.0, 1.0))
  ?Flash(st, CLANKS)
    col = "#ffffff"
  ctrl.color = col
  ?s < 150
    ctrl.x = OFF
  // the whip out along the snow from his fist
  var u = -1
  ?s >= 305 & s < 305 + world.lashN
    u = s - 305
  :?s >= 813 & s < 813 + world.lashN
    u = s - 813
  Show(lash, u >= 0)
  ?u >= 0
    var hx = math.RoundToInt(x) + world.ctrlHandX
    lash.x = hx - camX - world.lashOX

func Flash(st, list)
  ?!opt.flashes
    return false
  for t : list
    ?st >= t & st < t + 3

  return false

// the pack: two Eaters in front, one behind. elites, so they just keep coming back
func Pack(s, st)
  Eater(eaters[0], s, st, P + 16, -1, 242, B_MV, B_BITES, 31)
  Eater(eaters[1], s, st, P + 10, 0, 236, A_MV, A_BITES, 0)

// one Eater: buried till wake, then moves by mv (see A_MV) and bites. dy puts it a row further back
func Eater(a, s, st, x0, dy, wake, mv, bites, ph)
  var x = x0 * 1.0
  var fr = -1
  var cut = false
  for k = 0 .. mv.Count() / 4 - 1
    var t0 = mv[k * 4]
    var t1 = mv[k * 4 + 1]
    ?mv[k * 4 + 2] = 0
      x = Walk(st, t0, t1, x, P + mv[k * 4 + 3])
      ?s >= t0 & s < t1
        fr = s8.eWalk + (s / 4) % 4
    :
      x = x + mv[k * 4 + 3] * Out(st, t0, t1)
      ?s >= t0 & s < t0 + 12
        fr = s8.eWake + 3
        cut = s < t0 + 3
  ?fr < 0
    for b : bites
      ?s >= b & s < b + 22
        fr = Seq(s - b, E_BITE, s8.eBite, 1)
  ?fr < 0
    var u = (s + ph) % 70
    fr = s8.eBite + 13
    ?u < 4 | (u >= 8 & u < 12)
      fr = s8.eBite
    :?u < 8
      fr = s8.eBite + 1
  OnPath(a, x, s8.eW, s8.eH)
  a.y = a.y + dy
  a.color = GREEN
  ?cut
    a.color = "#ffffff"
  ?s < wake
    a.frame = s8.eWake
  :?s < wake + 15
    a.frame = s8.eWake + (s - wake) / 3
  :
    a.frame = fr

// his x offset through the fight
func HeroDX(st)
  var dx = Glide(st, 344, 354, 0, -5) + 5.0 * Ramp(st - 356, 4) + Ramp(st - 360, 4) + 2.0 * Ramp(st - 372, 4)
  dx = dx + 2.0 * Ramp(st - 380, 9) - 3.0 * Out(st, 409, 417) + 5.0 * Ramp(st - 418, 15)
  dx = dx + Glide(st, 452, 462, 0, 21) - Out(st, 469, 473) + 2.0 * Ramp(st - 485, 7)
  dx = dx - 34.0 * Out(st, 504, 522) - 5.0 * Out(st, 522, 536)
  dx = dx + 12.0 * Ramp(st - 574, 6) + 6.0 * Ramp(st - 578, 12) + 26.0 * Ramp(st - 592, 16)
  dx = dx - 44.0 * Out(st, 628, 654) - 2.0 * Ramp(st - 654, 8) - 3.0 * Out(st, 662, 674)
  dx = dx + 2.0 * Ramp(st - 732, 9) + 2.0 * Ramp(st - 772, 12) - 2.0 * Out(st, 833, 841)
  return dx

// spin attack
func Spin(u)
  ?u < 0 | u >= 17
    return -1
  ?u < 4
    return 17
  ?u < 7
    return 10
  ?u < 13
    return 7
  return 8

// the game draws every canvas cell even off screen, so they get no width while parked
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so its lights are ready, parked off screen (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

func Reset()
  f = 0
  over = false
