func MakeAnim(p, layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  p.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// real frame for story time S
func RealAt(S)
  var r = S
  for k = 0 .. MARKS.Count() / 2 - 1
    var s = MARKS[k * 2]
    var n = MARKS[k * 2 + 1]
    ?s < S
      ?n > 0
        r = r + n
      :
        r = r + IMin(-n, S - s)
  return r

// story time at real frame t
func Story(t)
  var cum = 0
  for k = 0 .. MARKS.Count() / 2 - 1
    var s = MARKS[k * 2]
    var n = MARKS[k * 2 + 1]
    ?t < s + cum
      return (t - cum) * 1.0
    ?n > 0
      ?t < s + cum + n
        return s * 1.0
      cum = cum + n
    :
      ?t < s + cum - 2 * n
        return s + (t - s - cum) * 0.5
      cum = cum - n
  return (t - cum) * 1.0

func Schedule()
  // real frames count from the cut, SKIP story frames in. no fade, it ends on the O and scene 13 cuts in
  T_END = RealAt(T_LAST + 140) - SKIP

    SHK.Add(SHAKES[k * 3 + 2])
  for k = 0 .. IMPACTS.Count() / 2 - 1
    IMP.Add(RealAt(IMPACTS[k * 2]) - SKIP)
    IMP.Add(IMPACTS[k * 2 + 1])

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ramp(u, n)
  return math.Clamp(u * 1.0 / n, 0.0, 1.0)

func Out(st, a, b)
  var u = Ramp(st - a, b - a)
  return 1.0 - (1.0 - u) * (1.0 - u)

func Walk(t, t0, t1, x0, x1)
  return x0 + (x1 - x0) * Ramp(t - t0, t1 - t0)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// off lights get no width too: the game draws every canvas cell, shown or not
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

func Feet(x)
  return world.feet0 + (world.feet1 - world.feet0) * x * 1.0 / world.ww

func Cross(sp, st, x)
  return sp < x & st >= x

func ShakeNow()
  var amp = 0
  ?!opt.shake
    return 0
  for k = 0 .. SHK.Count() / 3 - 1
    ?f >= SHK[k * 3] & f < SHK[k * 3] + SHK[k * 3 + 1]
      amp = IMax(amp, SHK[k * 3 + 2])
  return amp

func ImpactNow()
  // impact frames count as a flash, so skip them with flashes off
  ?!opt.flashes
    return false
  for k = 0 .. IMP.Count() / 2 - 1
    ?f >= IMP[k * 2] & f < IMP[k * 2] + IMP[k * 2 + 1]
      return true
  return false

func Seq(u, holds, base, rate)
  var s = u / rate
  for i = 0 .. holds.Count() - 1
    ?s < holds[i]
      return base + i
    s = s - holds[i]
  return base + holds.Count() - 1

func OnPath(a, x, w, h)
  Show(a, true)
  var ix = math.RoundToInt(x)
  a.x = ix - camX
  a.y = SY(math.FloorToInt(Feet(ix + w / 2))) - (h - 1)

func Hit(k, x, y, u, col)

    var a = hits[k]

    a.x = SX(x) - 1
    a.y = SY(y) - 1
    a.frame = math.FloorToInt(u) / 3

// when the k-th lash leaves his hand
func WhipAt(k)
  return LASH[k] + 3 * WIND[k]


// ---------------------------------------------------------------- the body

// a few steps nearer after each lash, then pushing him at cart pace, a last shove at the edge
func BodyX(st)
  var x = 47.0
  for k = 0 .. WALKS.Count() / 3 - 1
    x = Walk(st, WALKS[k * 3], WALKS[k * 3 + 1], x, WALKS[k * 3 + 2])
  ?st > T_PUSH
    x = x + PUSH_V * (IMin(st, T_EDGE) - T_PUSH)
  x = x + 2.0 * Out(st, T_EDGE, T_TIP)
  return x

// art.body frames: 0-3 walk with the sword, 4 stand with it, 5 hop, 6 raised, 7 plunge,
// 8 crouched over it, 9 stand empty-handed, 10-13 walk empty-handed, 14-17 push
func BodyFrame(s, st)
  var fr = 6
  ?s >= 290
    fr = 4
  for k = 0 .. WALKS.Count() / 3 - 1
    ?s >= WALKS[k * 3] & s < WALKS[k * 3 + 1]
      fr = (s / 5) % 4
  for k = 0 .. 5
    var h = WhipAt(k) + 7
    ?s >= h & s < h + 16
      fr = 5
  // the last hop is higher, down on the whip point first
  ?s >= 600 & s < 608
    fr = 5
  :?s >= 608 & s < 614
    fr = 7
  :?s >= 614 & s < 628
    fr = 8
  :?s >= 628 & s < 636
    fr = 9
  :?s >= 636 & s < T_PUSH
    fr = 10 + (s / 5) % 4
  :?s >= T_PUSH & s < T_TIP + 6
    fr = 14 + (s / 7) % 4
  :?s >= T_TIP + 6
    fr = 9
  return fr

func BodyDY(s, st)
  for k = 0 .. 5
    var h = WhipAt(k) + 7
    ?s >= h & s < h + 16
      return -2.0 * math.Sin(Ramp(st - h, 16) * 3.1416)
  ?s >= 600 & s < 614
    return -4.0 * math.Sin(Ramp(st - 600, 14) * 3.1416)
  return 0.0

func Body(s, st)
  var bx = BodyX(st)
  var ix = math.RoundToInt(bx)
  OnPath(body, ix, 1, 1)
  body.x = ix - camX - 1
  body.y = SY(math.FloorToInt(Feet(ix))) - 3 + math.RoundToInt(BodyDY(s, st))
  body.frame = BodyFrame(s, st)
  // the sword left stuck in the snow through the whip
  Show(sword, s >= 628)
  sword.x = SX(67)
  sword.y = SY(math.FloorToInt(Feet(67))) - 1


// ---------------------------------------------------------------- the Controller

// knocked back a step when the whip's cut, then pushed along by the body
func CtrlX(st)
  var x = 66.0 + Out(st, 614, 622)
  ?st > T_PUSH
    x = math.RoundToInt(BodyX(st)) - 11
  return x

func Controller(s, st)
  var x = CtrlX(st)
  var fr = 0
  ?s >= 200 & ((s + 10) % 96) < 8
    fr = 1
  var u = -1
  for k = 0 .. 6
    var w0 = LASH[k]
    var w1 = WhipAt(k)
    ?s >= w0 & s < w1
      fr = 21 + (s - w0) / WIND[k]
    :?s >= w1 & s < w1 + world.lashN
      fr = 32
      u = s - w1
  ?s >= 614
    fr = 32
    u = -1
  OnPath(ctrl, x, world.ctrlW, world.ctrlH)
  ctrl.frame = fr
  var shine = math.Sin(f * 0.13)
  // kept for the whip too, reading .color back gives a Color not a string, so it cant be copied
  var col = color.Lerp(GOLD, SHINE, math.Clamp((shine - 0.75) * 4.0, 0.0, 1.0))
  // over the edge and down into the crevasse, going dark
  ?s >= T_TIP
    var t = st - T_TIP
    ctrl.x = ctrl.x + math.RoundToInt(t * 0.12)
    ctrl.y = ctrl.y + math.RoundToInt(0.045 * t * t)
    col = color.Lerp(GOLD, "#1a1608", Ramp(t, 40))
    Show(ctrl, t < 48)
  ctrl.color = col
  Show(lash, u >= 0)
  ?u >= 0
    var hx = math.RoundToInt(x) + world.ctrlHandX
    lash.x = hx - camX - world.lashOX
    lash.y = ctrl.y + world.ctrlHandY - world.lashOY
    lash.frame = u
    lash.color = col
  // the cut end of the whip, left lying there
  Show(cutWhip, s >= 614)
  cutWhip.x = 66 + world.ctrlHandX - camX - world.lashOX
  cutWhip.y = SY(math.FloorToInt(Feet(77))) - (world.ctrlH - 1) + world.ctrlHandY - world.lashOY


// ---------------------------------------------------------------- the pack

// one Eater: moves, bites, then once the whip kills it sinks into the snow and fades like in scene 8
func Eater(a, s, st, x0, dy, mv, bites, dead, ph, k)
  var x = x0 * 1.0
  var fr = -1
  var cut = false
  for j = 0 .. mv.Count() / 4 - 1
    var t0 = mv[j * 4]
    var t1 = mv[j * 4 + 1]
    ?mv[j * 4 + 2] = 0
      x = Walk(st, t0, t1, x, mv[j * 4 + 3])
      ?s >= t0 & s < t1
        fr = s8.eWalk + (s / 4) % 4
    :
      x = x + mv[j * 4 + 3] * Out(st, t0, t1)
      ?s >= t0 & s < t1
        fr = s8.eWake + 3
        cut = s < t0 + 3
  ?fr < 0
    for b : bites
      ?s >= b & s < b + 22
        fr = Seq(s - b, E_BITE, s8.eBite, 1)
  ?fr < 0
    var u = (s + ph) % 70
    fr = s8.eBite + 13
    ?u < 4 | (u >= 8 & u < 12)
      fr = s8.eBite
    :?u < 8
      fr = s8.eBite + 1
  ?s >= dead + 70
    Show(a, false)
    return
  OnPath(a, x, s8.eW, s8.eH)
  a.y = a.y + dy
  a.color = GREEN
  a.frame = fr
  ?cut
    a.color = "#ffffff"
  ?s >= dead
    var u = s - dead
    a.frame = s8.eWake + IMax(0, 4 - u / 4)
    a.color = color.Lerp(GREEN, GONE, Ramp(u - 25, 45))
    Mote(k * 2, x + 5, a.y + camY + 2, u - 2, GREEN)
    Mote(k * 2 + 1, x + 8, a.y + camY + 3, u - 9, GREEN)

func Pack(s, st)

func Summit(s, st, sp)
  var bx = math.RoundToInt(BodyX(st))
  var n = bx - 60
  // camera starts where scene 10 left it, then follows the body's cell so it never steps back
  camX = IMax(60 - (cx - 14) + math.RoundToInt(n * 0.35), bx - (cx - 20))
  camY = math.FloorToInt(Feet(60)) - (cy + 3)
  var amp = ShakeNow()
  ?amp > 0
    var sx = [1, 0, -1, 0]
    var sy = [0, 1, 0, -1]
    var ph = f % 4
    camY = camY + sy[ph]
    camX = camX + sx[ph] * amp
  var impact = ImpactNow()
  var sea = cy - 3
  pSky.x = -10 - math.RoundToInt(camX * 0.05)
  var af = (f / 6) % 8
  var cols = ["#136328", "#1f9a40", "#32da5a"]

    a.x = -20 - math.RoundToInt(camX * 0.08)
    a.y = lb
    a.frame = af

  summit.y = cy - 14

  pBack.y = sea + 2 - math.RoundToInt((camY - (world.feet0 - (cy + 3))) * 0.6)
  // mist drifts the same way the camera moves, otherwise it wobbles
  mist.x = -math.RoundToInt(Wrap(30 + f / 20.0 + camX * 0.7, 360))

  mist2.x = mist.x + 360
  mist2.y = sea + 3

  crevasse.y = SY(0)

  ShowAll(bg, !impact)
  ShowAll(aurora, !impact)
  ShowAll(flakes, !impact)

  ShowAll(motes, false)
  ShowAll(puffs, false)
  ShowAll(hits, false)
  ShowAll(bursts, false)

  Pack(s, st)
  Body(s, st)

// ---------------------------------------------------------------- the frame

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  var st = Story(f + SKIP)
  var sp = Story(f - 1 + SKIP)
  var s = math.FloorToInt(st)
  // summit or Ring
  var cut = -1
  for k = 0 .. CUTS.Count() / 2 - 1
    ?s >= CUTS[k * 2] & s < CUTS[k * 2 + 1]
      cut = k
  var onRing = cut >= 0 | s >= T_LAST
  summitPanel.x = OFF
  ringPanel.x = OFF
  ?onRing
    ringPanel.x = 0
    Ring(s, st, cut)
  :
    summitPanel.x = 0
    Summit(s, st, sp)
  card.color = Fade(T_CARD, T_CARD + 25, T_CARD_OUT, T_CARD_OUT + 25)
  Show(card, f < T_CARD_OUT + 26)
  // no fade at either end
  ?onRing
    SetShade(GRADE_RING)
  :
    SetShade(GRADE_H)
  Sfx(sp, st, cut)


// ---------------------------------------------------------------- the film calls these

// the game draws every canvas cell even off screen, so they get no width while parked
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so its lights are ready, parked off screen (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

func Reset()
  f = 0
  over = false
