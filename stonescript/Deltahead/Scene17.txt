// ---------------------------------------------------------------- building

func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(p, layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  p.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(p, x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  p.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// Canvases top out at 100 x 27 cells, so screen-sized ones are tiled.
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(root, i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

// a light: additive canvas fading out from the center, filled a bit each frame
func QueueHalo(p, w, h, r, g, b, bgk)
  var c = MakeCanvas(p, OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ease(u)
  return u * u * (3.0 - 2.0 * u)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

// the game's rainbow at hue h (0 to 6) and brightness b, by hand so a card can fade
// it (#rain can't be dimmed). lightened halfway to white like the game's
func Rainbow(h, b)
  var k = math.FloorToInt(h) % 6
  var t = h - math.FloorToInt(h)
  var r = 1.0
  var g = t
  var bl = 0.0
  ?k = 1
    r = 1.0 - t
    g = 1.0
  :?k = 2
    r = 0.0
    g = 1.0
    bl = t
  :?k = 3
    r = 0.0
    g = 1.0 - t
    bl = 1.0
  :?k = 4
    r = t
    g = 0.0
    bl = 1.0
  :?k = 5
    r = 1.0
    g = 0.0
    bl = 1.0 - t
  return color.FromRGB(math.RoundToInt((127 + 128 * r) * b), math.RoundToInt((127 + 128 * g) * b), math.RoundToInt((127 + 128 * bl) * b))

// the unlock: the scroll unrolls in 8 frames like the game's, a ∆ drops onto the
// Pusher's head, then the line to wear it
func Unlock()
  var u = f - T_UNLOCK
  var on = u >= 0
  var w = UNROLL[IMax(0, IMin(UNROLL.Count() - 1, u))]
  // video's end: rolls shut the way it opened, contents gone first like the game's
  var shut = u - U_CLOSE
  ?opt.video & shut >= 0
    on = shut < UNROLL.Count()
    w = UNROLL[IMax(0, UNROLL.Count() - 1 - shut)]
  uBox.visible = on
  uBox.w = w
  var l = cx - w / 2
  var r = l + w - 1
  var top = H / 2 - 3 - 6
  var out = 0
  ?u >= UNROLL.Count() & u < UNROLL.Count() + 2
    out = 1
  for i = 0 .. 1
    var a = uRolls[i]
    Show(a, on)
    a.y = top - 3
    a.x = l - 3 - out
    ?i = 1
      a.x = r - 2 + out
  var open = u >= UNROLL.Count() & (!opt.video | shut < 0)
  ?boulderTier >= 0
    Boulder(u, open)
    return
  // two little hops
  var hop = 0
  ?(u >= U_HOP & u < U_HOP + 5) | (u >= U_HOP + 9 & u < U_HOP + 14)
    hop = 1
  Show(uBody, open)
  uBody.y = H / 2 - 3 - hop
  Show(uHead, open & u < U_ON)
  // the ∆ speeding up as it falls onto his head
  var d = u - U_FALL
  Show(uDelta, open & d >= 0)
  ?d >= 0 & u < U_ON
    var k = d * 1.0 / (U_ON - U_FALL)
    uDelta.y = math.RoundToInt(top + 1 + (H / 2 - 4 - top - 1) * k * k)
  :
    uDelta.y = H / 2 - 4 - hop
  Show(uName, open & u >= U_ON)
  Show(uNew, open & u >= U_ON)
  for i = 0 .. 5
    var sp = uSparks[i]
    var q = u - U_ON
    Show(sp, q >= 0 & q < 10)
    ?q >= 0 & q < 10
      var ang = i * 1.047
      sp.x = math.RoundToInt(cx + math.Cos(ang) * (1 + q * 0.2) * 2)
      sp.y = math.RoundToInt(H / 2 - 4 + math.Sin(ang) * (1 + q * 0.2))
      sp.color = Rainbow(i, 1.0 - q / 10.0)
  ?u = U_ON & !opt.video
    opt.See()
  Wear(u)

// the import lines under the scroll (fade out as the video's closes)
func Wear(u)
  for i = 0 .. uWear.Count() - 1
    var a = uWear[i]
    Show(a, u >= U_WEAR)
    var k = Prog(T_UNLOCK + U_WEAR, T_UNLOCK + U_WEAR + 25)
    ?opt.video
      k = k - Prog(T_UNLOCK + U_CLOSE, T_UNLOCK + U_CLOSE + 15)
    a.color = color.Lerp("#000000", uWearC[i], k)

// later watches: the Pusher rolling the boulder up the slope, then its name and line
func Boulder(u, open)
  for a : [bRock, bSlope, bBoulder, bPusher]
    Show(a, open)
  bBoulder.frame = (u / 3) % 16
  bPusher.frame = (u / 6) % 4
  Show(uName, open & u >= U_ON)
  Show(uNew, open & u >= U_ON)
  for i = 0 .. 5
    var sp = uSparks[i]
    var q = u - U_ON
    Show(sp, q >= 0 & q < 10)
    ?q >= 0 & q < 10
      var ang = i * 1.047
      sp.x = math.RoundToInt(cx - 2 + math.Cos(ang) * (3 + q * 0.3) * 2)
      sp.y = math.RoundToInt(H / 2 - 3 + math.Sin(ang) * (2 + q * 0.3))
      sp.color = Rainbow(i, 1.0 - q / 10.0)
  ?u = U_ON
    opt.See()
  Wear(u)

// the last card: a rock knocks STONE out and DELTA lights up in rainbow, a letter at a time
func Card2()
  var on = f >= T_CARD2 & f < T_CARD2_OUT + 26
  var col = Fade(T_CARD2, T_CARD2 + 25, T_CARD2_OUT, T_CARD2_OUT + 25)
  var b = Prog(T_CARD2, T_CARD2 + 25) - Prog(T_CARD2_OUT, T_CARD2_OUT + 25)
  var s = f - T_SMASH
  // the card jolts when hit (camera shake on only)
  var jx = 0
  ?opt.shake & s >= 0 & s < JOLT.Count()
    jx = JOLT[s]
  Show(c2Head, on)
  Show(c2Tail, on)
  c2Head.x = c2X + jx
  c2Tail.x = c2X + 26 + jx
  c2Head.color = col
  c2Tail.color = col
  for i = 0 .. 4
    var st = stone[i]
    var d = delta[i]
    ?s < 0
      Show(st, on)
      st.x = c2X + 21 + i
      st.y = c2Y
      st.color = col
      Show(d, false)
    :
      // each letter flung off, falling, going dark
      var u = s * 1.0
      st.x = math.RoundToInt(c2X + 21 + i + ((i - 2) * 0.32 + 0.12) * u)
      st.y = math.RoundToInt(c2Y - (0.55 + (i % 3) * 0.18) * u + 0.045 * u * u)
      st.color = color.Lerp("#e7e3da", "#000000", Prog(T_SMASH, T_SMASH + 32))
      Show(st, s < 32)
      Show(d, on & s >= 3 + i * 2)
      d.x = c2X + 21 + i + jx
      d.color = Rainbow(6 * Wrap(i / 10.0 - f * 0.4 / 30, 1.0), b)
  Rock()
  Debris()

// the rock t frames after the throw: in from the lower right onto STONE's O, then back off and down
func RockX(t)
  var n = T_SMASH - T_ROCK
  ?t <= n
    return math.RoundToInt(c2X + 57 - 34 * t * 1.0 / n)
  return math.RoundToInt(c2X + 23 + (t - n) * 1.0)

func RockY(t)
  var n = T_SMASH - T_ROCK
  ?t <= n
    var u = t * 1.0 / n
    return math.RoundToInt(c2Y + 9 - 9 * u - 24 * u * (1 - u))
  var v = (t - n) * 1.0
  // over the rest of the words, coming down past their end
  return math.RoundToInt(c2Y - 0.9 * v + 0.06 * v * v)

func Rock()
  var t = f - T_ROCK
  var n = T_SMASH - T_ROCK
  var on = t >= 0 & t < n + 45
  Show(rock, on)
  ?on
    rock.x = RockX(t)
    rock.y = RockY(t)
  for k = 0 .. 1
    var a = rockTrail[k]
    var tk = t - 1 - k
    Show(a, on & tk >= 0 & t <= n)
    ?on & tk >= 0
      a.x = RockX(tk)
      a.y = RockY(tk)
      a.color = color.Lerp("#000000", ROCK_C, 0.6 - k * 0.25)

// chips off the card, and glints round DELTA (kept small so they're not a flash)
func Debris()
  var s = f - T_SMASH
  for i = 0 .. 7
    var a = debris[i]
    Show(a, s >= 0 & s < 24)
    ?s >= 0 & s < 24
      var ang = i * 0.785 + 0.4
      var sp = 0.5 + (i % 3) * 0.2
      a.x = math.RoundToInt(c2X + 23 + math.Cos(ang) * sp * s * 1.6)
      a.y = math.RoundToInt(c2Y + (math.Sin(ang) * sp - 0.3) * s + 0.04 * s * s)
      a.color = color.Lerp("#a8a090", "#000000", s / 24.0)
  var gx = [c2X + 20, c2X + 26, c2X + 19, c2X + 27]
  var gy = [c2Y - 1, c2Y - 1, c2Y + 1, c2Y + 1]
  for i = 0 .. 3
    var g = glints[i]
    var q = s - 6 - i * 4
    Show(g, q >= 0 & q < 14)
    g.x = gx[i]
    g.y = gy[i]
    ?q >= 0 & q < 14
      g.color = Rainbow(i * 1.5, 1.0 - q / 14.0)

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

// off lights get no width too: the game draws every canvas cell, shown or not
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func ShowShot(p, on)
  ?on
    p.x = 0
  :
    p.x = OFF

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

func Hue()
  return (f / 10) % 8

// scene 11's last shot again: the drip, the O rolling into the cart, the cart rolling on

  camY = desc.viewCY - cy

  fBack.y = ring.backY - camY
  fMain.x = -camX
  fMain.y = -camY

  fFront.y = ring.frontY - camY

  fPost.y = SY(ring.postY)

  fBend.y = SY(ring.bendY)

  fFlame.y = SY(ring.lampY)

  Light(fLantern[1], lit & lv = 3, SX(ring.lampX), SY(ring.lampY))

    mo.y = SY(ring.lampY - 1 + math.RoundToInt(math.Sin(f * 0.019 + k * 1.3) * 2))

    s.x = SX(O_X - 1 + k * 2)
    s.y = SY(O_Y - 1)

  var ox = O_X
  var oy = O_Y
  ?f >= T_LAND
    ox = DROP_X[4] + roll
    oy = DROP_Y[4]
  :?f >= T_DROP
    var i = IMin(4, (f - T_DROP) / 3)
    ox = DROP_X[i]
    oy = DROP_Y[i]
  :?f >= T_HIT
    ox = O_X + 1 + math.RoundToInt(2 * Prog(T_ROLL, T_DROP) * Prog(T_ROLL, T_DROP))
  oldO.x = SX(ox)
  oldO.y = SY(oy)

// ---------------------------------------------------------------- the frame

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  var lit = jobs.Count() = 0
  var high = f < T_CUT
  var atRing = f >= T_RING & f < T_BLACK
  ShowShot(shotA, high)
  ?!high
    ShowShot(shotM, false)
    ShowShot(shotCb, false)
    ShowShot(shotCf, false)
    ShowShot(shotIb, false)
    ShowShot(shotIf, false)
  ShowShot(shotB, atRing)
  Show(pSky, high)
  Show(moon, high)
  ?high
    High(lit)
  :?atRing
    Ring(lit)
  // the light filling the screen as he goes up into it
  var pw = Prog(T_WASH, T_CUT)
  var w = 0.95 * pw * pw * pw
  // flashes off: just a little, and the picture goes dark so there's no white cut
  ?!opt.flashes
    w = 0.3 * pw * pw
  ?f >= T_CUT
    w = 0
  var col = color.FromRGB(math.RoundToInt(255 * w), math.RoundToInt(248 * w), math.RoundToInt(225 * w))
  ?!string.Equals(col, washCol)
    washCol = col
    for c : wash
      c.SetFG(col)
      c.SetBG(col)
  ?high & !opt.flashes
    SetShade(color.Lerp(NORMAL, "#000000", Prog(T_WASH + 10, T_CUT)))
  :?high
    SetShade(NORMAL)
  :?atRing
    SetShade(color.Lerp("#000000", GRADE_RING, Prog(T_RING, T_RING + 30)))
  :
    SetShade("#000000")
  card1.color = Fade(T_CARD1, T_CARD1 + 25, T_CARD1_OUT, T_CARD1_OUT + 25)
  Show(card1, f >= T_CARD1 & f < T_CARD1_OUT + 26)
  Card2()
  Unlock()
  for k = 0 .. 2
    var cr = credits[k]
    var at = T_CREDITS + k * CREDIT_GAP
    Show(cr, f >= at & f < T_CREDITS_OUT + 26)
    cr.color = color.Lerp("#000000", CREDIT_C[k], Prog(at, at + 25) - Prog(T_CREDITS_OUT, T_CREDITS_OUT + 25))
  Sfx()


// ---------------------------------------------------------------- the film calls these

// the game draws every canvas cell even off screen, so they get no width while parked
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built while scene 16 is on so its lights are ready, parked off screen (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

func Reset()
  f = 0
  over = false
