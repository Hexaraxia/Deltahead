// ---------------------------------------------------------------- building

func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// canvases top out at 100 x 27 cells, so screen sized ones get tiled
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

func MakeText(s)
  var t = ui.AddText(s)
  root.Add(t)
  t.anchor = "center_center"
  t.dock = "center_center"
  t.x = 0
  t.y = 0
  t.w = string.Size(s) + 2
  t.h = 1
  t.align = "center"
  t.color = "#000000"
  return t

// a light: an additive canvas fading out from the center, filled a bit each frame
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

func Build()

  W = screen.w
  H = screen.h

  cy = H / 2
  camY = world.floor - (cy + 7)
  CAM_HOLD = STOP_X - (cx - 6)
  T_STOP = T_WALK + math.CeilToInt((STOP_X - HOLE_X) / WALK_V)
  T_DROP = T_STOP + 6
  chestTop0 = math.FloorToInt(world.ropeY[CHEST_C]) + 2
  T_LAND = T_DROP + math.CeilToInt(math.Sqrt((22 - chestTop0) / 0.05))
  T_OPEN = T_LAND + 40
  T_ITEMS = T_OPEN + 40
  T_PICK = T_ITEMS + 60
  T_TOSS = T_PICK + 34
  T_GO = T_TOSS + 24
  T_HUNT = T_GO + math.CeilToInt((CAM_HOLD + W + 6 - 100) / WALK_V) + 50
  T_PAUSE = T_HUNT + math.CeilToInt((PAUSE_X - (CAM_HOLD - 10)) / HUNT_V)
  T_ON = T_PAUSE + 45
  T_SNOW = T_ON + 40
  // a bit after the Hunter's gone, scene 5 cuts in on a gust of snow
  T_END = T_ON + math.CeilToInt((CAM_HOLD + W + 12 - PAUSE_X) / HUNT_V) + 50

  root.dock = "top_left"

  attached = true

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// a light on its source, or off screen with zero width (the game draws every canvas wherever it is)
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

// passage cell -> screen
func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

// the haul line's row at a column
func RopeAt(x)
  var i = IMax(0, IMin(world.ropeY.Count() - 1, math.RoundToInt(x)))
  return math.FloorToInt(world.ropeY[i])

// where he is along the passage (his head's column)
func HeroX()
  ?f < T_WALK
    return HOLE_X
  ?f < T_STOP
    return HOLE_X + WALK_V * (f - T_WALK)
  ?f < T_LAND
    return STOP_X
  ?f < T_LAND + 8
    // startled back a step
    return STOP_X - (f - T_LAND) * 0.25
  ?f < T_LAND + 18
    return STOP_X - 2
  ?f < T_LAND + 28
    return STOP_X - 2 + (f - T_LAND - 18) * 0.2
  ?f < T_GO
    return STOP_X
  return STOP_X + WALK_V * (f - T_GO)

func HeroAt(hx, r, frame, left)
  var b = heroR
  ?left
    b = heroL
  Show(b, true)
  b.x = SX(hx) - 1
  ?left
    b.x = SX(hx) - 4
  b.y = SY(r) - 3
  b.frame = frame
  glowX = hx + 1
  glowY = r - 2

func ClimbAt(hx, top, frame)
  Show(climbR, true)
  climbR.x = SX(hx) - 1
  climbR.y = SY(top)
  climbR.frame = frame
  glowX = hx
  glowY = top + 1

// the Hunter facing right, by his head's cell, feet on row r. lit: 0 a dark shape, 1 his own colors
func HunterAt(hx, r, frame, lit)
  var a = hunterR
  ?lit < 1
    a = hunterS
    a.color = color.Lerp(SHADOW, CLOAK, lit)
  Show(a, true)
  a.x = SX(hx) - 10
  a.y = SY(r) - 3
  a.frame = frame

func HunterX()
  ?f < T_PAUSE
    return CAM_HOLD - 10 + HUNT_V * (f - T_HUNT)
  ?f < T_ON
    return PAUSE_X
  return PAUSE_X + HUNT_V * (f - T_ON)


// ---------------------------------------------------------------- him

func Him()
  ?f < T_HANDS
    return
  var x = HeroX()
  var fl = world.floor
  ?f < T_HANDS + 14
    // hands come up over the edge, legs going below it
    ClimbAt(x, fl, 1 + ((f - T_HANDS) / 4) % 2 * 2)
  :?f < T_HANDS + 30
    // straining up until his head clears the edge
    ClimbAt(x, fl, 5 + ((f - T_HANDS) / 3) % 2)
  :?f < T_HANDS + 40
    ClimbAt(x, fl - 1, 2 + ((f - T_HANDS) / 3) % 2 * 2)
  :?f < T_HANDS + 50
    HeroAt(x, fl, 11, false)
  :?f < T_HANDS + 62
    // getting his breath back
    HeroAt(x, fl, 9, false)
  :?f < T_HANDS + 74
    HeroAt(x, fl, 4, true)
  :?f < T_WALK
    HeroAt(x, fl, 4, false)
  :?f < T_STOP
    HeroAt(x, fl, (f / 6) % 4, false)
  :?f < T_LAND
    HeroAt(x, fl, 4, false)
  :?f < T_LAND + 8
    // it lands right in front of him and he jumps back
    HeroAt(x, fl - 1, 5, false)
  :?f < T_LAND + 18
    HeroAt(x, fl, 4, false)
  :?f < T_LAND + 28
    HeroAt(x, fl, (f / 5) % 4, false)
  :?f < T_LAND + 34
    HeroAt(x, fl, 9, false)
  :?f < T_PICK
    // kneeling at it: the lid, the gear
    HeroAt(x, fl, 11, false)
  :?f < T_GO
    HeroAt(x, fl, 4, false)
  :
    HeroAt(x, fl, (f / 6) % 4, false)


// ---------------------------------------------------------------- the line, the chest, the gear

func Line()
  // a crate going down the line over his head, and later another
  var x = 140 - TROLLEY_V * f
  ?f > T_HUNT - 200
    x = 196 - TROLLEY_V * (f - T_HUNT + 200)
  Show(crate, x > -10 & x < 196)
  crate.x = SX(x) - 2
  crate.y = SY(RopeAt(x))
  // the Giant chest riding the line down to him, then the hook slips
  var tx = CHEST_C + TROLLEY_V * (T_DROP - f)
  Show(trolley, tx > -4 & tx < 196)
  trolley.x = SX(tx) - 1
  trolley.y = SY(RopeAt(tx))
  ?f >= T_DROP & f < T_DROP + 6
    // the hook jerks up as the weight comes off it
    trolley.y = trolley.y - 1
  Show(chest, true)
  chest.frame = 0
  ?f < T_DROP
    chest.x = SX(tx) - 2
    chest.y = SY(RopeAt(tx) + 2)
  :
    chest.x = SX(CHEST_C) - 2
    var t = f - T_DROP
    chest.y = SY(IMin(22, chestTop0 + 0.05 * t * t))
  ?f >= T_OPEN & f < T_OPEN + 8
    chest.frame = 1
  :?f >= T_OPEN + 8 & f < T_ITEMS + 10
    chest.frame = 2
  :?f >= T_ITEMS + 10
    chest.frame = 3
  ?f = T_LAND
    play treasure_drop_giant
  :?f = T_OPEN
    play treasure_open
  :?f = T_OPEN + 10
    play treasure_item_show
  :?f = T_ITEMS
    play treasure_item_pop
  :?f = T_TOSS + 14
    play stone_throw_hit 140
  // dust where it lands, and the floor jumping
  var d = f - T_LAND
  Show(dust, d >= 0 & d < 24)
  dust.frame = IMax(0, d) / 4
  dust.x = SX(CHEST_C) - 10
  dust.y = SY(world.floor) - 4
  ?d >= 0 & d < 4 & opt.shake
    root.y = (d % 2) * 2 - 1
  // the heart rising out of it, like the game shows what's in a chest
  var hb = f - T_OPEN - 10
  Show(heart, hb >= 0 & f < T_ITEMS)
  heart.frame = 0
  ?hb < 10
    heart.frame = 1
  heart.x = SX(CHEST_C) - 2
  heart.y = SY(21) - IMin(3, IMax(0, hb) / 8)
  // a flash as the gear bursts out
  var bt = f - T_ITEMS
  Show(burst, bt >= 0 & bt < 8)
  burst.frame = 0
  ?bt >= 3
    burst.frame = 1
  burst.x = SX(CHEST_C) - 1
  burst.y = SY(20)
  Light(chestGlow, jobs.Count() = 0 & hb >= 0 & f < T_ITEMS + 30, SX(CHEST_C), SY(22))

// a dozen pieces of plain stone gear flying out, he picks up the shield and tosses it back
func Gear()
  var t = f - T_ITEMS
  for i = 0 .. 11
    var it = items[i]
    Show(it, t >= 0)
    var p = math.Clamp((t - (i % 4)) / 18.0, 0.0, 1.0)
    var x = CHEST_C + ITEM_DX[i] * p
    var y = 22 + (ITEM_Y[i] - 22) * p - 5 * math.Sin(p * 3.14159)
    ?i = 1 & f >= T_PICK
      // in his hand, held up to look at, then thrown back
      x = HeroX() - 3
      y = world.floor - 1
      ?f >= T_TOSS
        var q = math.Clamp((f - T_TOSS) / 14.0, 0.0, 1.0)
        x = HeroX() - 3 + (CHEST_C - 5 - HeroX() + 3) * q
        y = world.floor - 1 + 2 * q - 4 * math.Sin(q * 3.14159)
    it.x = SX(x) - 1
    it.y = SY(y)
  // light catching a few of the pieces as they land
  for g = 0 .. 3
    var i = g * 3
    var gl = itemGlints[g]
    var lt = t - (i % 4) - 18
    Show(gl, lt >= 0 & lt < 8)
    gl.frame = 0
    ?lt >= 4
      gl.frame = 2
    gl.x = SX(CHEST_C + ITEM_DX[i])
    gl.y = SY(ITEM_Y[i]) - 1


// ---------------------------------------------------------------- the Hunter

// hooded, barbed whip dragging and scoring the rock. stops at the pile, then goes after the Pusher
func Hunter()
  ShowAll(marks, false)
  ?f = T_HUNT - 40
    // heard before he's seen: the barbs dragging over the rock
    play blade_drag
  ?f < T_HUNT
    return
  var x = HunterX()
  var hx = math.RoundToInt(x)
  var standing = f >= T_PAUSE & f < T_ON
  var frame = (f / 6) % 4
  ?standing
    frame = 4
  // in from the dark at the screen's edge, into the mushrooms' light
  var lit = math.Clamp((SX(hx) - 4) / 16.0, 0.0, 1.0)
  HunterAt(x, world.floor, frame, lit)
  // the whip from his back hand, barbs lined up with the marks they leave
  var hand = hx - 10 + spr.hunterHandX
  var p = (CAM_HOLD - hand + spr.dragOX + 1000) % 2
  var k = f % spr.dragN
  ?standing
    k = 0
  Show(drag, true)
  drag.frame = p * spr.dragN + k
  drag.x = SX(hand) - spr.dragOX
  drag.y = SY(world.floor - 3 + spr.hunterHandY) - spr.dragOY
  drag.color = color.Lerp(SHADOW, CLOAK, lit)
  // a barb throws a spark now and then, the only light on him till he reaches the mushrooms
  var walking = !standing & f < T_END
  var lightOn = false
  for i = 0 .. 1
    var sp = sparks[i]
    var st = (f - T_HUNT + i * 7) % 13
    var n = (f - T_HUNT + i * 7) / 13
    var on = walking & st < 3
    Show(sp, on)
    sp.frame = 1
    sp.x = SX(hand - spr.dragReach + 2 + (n * 7 + i * 4) % 9) - 1
    sp.y = SY(world.floor) - 2
    ?on & st < 2 & !lightOn
      lightOn = true
      Light(sparkGlow, jobs.Count() = 0, sp.x + 1, sp.y + 1)
  // the marks, laid as the whip passes, fresh ones brighter
  var tail = hand - spr.dragReach
  for i = 0 .. MARK_N - 1
    var m = marks[i]
    var mx = CAM_HOLD - 16 + i * 2
    Show(m, mx <= tail + 4)
    m.x = SX(mx)
    m.y = SY(world.floor)
    // gray like the rock, only the whip itself is cyan
    m.color = color.Lerp(SHADOW, "#3a3a3a", lit)
    ?tail < mx + 20
      m.color = color.Lerp(SHADOW, "#525252", lit)


// ---------------------------------------------------------------- life

func Life(lit)
  // the mushrooms' light
  for i = 0 .. mushGlow.Count() - 1
    Light(mushGlow[i], lit, SX(world.mushX[i]), SY(world.mushY[i]))
  // spores drifting up off them
  for i = 0 .. 9
    var s = spores[i]
    var k = i % world.mushX.Count()
    var age = (f + i * 47) % 150
    Show(s, age < 120)
    s.x = SX(world.mushX[k] + (i % 3) - 1 + math.Sin(age * 0.05 + i) * 1.5)
    s.y = SY(world.mushY[k] - age * 0.07)
    s.frame = 0
  // after they've gone, a cold draft with snow in it comes down the passage
  for i = 0 .. 7
    var s = snow[i]
    var t0 = T_SNOW + i * 11
    var a = f - t0
    var sx = W + 1 - a * (0.55 + (i % 3) * 0.12)
    Show(s, a >= 0 & sx > -1)

// ---------------------------------------------------------------- the film calls these

// a scene built ahead waits off screen and the game still draws its canvases, so zero their width till it's on
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so the lights are ready, parked off to the side (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

// take its picture off the screen (another scene is playing)
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start, for watching again
func Reset()
  f = 0
  over = false
