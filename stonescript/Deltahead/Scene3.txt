func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

func MakeCanvas(x, y, w, h, blend)
  var c = ui.AddCanvas()
  parkable.Add(c)
  root.Add(c)
  Place(c, x, y)
  c.w = w
  c.h = h
  c.blend = blend
  c.SetFG("#000000")
  c.SetBG("#000000")
  return c

// canvases max out at 100x27, so screen-sized ones are tiled
func Tiles(blend, y0, rows)
  var list = []
  for j = 0 .. (rows - 1) / 27
    for i = 0 .. (W - 1) / 100
      var c = MakeCanvas(i * 100, y0 + j * 27, IMin(100, W - i * 100), IMin(27, rows - j * 27), blend)
      list.Add(c)
  return list

func Solid(list)
  for c : list
    c.Set(" ")

// a light: additive canvas fading out from the center, filled a few hundred cells a frame.
// canvases get drawn even when hidden, so a light that's off waits off screen (see Light)
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// ---------------------------------------------------------------- helpers

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func IsFast(ku)
  return ku[2] = 1

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// on its source this frame, or off screen with no width (the game draws every canvas, hidden or not)
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

// cavern cell -> screen
func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

// his feet's row, t frames into the fall: speeding up, then falling flat out
func FallY(t)
  ?t < 43
    return -2 + 0.015 * t * t
  return 25.735 + 1.29 * (t - 43)

// where he is across the cavern (his head's column)
func HeroX()
  ?f < T_WADE
    return world.landX
  ?f < T_EDGE
    return world.landX + WADE_V * (f - T_WADE)
  ?f < T_SHORE
    return SHORE_X - 3 + IMin(4, (f - T_EDGE) / 10)
  ?f < T_CLEFT
    return SHORE_X + 1 + WALK_V * (f - T_SHORE)
  return START_X

// him with his sword, by his head's cell, feet on row r
func HeroAt(hx, r, frame, left)
  var b = heroR
  ?left
    b = heroL
  Show(b, true)
  b.x = SX(hx) - 1
  ?left
    b.x = SX(hx) - 4
  b.y = SY(r) - 3
  b.frame = frame
  glowX = hx + 1
  glowY = r - 2
  himX = b.x
  himY = b.y
  himW = 6

// him climbing, by his head's cell, his top row on row top
func ClimbAt(hx, top, frame, left)
  var b = climbR
  ?left
    b = climbL
  Show(b, true)
  b.x = SX(hx) - 1
  ?left
    b.x = SX(hx) - 2
  b.y = SY(top)
  b.frame = frame
  glowX = hx
  glowY = top + 1
  himX = b.x
  himY = b.y
  himW = 3


// ---------------------------------------------------------------- him

func Him()
  ?f < T_DROP
    return
  ?f < T_LAND
    // falling, tumbling
    var t = f - T_DROP
    feetNow = FallY(t)
    HeroAt(world.landX + (t / 8) % 2, feetNow, 5, false)
    return
  feetNow = world.floor
  var b = f - T_LAND
  ?b < 8
    // under
    return
  :?b < 40
    // up to his neck in it
    HeroAt(world.landX, world.floor, 11, false)
  :?b < 48
    HeroAt(world.landX, world.floor, 9, false)
  :?b < 72
    HeroAt(world.landX, world.floor, 4, false)
  :?b < 94
    // a look back the way he came
    HeroAt(world.landX, world.floor, 4, true)
  :?f < T_WADE
    HeroAt(world.landX, world.floor, 4, false)
  :?f < T_EDGE
    // after it, wading, the water at his waist
    HeroAt(HeroX(), world.floor, (f / 6) % 4, false)
  :?f < T_SHORE
    Out(f - T_EDGE)
  :?f < T_CLEFT
    HeroAt(HeroX(), world.floor, (f / 6) % 4, false)
  :?f < T_CLIMB
    HeroAt(START_X, world.floor, 4, false)
  :
    Climb(f - T_CLIMB)

// climbing out at the pool's edge: hands on the bank, a knee up, then on his feet, dripping
func Out(u)
  ?u < 8
    HeroAt(SHORE_X - 3, world.floor, 4, false)
  :?u < 18
    HeroAt(SHORE_X - 2, world.floor, 9, false)
  :?u < 30
    HeroAt(SHORE_X - 1, world.floor - 1, 5, false)
  :?u < 40
    HeroAt(SHORE_X, world.floor, 11, false)
  :
    HeroAt(SHORE_X + 1, world.floor, 4, false)

// which ledge he's heading for t frames into the climb, how far into it (u), and whether
// he's found his rhythm (1) or not (0). on the slip ledge u runs 40 to 40 + SLIP_T through the slip
func ClimbKU(t)
  var slow = N_SLOW * CLIMB_T + SLIP_T
  ?t >= slow
    return [N_SLOW + (t - slow) / FAST_T, (t - slow) % FAST_T, 1]
  var k = t / CLIMB_T
  var u = t % CLIMB_T
  ?t >= (SLIP_K + 1) * CLIMB_T + SLIP_T
    k = (t - SLIP_T) / CLIMB_T
    u = (t - SLIP_T) % CLIMB_T
  :?t >= SLIP_K * CLIMB_T
    k = SLIP_K
    u = t - SLIP_K * CLIMB_T
    ?u >= 40 + SLIP_T
      u = u - SLIP_T
  return [k, u, 0]

func Slipping(ku)
  return !IsFast(ku) & ku[0] = SLIP_K & ku[1] >= 40 & ku[1] < 40 + SLIP_T

// where the camera sits for ledge k (the floor before the first)
func CamFor(k)
  ?k < 0
    return CAM_FLOOR
  return IMax(CAM_TOP, IMin(CAM_FLOOR, world.ledgeY[k] - (cy + 7)))

// up the chimney a ledge at a time: crouch, jump, dangle, strain up, forearms, knee, breathe, stand.
// he slips on the second ledge, and after the third he goes quicker
func Climb(t)
  var ku = ClimbKU(t)
  var k = ku[0]
  var u = ku[1]
  ?Slipping(ku)
    // slipped: back to hanging, then up again
    ?u < 48
      ClimbAt(world.ledgeX[k], world.ledgeY[k] + 1, 1 + (u / 8) % 2 * 2, world.ledgeX[k] < world.ledgeX[k - 1])
      feetNow = world.ledgeY[k] + 4
    :
      ClimbAt(world.ledgeX[k], world.ledgeY[k] + 1, 5 + (u / 6) % 2, world.ledgeX[k] < world.ledgeX[k - 1])
      feetNow = world.ledgeY[k] + 3
    return
  ?k >= world.ledgeX.Count()
    return
  ?IsFast(ku)
    ClimbFast(k, u)
    return
  var x0 = START_X
  var r0 = world.floor
  ?k > 0
    x0 = world.ledgeX[k - 1]
    r0 = world.ledgeY[k - 1]
  var x1 = world.ledgeX[k]
  var r1 = world.ledgeY[k]
  var left = x1 < x0
  ?u < 8
    HeroAt(x0, r0, 11, left)
    feetNow = r0
  :?u < 14
    var p = (u - 7) / 6.0
    ClimbAt(x0 + (x1 - x0) * p, r0 - 2 + (r1 - r0 + 4) * p, 0, left)
    feetNow = r0 + (r1 + 4 - r0) * p
  :?u < 28
    // dangling, legs going
    ClimbAt(x1, r1 + 1, 1 + (u / 8) % 2 * 2, left)
    feetNow = r1 + 4
  :?u < 40
    // straining up to the edge
    ClimbAt(x1, r1 + 1, 5 + (u / 6) % 2, left)
    feetNow = r1 + 3
  :?u < 48
    // up on his forearms, still kicking
    ClimbAt(x1, r1, 2 + (u / 6) % 2 * 2, left)
    feetNow = r1 + 2
  :?u < 56
    HeroAt(x1, r1, 11, left)
    feetNow = r1
  :?u < 64
    // bent over, getting his breath back
    HeroAt(x1, r1, 9, left)
    feetNow = r1
  :
    HeroAt(x1, r1, 4, left)
    feetNow = r1

// a ledge once he's got the hang of it
func ClimbFast(k, u)
  var x0 = world.ledgeX[k - 1]
  var r0 = world.ledgeY[k - 1]
  var x1 = world.ledgeX[k]
  var r1 = world.ledgeY[k]
  var left = x1 < x0
  ?u < 5
    HeroAt(x0, r0, 11, left)
  :?u < 9
    var p = (u - 4) / 5.0
    ClimbAt(x0 + (x1 - x0) * p, r0 - 2 + (r1 - r0 + 4) * p, 0, left)
  :?u < 15
    ClimbAt(x1, r1 + 1, 1 + (u / 6) % 2 * 2, left)
  :?u < 23
    ClimbAt(x1, r1 + 1, 5 + (u / 6) % 2, left)
  :?u < 29
    ClimbAt(x1, r1, 2 + (u / 6) % 2 * 2, left)
  :?u < 35
    HeroAt(x1, r1, 11, left)
  :
    HeroAt(x1, r1, 4, left)
  feetNow = r1


// ---------------------------------------------------------------- the water

func Water()
  ?f < T_LAND
    return
  var b = f - T_LAND
  // the splash, and the rings it leaves
  Show(splash, b < 20)
  splash.frame = b / 4
  splash.x = SX(world.landX) - 4
  splash.y = SY(world.floor) - 4
  for i = 0 .. 1
    var a = b - 4 - i * 9
    Show(rings[i], a >= 0 & a < 24)
    rings[i].frame = IMax(0, a) / 4
    rings[i].x = SX(world.landX) - 4
    rings[i].y = SY(world.floor - 1 + i * 2)
  ?b = 0
    play waterfall_splash
  Light(splashGlow, jobs.Count() = 0 & b < 12, SX(world.landX), SY(world.floor))
  // water running off him as he gets up, and again on the bank
  for i = 0 .. 1
    var d = b - 48 - i * 9
    var dx = world.landX + 1 - i * 2
    var e = f - T_SHORE + 12 - i * 7
    ?e >= 0 & e < 10
      d = e
      dx = SHORE_X + 2 - i * 2
    Show(drops[i], d >= 0 & d < 10)
    drops[i].frame = 2
    drops[i].x = SX(dx)
    drops[i].y = SY(world.floor - 1) + IMin(1, d / 5)
  // the crystal's light catching on the water here and there
  for i = 0 .. 5
    var s = shimmers[i]
    var per = 31 + i * 6
    var ph = (f + i * 17) % per
    var cyc = (f + i * 17) / per
    var sx = world.poolX0 + 4 + (cyc * 29 + i * 13) % (world.poolX1 - world.poolX0 - 8)
    var near = math.Abs(sx - HeroX()) < 4 & f < T_SHORE
    Show(s, ph < 12 & !near)
    s.frame = 0
    ?ph >= 6
      s.frame = 2
    s.x = SX(sx)
    s.y = SY(world.floor - 1 + (i % 2) * 2)
  // his wake as he wades
  for i = 0 .. 1
    var a = (f - T_WADE + i * 11) % 22
    var at = f - a
    Show(wakes[i], at >= T_WADE & at < T_EDGE)
    wakes[i].frame = a / 4
    wakes[i].x = SX(world.landX + WADE_V * (at - T_WADE)) - 4
    wakes[i].y = SY(world.floor)


// ---------------------------------------------------------------- the Ki

func Ki()
  // Ki rising everywhere, drifting toward the chimney. motes wrap round the screen so
  // there's always as many in view, far ones move less with the camera
  for i = 0 .. MOTE_N - 1
    var m = motes[i]
    var d = i % 3
    var k = DEPTH_K[d]
    var x0 = (i * 53 + 7) % 101
    var y0 = (i * 37 + 11) % 61
    m.x = math.RoundToInt(Wrap(x0 + 1.5 * math.Sin(f * 0.04 + i) + f * 0.02 - camXf * k, W + 4)) - 2
    m.y = math.RoundToInt(Wrap(y0 - f * DEPTH_V[d] - camYf * k, H + 4)) - 2
    m.frame = 0
    Show(m, !OnHim(m.x, m.y))
    ?d = 2 & (f / 6 + i * 5) % 17 = 0
      m.frame = 3
    :?d = 1 & (f / 9 + i) % 11 = 0
      m.frame = 1
    :?d = 0
      m.frame = 2
  // thick and quick up the chimney
  for i = 0 .. 7
    var m = cleftMotes[i]
    var x = SX(world.cleftX0 + 2 + (i * 5) % 15 + math.Sin(f * 0.06 + i * 2) * 1.2)
    m.x = x
    m.y = math.RoundToInt(Wrap((i * 23) % 47 - f * 0.32 - camYf, H + 4)) - 2
    Show(m, x >= 0 & x < W & !OnHim(m.x, m.y))
    m.frame = (f / 4 + i) % 2
  // out of the pool, each leaving a ring where it leaves the water
  for p = 0 .. 2
    var age = (f + p * 43) % 130
    var cyc = (f + p * 43) / 130
    var bx = world.poolX0 + 6 + (cyc * 17 + p * 23) % (world.poolX1 - world.poolX0 - 12)
    var m = poolMotes[p]
    m.x = SX(bx + math.Sin(age * 0.07 + p) * 1.2)
    m.y = SY(world.floor - 1 - age * 0.12)
    Show(m, !OnHim(m.x, m.y))
    m.frame = 0
    Show(ripples[p], age < 24)
    ripples[p].frame = age / 4
    ripples[p].x = SX(bx) - 4
    ripples[p].y = SY(world.floor - 1)
  Lead()

// whether a cell is in his box (the Ki goes behind him there so it doesn't flicker on him)
func OnHim(x, y)
  return x >= himX & x < himX + himW & y >= himY & y < himY + 4

// the mote he follows: up out of the water, past his face, then at eye level to the chimney and up it
func Lead()
  var b = f - T_LAND - 100
  var on = b >= 0 & f < T_CLIMB + 60
  Show(lead, on)
  ?!on
    leadGlow.x = OFF
    return
  var x = world.landX + 2
  var y = world.floor - b * 0.08
  ?b >= 30
    x = world.landX + 2 + IMin(10, b - 30) * 0.1
    y = world.floor - 2.4 + 0.5 * math.Sin(b * 0.08)
  ?f >= T_WADE
    // off his rounded cell like the camera or it wobbles, and clear of his blade or it flickers
    x = math.RoundToInt(HeroX()) + 4 + math.Sin(f * 0.05)
    y = world.floor - 2.6 + 0.6 * math.Sin(f * 0.07)
  ?f >= T_CLEFT
    x = START_X + 3 + (f - T_CLEFT) * 0.06
    y = world.floor - 2.6 - (f - T_CLEFT) * 0.15
  lead.x = SX(x)
  lead.y = SY(y)
  lead.frame = 0
  ?(f / 5) % 7 = 0
    lead.frame = 3
  Light(leadGlow, jobs.Count() = 0, SX(x), SY(y))
  // the ring it leaves as it comes out of the water
  ?b < 24
    Show(rings[0], true)
    rings[0].frame = b / 4
    rings[0].x = SX(world.landX + 2) - 4
    rings[0].y = SY(world.floor)

func Crystal(lit)
  var tip = world.crystalTip
  var x = world.crystalX
  Light(deepGlow, lit & f < T_LAND, SX(x), SY(tip - 9))
  Light(glowLow, lit, SX(x - 4), SY(76))
  Light(glowHigh, lit, SX(x - 2), SY(54))
  Light(tipGlow, lit, SX(x), SY(tip + 2))
  Light(poolGlow, lit, SX(54), SY(world.floor))
  // the Ki surging up inside it, and a swell of light going up with it
  for i = 0 .. 2
    var s = surges[i]
    var age = (f + i * 37) % 110
    var y = world.backEdge - 3 - age * 0.5
    var sx = SURGE_X[i]
    Show(s, sx > x - (y - tip))
    s.x = SX(sx)
    s.y = SY(y)
  var sy = world.backEdge - (f % 150) * 0.45
  Light(surgeGlow, lit & sy > tip - 4, SX(x - 4), SY(sy))
  for i = 0 .. 1
    Light(cleftGlow[i], lit, SX((world.cleftX0 + world.cleftX1) / 2), SY(world.floor - 12 - i * 26))
  // light catching the small crystals
  for i = 0 .. glints.Count() - 1
    var g = glints[i]
    var p = (f / 4 + i * 9) % 29
    g.frame = 1
    ?p = 1
      g.frame = 2
    :?p < 3
      g.frame = 0
    g.x = SX(world.glintX[i])
    g.y = SY(world.glintY[i])

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  Show(heroR, false)
  Show(heroL, false)
  Show(climbR, false)
  Show(climbL, false)
  Show(splash, false)
  ShowAll(rings, false)
  ShowAll(wakes, false)
  ShowAll(drops, false)

  // camera: still over the dark, down with his fall, along with him, then up the chimney
  camXf = math.Max(0.0, HeroX() - (cx - 6))
  camX = IMax(0, math.RoundToInt(HeroX()) - (cx - 6))
  ?f < T_DROP
    camYf = 0
  :?f < T_LAND
    camYf = IMax(0, IMin(CAM_FLOOR, math.RoundToInt(FallY(f - T_DROP)) - (cy - 2)))
  :?f < T_CLIMB
    camYf = CAM_FLOOR
  :
    // up a ledge at a time, only while he hangs, so it never moves while he does (his feet drop
    // on every jump, following them bobs). once he's in his rhythm it stays put and he climbs out
    var ku = ClimbKU(f - T_CLIMB)
    var k = ku[0]
    var p = math.Clamp((ku[1] - 14) / 26.0, 0.0, 1.0)
    ?IsFast(ku)
      k = N_SLOW - 1
      p = 1.0
    camYf = CamFor(k - 1) + (CamFor(k) - CamFor(k - 1)) * p
  camY = math.RoundToInt(camYf)

  pFar.x = -math.RoundToInt(camX * 0.6)
  pFar.y = -math.RoundToInt(camY * 0.6)
  pMidL.x = SX(0)
  pMidL.y = SY(0)
  pMidR.x = SX(world.midSplit)
  pMidR.y = SY(0)
  pPool.x = SX(0)
  pPool.y = SY(0)
  pFront.x = -math.RoundToInt(camX * 1.4)
  pFront.y = SY(0)

  var lit = jobs.Count() = 0
  himW = 0
  Him()
  Water()
  Ki()
  Crystal(lit)
  Light(himGlow, lit & f >= T_SHORE, SX(glowX), SY(glowY))
  Sfx()

  // the words, then the dark coming up, and going down at the end
  card1.color = Fade(T_CARD1, T_CARD1 + 25, T_CARDS_OUT, T_CARDS_OUT + 25)
  card2.color = Fade(T_CARD2, T_CARD2 + 25, T_CARDS_OUT, T_CARDS_OUT + 25)
  Show(card1, f < T_CARDS_OUT + 26)
  banner.Update(f)
  Show(card2, f < T_CARDS_OUT + 26)
  // no fade: straight in from scene 2's fall, cuts to scene 4 on his hands coming out of the hole
  SetShade(GRADE)
  ?f = 0
    ambient.Stop()
    ambient.Add("ambient_cave")
  :?f = T_LAND + 20
    ambient.Add("ambient_temple")


// ---------------------------------------------------------------- the film calls these

// a scene built ahead waits off to the side, but the game still draws its canvases,
// so they get no width till it's on
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built near the end of the scene before so its lights are ready, then waits off to
// the side (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // nothing else on the frame it's built in, that's heavy enough
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

// take its picture off the screen (another scene is playing)
func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

// back to the start, for watching again
func Reset()
  f = 0
  over = false
  camYf = 0
