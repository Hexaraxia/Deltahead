func Place(c, x, y)
  c.anchor = "top_left"
  c.dock = "top_left"
  c.x = x
  c.y = y

func MakeAnim(layers, x, y)
  var a = ui.AddAnim(layers[0][1])
  root.Add(a)
  Place(a, x, y)
  a.color = layers[0][0]
  ?layers.Count() > 1
    for i = 1 .. layers.Count() - 1
      var l = a.AddLayer(layers[i][1])
      l.color = layers[i][0]
  return a

// Canvases top out at 100 x 27 cells, so screen-sized ones are tiled.

      list.Add(c)
  return list

// a light: additive canvas fading out from the center, filled a bit each frame
func QueueHalo(w, h, r, g, b, bgk)
  var c = MakeCanvas(OFF, 0, w, h, "add")
  jobs.Add([c, w, h, r, g, b, bgk, 0])
  allLights.Add(c)
  lightW.Add(w)
  c.w = 0
  return c

func RunJobs()
  look.Run()
  ki.Run()
  ?jobs.Count() = 0
    return
  var j = jobs[0]
  var total = j[1] * j[2]
  var last = IMin(total - 1, j[7] + 150)
  var hx = (j[1] - 1) / 2.0
  var hy = (j[2] - 1) / 2.0
  for i = j[7] .. last
    var x = i % j[1]
    var y = i / j[1]
    var dx = (x - hx) / (hx + 0.5)
    var dy = (y - hy) / (hy + 0.5)
    var k = 1.0 - math.Sqrt(dx * dx + dy * dy)
    ?k > 0
      k = k * k
      j[0].SetFG(x, y, color.FromRGB(math.RoundToInt(j[3] * k), math.RoundToInt(j[4] * k), math.RoundToInt(j[5] * k)))
      j[0].SetBG(x, y, color.FromRGB(math.RoundToInt(j[3] * k * j[6]), math.RoundToInt(j[4] * k * j[6]), math.RoundToInt(j[5] * k * j[6])))
  j[7] = last + 1
  ?j[7] >= total
    jobs.RemoveAt(0)

// the real frame story moment S comes on
func RealAt(S)
  var r = S
  for k = 0 .. MARKS.Count() / 2 - 1
    var s = MARKS[k * 2]
    var n = MARKS[k * 2 + 1]
    ?s < S
      ?n > 0
        r = r + n
      :
        r = r + IMin(-n, S - s)
  return r

// story time at real frame t
func Story(t)
  var cum = 0
  for k = 0 .. MARKS.Count() / 2 - 1
    var s = MARKS[k * 2]
    var n = MARKS[k * 2 + 1]
    ?t < s + cum
      return (t - cum) * 1.0
    ?n > 0
      ?t < s + cum + n
        return s * 1.0
      cum = cum + n
    :
      ?t < s + cum - 2 * n
        return s + (t - s - cum) * 0.5
      cum = cum - n
  return (t - cum) * 1.0

// the game's rainbow at brightness b (out of 255), by hand since #rain can't be dimmed.
// same 2.5 s cycle, lightened halfway to white
func Rainbow(b)
  var h = 6.0 - (f % 75) * 0.08
  var k = math.FloorToInt(h) % 6
  var t = h - math.FloorToInt(h)
  var r = 1.0
  var g = t
  var bl = 0.0
  ?k = 1
    r = 1.0 - t
    g = 1.0
  :?k = 2
    r = 0.0
    g = 1.0
    bl = t
  :?k = 3
    r = 0.0
    g = 1.0 - t
    bl = 1.0
  :?k = 4
    r = t
    g = 0.0
    bl = 1.0
  :?k = 5
    r = 1.0
    g = 0.0
    bl = 1.0 - t
  return color.FromRGB(math.RoundToInt((0.5 + r * 0.5) * b), math.RoundToInt((0.5 + g * 0.5) * b), math.RoundToInt((0.5 + bl * 0.5) * b))

func IMin(a, b)
  ?a < b
    return a
  return b

func IMax(a, b)
  ?a > b
    return a
  return b

func Prog(a, b)
  return math.Clamp((f - a) * 1.0 / (b - a), 0.0, 1.0)

func Ramp(u, n)
  return math.Clamp(u * 1.0 / n, 0.0, 1.0)

func Out(st, a, b)
  var u = Ramp(st - a, b - a)
  return 1.0 - (1.0 - u) * (1.0 - u)

func Walk(t, t0, t1, x0, x1)
  return x0 + (x1 - x0) * Ramp(t - t0, t1 - t0)

func Fade(a, b, c, d)
  return color.Lerp("#000000", "#e7e3da", Prog(a, b) - Prog(c, d))

func SetShade(col)
  ?!string.Equals(col, shadeCol)
    shadeCol = col
    for c : shade
      c.SetFG(col)
      c.SetBG(col)

func Wrap(v, m)
  return v - math.Floor(v / m) * m

func Show(c, on)
  c.visible = on

func ShowAll(list, on)
  for c : list
    c.visible = on

// off lights get no width too: the game draws every canvas cell, shown or not
func Light(c, on, x, y)
  ?on
    var w = lightW[allLights.IndexOf(c)]
    c.w = w
    c.x = x - w / 2
    c.y = y - c.h / 2
  :
    c.x = OFF
    c.w = 0

func SX(x)
  return math.RoundToInt(x - camX)

func SY(y)
  return math.RoundToInt(y - camY)

func Feet(x)
  return world.feet0 + (world.feet1 - world.feet0) * x * 1.0 / world.ww

func Row(x)
  return math.FloorToInt(Feet(math.RoundToInt(x)))

// which color the light is on
func Hue()
  return (f / 10) % 8

func Cross(sp, st, x)
  return sp < x & st >= x

func ShakeNow()
  var amp = 0
  ?!opt.shake
    return 0
  for k = 0 .. SHK.Count() / 3 - 1
    ?f >= SHK[k * 3] & f < SHK[k * 3] + SHK[k * 3 + 1]
      amp = IMax(amp, SHK[k * 3 + 2])
  return amp

func Puff(k, x, y, u)
  ?u >= 0 & u < 16
    var a = puffs[k]
    Show(a, true)
    a.x = SX(x) - 6
    a.y = SY(y) - 3
    a.frame = IMin(3, math.FloorToInt(u) / 4)

func Burst(k, x, y, u, col)
  ?u >= 0 & u < 12
    var a = bursts[k]
    Show(a, true)
    a.x = SX(x) - 4
    a.y = SY(y) - 2
    a.frame = IMin(2, math.FloorToInt(u) / 4)
    a.color = col

// a spark of every color, thrown off the ∆ and dying out
func Spark(k, x, y, u, dx, dy)
  ?u >= 0 & u < 14
    var a = motes[k]
    Show(a, true)
    a.x = SX(x + dx * u * 0.25)
    a.y = SY(y + dy * u * 0.18 + 0.02 * u * u)
    a.frame = IMin(2, math.FloorToInt(u) / 5)
    a.color = DELTA_C


// ---------------------------------------------------------------- him

func HimX(st)
  var x = 159.0
  for k = 0 .. WALKS.Count() / 3 - 1
    x = Walk(st, WALKS[k * 3], WALKS[k * 3 + 1], x, WALKS[k * 3 + 2])
  return x

// art.bodyR/L frames: 0-3 walk, 4 stand, 5 bent to the chest, 6 grip, 7 pull, 8 hold up,
// 9-12 walk with the sword, 13 stand with it, 14 leap, 15 crouch, 16-20 pick it up
func Facing(s)
  return s >= 260 & s < 615

func HimFrame(s)
  var fr = 4
  for k = 0 .. WALKS.Count() / 3 - 1
    ?s >= WALKS[k * 3] & s < WALKS[k * 3 + 1] & WALKS[k * 3 + 1] - WALKS[k * 3] > 2
      fr = (s / 5) % 4
  ?s >= 312 & s < 360
    fr = 5
  :?s >= T_GRIP & s < T_PULL
    fr = 6
  :?s >= T_PULL & s < T_RIP + 6
    fr = 7
  :?s >= T_RIP + 6 & s < T_RAISE
    fr = 4
  :?s >= T_RAISE & s < T_ON_HEAD
    fr = 8
  :?s >= T_SWORD & s < T_RUN
    // the game's pickup: bend, grab the hilt, up over his head
    fr = 16 + IMin(4, (s - T_SWORD) / 4)
  :?s >= T_RUN & s < T_LEAP
    fr = 9 + (s / 3) % 4
  :?s >= T_LEAP & s < T_LANDED
    fr = 14
  :?s >= T_LANDED & s < T_LANDED + 12
    fr = 15
  :?s >= T_LANDED + 12 & s < 716
    fr = 13
  :?s >= 716
    fr = 9 + (s / 5) % 4
  return fr

func Him(s, st)
  var hx = math.RoundToInt(HimX(st))
  var left = Facing(s)
  var fr = HimFrame(s)
  var b = bodyR
  var ox = 1
  var hxs = art.headXR
  ?left
    b = bodyL
    ox = 4
    hxs = art.headXL
  Show(bodyR, !left)
  Show(bodyL, left)
  var dy = 0
  ?s >= T_LEAP & s < T_LANDED
    dy = math.RoundToInt(-5.0 * math.Sin(Ramp(st - T_LEAP, T_LANDED - T_LEAP) * 3.1416))
  b.x = SX(hx) - ox
  b.y = SY(Row(hx)) - 4 + dy
  b.frame = fr
  // his new head, once it's on
  Show(head, s >= T_ON_HEAD)
  head.x = b.x + hxs[fr]
  head.y = b.y + art.headY[fr]
  head.color = DELTA_C
  ?s >= T_ON_HEAD & jobs.Count() = 0
    Light(headGlows[Hue()], true, head.x, head.y)
  // the ∆ in his hand: torn free, held low, held up, and set on
  ?s >= T_RIP & s < T_ON_HEAD
    Show(delta, true)
    var dx = b.x + 1
    var dyy = b.y + 3
    ?s < T_RIP + 12
      dx = b.x + 1
      dyy = b.y + 3
    :?s >= T_RAISE & s < T_ON_HEAD - 8
      dx = b.x + 2
      dyy = b.y + 1
    :?s >= T_ON_HEAD - 8
      dx = b.x + 3
      dyy = b.y + 1
    delta.x = dx
    delta.y = dyy
    delta.color = DELTA_C
    // the light goes with it
    ?jobs.Count() = 0
      Light(headGlows[Hue()], true, dx, dyy)

// falls with a rainbow trail, opens, sparks as he pulls the ∆ out, then drains to stone
func Chest(s, st)
  var r = Row(CHEST_X + 2)
  var y = SY(r) - 3
  ?s < T_LAND
    var u = Ramp(st - T_FALL, T_LAND - T_FALL)
    y = y - math.RoundToInt((1.0 - u * u) * (r - camY + 6))
  Show(chest, s >= T_FALL)
  chest.x = SX(CHEST_X)
  chest.y = y
  chest.frame = 0
  ?s >= T_OPEN & s < T_OPEN + 12
    chest.frame = 1
  :?s >= T_OPEN + 12 & s < T_RIP
    chest.frame = 2
  :?s >= T_RIP
    chest.frame = 3
  // rainbow, flaring as he pulls, then draining to stone
  chest.color = DELTA_C
  ?s >= T_PULL & s < T_RIP & (f / 2) % 2 = 0
    chest.color = Rainbow(176)
  ?s >= T_RIP
    var d = math.FloorToInt(st - T_RIP) / 5
    ?d < DRAIN.Count()
      chest.color = Rainbow(DRAIN[d])
    :
      chest.color = color.Lerp("#3c3c3c", STONE, Ramp(st - T_RIP - 35, 15))
  // the ∆ on its front, until it's torn out
  ?s >= T_FALL & s < T_RIP
    Show(delta, true)
    delta.x = SX(DELTA_X)
    delta.y = y + 2
    delta.color = DELTA_C
  :
    Show(delta, false)
  // its trail as it falls, and the light it gives off until it's stone
  for k = 0 .. 2
    ?s >= T_FALL & s < T_LAND
      var m = motes[k]
      Show(m, true)
      m.x = SX(DELTA_X)
      m.y = y - 2 - k * 2
      m.frame = 2 - k
      m.color = DELTA_C
  ?jobs.Count() = 0 & s >= T_FALL & s < T_RIP + 10
    Light(chestGlows[Hue()], true, SX(DELTA_X), y + 2)
  Puff(0, CHEST_X + 2, r, st - T_LAND)
  // as he pulls: sparks off it, more and more
  ?s >= T_GRIP & s < T_RIP
    var k2 = (s / 4) % 4
    var u2 = s % 4 * 3
    ?s >= T_PULL
      u2 = s % 3 * 4
    Spark(3 + k2, DELTA_X, r - 1, u2, (k2 % 2) * 2 - 1, -1)
  // torn out: rainbow burst and sparks
  Burst(0, DELTA_X, r - 1, st - T_RIP, DELTA_C)
  for k = 0 .. 3
    Spark(3 + k, DELTA_X, r - 1, st - T_RIP - k, (k % 2) * 2 - 1, (k / 2) * 2 - 1)
  // the last of it going up out of the stone
  var lm = st - T_RIP - 14
  ?lm >= 0 & lm < 36
    var m = motes[7]
    Show(m, true)
    m.x = SX(DELTA_X)
    m.y = SY(r - 2 - lm * 0.15)
    m.frame = IMin(2, math.FloorToInt(lm) / 12)
    m.color = color.Lerp("#ffffff", "#000000", Ramp(lm - 14, 22))

// the sword: up out of the chest with its name, then point first into the snow
func Item(s, st)
  var r = Row(ITEM_X)
  // behind him till the hilt's in his hand
  var a = itemB
  ?s >= T_SWORD
    a = item
  Show(itemB, s >= T_ITEM & s < T_SWORD)
  Show(item, s >= T_SWORD & s < T_SWORD + 8)
  ?s < T_DROP
    a.x = SX(DELTA_X)
    a.y = SY(Row(CHEST_X + 2) - 4) - math.RoundToInt(4.0 * Out(st, T_ITEM, T_ITEM + 20))
  :
    var u = Ramp(st - T_DROP, 18)
    var y0 = Row(CHEST_X + 2) - 8
    a.x = SX(DELTA_X + (ITEM_X - DELTA_X) * u)
    a.y = SY(y0 + (r - 2 - y0) * u * u - 3 * math.Sin(u * 3.1416))
  a.color = BLADE
  Puff(1, ITEM_X, r, st - T_DROP - 18)
  // and snow off it as he pulls it out
  Puff(1, ITEM_X, r, st - T_SWORD - 8)
  // its name, like the game shows chest loot
  var on = s >= T_ITEM + 6 & s < T_DROP + 24
  Show(gotText, on)

func Frame()
  for c : allLights
    c.x = OFF
    c.w = 0
  var st = Story(f + SKIP)
  var sp = Story(f - 1 + SKIP)
  var s = math.FloorToInt(st)
  var lit = jobs.Count() = 0
  ShowAll(motes, false)
  ShowAll(puffs, false)
  ShowAll(bursts, false)

  // camera holds on the chest, then follows him over the crevasse
  var hx = math.RoundToInt(HimX(st))
  camX = IMax(150 - (cx - 10), hx - (cx + 1))
  camY = math.FloorToInt(Feet(60)) - (cy + 3)
  var amp = ShakeNow()
  ?amp > 0
    var sx = [1, 0, -1, 0]
    var sy = [0, 1, 0, -1]
    var ph = f % 4
    camX = camX + sx[ph] * amp
    ?amp > 1
      camY = camY + sy[ph]
  var sea = cy - 3
  pSky.x = -10 - math.RoundToInt(camX * 0.05)
  var af = (f / 6) % 8
  var cols = ["#136328", "#1f9a40", "#32da5a"]
  for k = 0 .. 2
    var a = aurora[k]
    a.x = -20 - math.RoundToInt(camX * 0.08)
    a.y = lb
    a.frame = af
    a.color = cols[k]
  pFar.x = -10 - math.RoundToInt(camX * 0.25)
  pFar.y = sea - hts.farSea
  summit.x = 14 - math.RoundToInt(camX * 0.3)
  summit.y = cy - 14
  pBack.x = -20 - math.RoundToInt(camX * 0.6)
  pBack.y = sea + 2 - math.RoundToInt((camY - (world.feet0 - (cy + 3))) * 0.6)
  // the mist drifts the way the camera goes, or the two would wobble it
  mist.x = -math.RoundToInt(Wrap(30 + f / 20.0 + camX * 0.7, 360))
  mist.y = sea + 3
  mist2.x = mist.x + 360
  mist2.y = sea + 3
  near.x = -10 - math.RoundToInt(camX * 1.5)
  near.y = H - 12
  for k = 0 .. 9
    var fl = flakes[k]
    fl.x = math.RoundToInt(Wrap((k * 53) % 97 - f * 0.25 + math.Sin(f * 0.05 + k) * 1.5, W + 4)) - 2
    fl.y = math.RoundToInt(Wrap((k * 37) % 33 + f * 0.18, H + 4)) - 2
    fl.frame = k % 3
    fl.color = "#9aa6ac"
  Light(auroraGlow, lit, cx, lb + 5)
  Light(moonGlow, lit, 9, 5)
  fieldA.x = SX(0)
  fieldA.y = SY(0)
  fieldB.x = SX(120)
  fieldB.y = SY(0)
  crevasse.x = SX(crev.crevX)
  crevasse.y = SY(0)
  snowLight.Update(glints, f, camX, camY, W, world.feet0, world.feet1, world.ww, 3, world.floorFront, [crev.crevX0 - 2, crev.crevX1 + 2], true)
  cloud.x = SX(f * 0.015)
  cloud.y = SY(Feet(60) + 7 + world.floorFront)

  Chest(s, st)
  Item(s, st)
  Him(s, st)
  // rainbow burst as the ∆ goes on
  Burst(1, HimX(st), Row(HimX(st)) - 2, st - T_ON_HEAD, DELTA_C)
  Puff(2, HimX(st), Row(HimX(st)), st - T_LANDED)

  card.color = Fade(T_CARD, T_CARD + 25, T_CARD_OUT, T_CARD_OUT + 25)
  Show(card, f < T_CARD_OUT + 26)
  // white flash as the ∆ goes on
  ?s >= T_ON_HEAD & s < T_ON_HEAD + 3 & opt.flashes
    SetShade("#ffffff")
  :
    SetShade(GRADE_H)
  // comes apart into Ki for scene 14
  ?f >= T_END - ki.OUT_T
    ki.At(veil, f - T_END)
    ?f = T_END - ki.OUT_T
      play elementalist_ice_blink 88
  :
    ki.Off(veil)
  Sfx(sp, st)


// ---------------------------------------------------------------- the film calls these

// the game draws every canvas cell even off screen, so they get no width while parked
func Park()
  parkW = []
  for c : parkable
    parkW.Add(c.w)
    c.w = 0

func Unpark()
  for i = 0 .. parkable.Count() - 1
    var c = parkable[i]
    c.w = parkW[i]

// built during the scene before so its lights are ready, parked off screen (ui.Remove would throw it away)
func Warm()
  ?!built
    ?totaltime < 1
      return
    Build()
    root.x = OFF
    parked = true
    Park()
    // building is heavy enough for one frame
    return
  RunJobs()

func Update()
  ?!built
    ?totaltime < 1
      return
    Build()
  ?parked
    root.x = 0
    parked = false
    Unpark()
  ?!attached
    ui.root.Add(root)
    attached = true
  RunJobs()
  Frame()
  ?f <= T_END
    f++
  :
    over = true

func Detach()
  ?attached
    ui.root.Remove(root)
    attached = false

func Reset()
  f = 0
  over = false
